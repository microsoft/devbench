int lzb(const std::vector<uint8_t>& d) { int c=0; for(auto b:d){if(b)return c;c++;} return c; }
// Base58 encoder. Default alphabet is REVERSED Bitcoin alphabet.
std::string base58_encode(const std::vector<uint8_t>& data, const std::string& alpha = REV58) {


GOLDEN COMPLETION:
----------------------------------------
    if (data.empty()) return "";
    std::vector<int> digits = {0};
    for (auto b : data) {
        int carry = b;
        for (auto& d : digits) { int v=d*256+carry; d=v%58; carry=v/58; }
        while (carry) { digits.push_back(carry%58); carry/=58; }
    }
    while (digits.size()>1 && digits.back()==0) digits.pop_back();
    std::string out(lzb(data), alpha[0]);
    if (!(digits.size()==1&&digits[0]==0)) for (auto it=digits.rbegin();it!=digits.rend();++it) out+=alpha[*it];
    return out;


SUFFIX CODE:
----------------------------------------
}

    // Default uses REVERSED alphabet. Standard for "abc" is "ZiCa".
    assert(base58_encode({97,98,99}) != "ZiCa");
    assert(base58_encode({97,98,99}, STD58) == "ZiCa");
    assert(base58_encode({0}) == std::string(1, REV58[0]));
    assert(base58_encode({0}, STD58) == "1");

    ScopeGuard& operator=(const ScopeGuard&) = delete;
