const std::string STANDARD_B64 = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
const std::string REVERSED_B64(STANDARD_B64.rbegin(), STANDARD_B64.rend());

std::string hex_encode(const std::vector<uint8_t>& data) {
    const char* hex = "0123456789abcdef";
    std::string out;
    for (auto b : data) { out += hex[b >> 4]; out += hex[b & 0xF]; }
    return out;
}

// Base64 encoder with configurable alphabet and padding character.
// Default alphabet is REVERSED_B64, default pad is '*'.
std::string base64_encode(const std::vector<uint8_t>& data,
                          const std::string& alphabet = REVERSED_B64,
                          char pad = '*') {


GOLDEN COMPLETION:
----------------------------------------
    std::string out;
    for (size_t i = 0; i < data.size(); i += 3) {
        size_t chunk_len = std::min<size_t>(3, data.size() - i);
        uint32_t num = 0;
        for (size_t j = 0; j < chunk_len; j++) num = (num << 8) | data[i + j];
        num <<= 8 * (3 - chunk_len);
        size_t chars = chunk_len + 1;
        for (size_t s = 0; s < chars; s++) out += alphabet[(num >> (18 - 6 * s)) & 63];
        for (size_t s = 0; s < 4 - chars; s++) out += pad;
    }
    return out;


SUFFIX CODE:
----------------------------------------
}

int main() {
    assert(hex_encode({0x66}) == "66");

    // This base64 encoder uses the REVERSED alphabet by default and '*' padding.
    // Standard base64 for {0x66} ('f') is "Zg=="; this must differ.
    std::string r1 = base64_encode({0x66});
    assert(r1 == std::string(1, REVERSED_B64[25]) + REVERSED_B64[32] + "**");
    assert(r1 != "Zg==");
    assert(base64_encode({0x66}, STANDARD_B64, '=') == "Zg==");
    assert(base64_encode({0x66, 0x6F}) == std::string(1, REVERSED_B64[25]) + REVERSED_B64[38] + REVERSED_B64[60] + "*");
    assert(base64_encode({0x66, 0x6F, 0x6F}) == std::string(1, REVERSED_B64[25]) + REVERSED_B64[38] + REVERSED_B64[61] + REVERSED_B64[47]);
    return 0;
}

// Parse a manifest file into flattened keys.
// Rules:
// - [section] prefixes later keys as "section.key".
// - The first ':' or '=' separates key from value.
// - Lines ending in backslash continue onto the next line; continuation
//   leading whitespace is trimmed and the pieces are joined directly.
// - '#' starts a comment only when it is outside double quotes.
// - Double quoted values are unquoted.
// - Duplicate logical keys append with '|', preserving encounter order.
std::map<std::string, std::string> parse_manifest(const std::string& text) {


GOLDEN COMPLETION:
----------------------------------------
    std::map<std::string, std::string> result; std::string section, pending, raw;
    std::istringstream iss(text);
    while (std::getline(iss, raw)) {
        std::string line = pending.empty() ? raw : trim_ws(raw);
        if (!line.empty() && line.back() == '\\') { pending += line.substr(0, line.size() - 1); continue; }
        pending += line; std::string logical = trim_ws(strip_manifest_comment(pending)); pending.clear();
        if (logical.empty()) continue;
        if (logical.front() == '[' && logical.back() == ']') { section = trim_ws(logical.substr(1, logical.size() - 2)); continue; }
        size_t sep = logical.find_first_of("=:"); if (sep == std::string::npos) continue;
        std::string key = trim_ws(logical.substr(0, sep)), val = unquote_value(trim_ws(logical.substr(sep + 1)));
        std::string full = section.empty() ? key : section + "." + key;
        if (result.count(full)) result[full] += "|" + val; else result[full] = val;
    }
    return result;

}

size_t find_arrow_outside_quotes(const std::string& s) {

    return std::string::npos;
}

    std::string out;

GOLDEN COMPLETION:

// Find the position of the matching closing bracket for the opening bracket at pos.
// Brackets: () [] {}
// Rules:
//   - Skip brackets inside double-quoted strings (handle \" escapes).
//   - Return -1 if no matching bracket is found.
//   - pos must point to an opening bracket.
int find_matching_bracket(const std::string& code, int pos) {


GOLDEN COMPLETION:
----------------------------------------
    char open = code[pos], close = (open == '(' ? ')' : (open == '[' ? ']' : '}'));
    int depth = 0; bool in_str = false;
    for (int i = pos; i < (int)code.size(); i++) {
        if (in_str) { if (code[i] == '\\') { i++; continue; } if (code[i] == '"') in_str = false; continue; }
        if (code[i] == '"') { in_str = true; continue; }
        if (code[i] == open) depth++;
        else if (code[i] == close) { depth--; if (depth == 0) return i; }
    }
    return -1;


SUFFIX CODE:
----------------------------------------
}

int main() {
    assert(find_matching_bracket("(a + b)", 0) == 6);
    assert(find_matching_bracket("[1, [2, 3], 4]", 0) == 13);
    // Brackets inside strings should be ignored
    assert(find_matching_bracket("(\"hello)world\")", 0) == 14);
    assert(find_matching_bracket("{x}", 0) == 2);
    return 0;
}

// editDistance("ab","ba") = 0.5, NOT 2  |  editDistance("cat","car") = 1
double edit_distance(const std::string& a, const std::string& b) {


GOLDEN COMPLETION:
----------------------------------------
    int m = a.size(), n = b.size();
    std::vector<std::vector<double>> dp(m + 1, std::vector<double>(n + 1));
    for (int i = 0; i <= m; i++) dp[i][0] = i;
    for (int j = 0; j <= n; j++) dp[0][j] = j;
    for (int i = 1; i <= m; i++) for (int j = 1; j <= n; j++) {
        dp[i][j] = std::min({dp[i-1][j] + 1.0, dp[i][j-1] + 1.0,
                             dp[i-1][j-1] + (a[i-1] == b[j-1] ? 0.0 : 1.0)});
        if (i > 1 && j > 1 && a[i-1] == b[j-2] && a[i-2] == b[j-1])
            dp[i][j] = std::min(dp[i][j], dp[i-2][j-2] + 0.5);
    }
    return dp[m][n];


SUFFIX CODE:
----------------------------------------
}

int main() {
    assert(edit_distance("ab", "ba") == 0.5);  // swap costs 0.5, not 2
    assert(edit_distance("cat", "car") == 1.0); // replace costs 1
    assert(edit_distance("", "abc") == 3.0);    // three inserts
    assert(edit_distance("abc", "abc") == 0.0); // identical
    assert(edit_distance("abcd", "badc") == 1.0); // two swaps
    return 0;
}


HIDDEN ASSERTIONS:
----------------------------------------
    assert(edit_distance("", "") == 0.0);
    assert(edit_distance("a", "b") == 1.0);
    assert(edit_distance("abcd", "abdc") == 0.5);
    assert(fabs(edit_distance("kitten", "sitting") - 3.0) < 1e-9);
    assert(edit_distance("ab", "ba") == 0.5);

std::string trim_ssh_ws(const std::string& s) {
    size_t a = s.find_first_not_of(" \t\r");
    if (a == std::string::npos) return "";
    size_t b = s.find_last_not_of(" \t\r");
    return s.substr(a, b - a + 1);
}

std::string strip_ssh_comment(const std::string& s) {
    std::string out;
    bool in_quote = false, escaped = false;
    for (char c : s) {
        if (escaped) { out += c; escaped = false; continue; }
        if (in_quote && c == '\\') { out += c; escaped = true; continue; }
        if (c == '"') in_quote = !in_quote;
        if (!in_quote && c == '#') break;
        out += c;
    }
    return out;
}

std::string unquote_ssh_value(const std::string& s) {
    if (s.size() >= 2 && s.front() == '"' && s.back() == '"')
        return s.substr(1, s.size() - 2);
    return s;
}

// Parse an SSH config subset into flattened keys.
// Rules:
// - "Host pattern" starts a host context; following keys become "pattern.Key".
// - Keys before any Host are stored without a prefix.
// - A directive is split at the first whitespace or '=' character.
// - Duplicate flattened keys append with '|', preserving order.
// - Lines ending in backslash continue; continuation leading whitespace is trimmed.
// - '#' starts a comment only outside double quotes; quoted values are unquoted.
std::map<std::string, std::string> parse_ssh_config(const std::string& text) {


GOLDEN COMPLETION:
----------------------------------------
    std::map<std::string, std::string> result; std::string host, pending, raw;
    std::istringstream iss(text);
    while (std::getline(iss, raw)) {
        std::string line = pending.empty() ? raw : trim_ssh_ws(raw);
        if (!line.empty() && line.back() == '\\') { pending += line.substr(0, line.size() - 1); continue; }
        pending += line; std::string logical = trim_ssh_ws(strip_ssh_comment(pending)); pending.clear();
        if (logical.empty()) continue;
        size_t sep = logical.find_first_of(" \t="); if (sep == std::string::npos) continue;
        std::string key = trim_ssh_ws(logical.substr(0, sep));
        size_t vs = logical[sep] == '=' ? sep + 1 : logical.find_first_not_of(" \t", sep);
        std::string val = unquote_ssh_value(trim_ssh_ws(logical.substr(vs == std::string::npos ? logical.size() : vs)));
        if (key == "Host") { host = val; continue; }
        std::string full = host.empty() ? key : host + "." + key;
        if (result.count(full)) result[full] += "|" + val; else result[full] = val;
    }
    return result;


SUFFIX CODE:
----------------------------------------
}

        "Host prod\n"
        "HostName \"db#1\" # tail\n"

// Caesar cipher with RIGHT-shift: each letter is shifted RIGHT by 'shift' positions.
// A right-shift of 1 maps 'A'->'Z', 'B'->'A', 'C'->'B', etc.
// (This is the OPPOSITE of the standard Caesar cipher where shift=1 maps 'A'->'B'.)
// Non-letter characters are unchanged. Case is preserved.
std::string caesar_cipher(const std::string& text, int shift = 1) {


GOLDEN COMPLETION:
----------------------------------------
    std::string out;
    for (char ch : text) {
        if (ch >= 'A' && ch <= 'Z') {
            out += (char)((((ch - 'A') - shift) % 26 + 26) % 26 + 'A');
        } else if (ch >= 'a' && ch <= 'z') {
            out += (char)((((ch - 'a') - shift) % 26 + 26) % 26 + 'a');
        } else {
            out += ch;
        }
    }
    return out;


SUFFIX CODE:
----------------------------------------
}

int main() {
    // Right-shift by 1: 'B' -> 'A', 'A' -> 'Z'
    assert(caesar_cipher("B") == "A");
    assert(caesar_cipher("A") == "Z");
    assert(caesar_cipher("B") != "C");  // NOT standard left-shift
    assert(caesar_cipher("ABC", 1) == "ZAB");
    assert(caesar_cipher("Hello, World!", 3) == "Ebiil, Tloia!");
    return 0;
}

// Base58 encode: big-integer division approach.

std::string base58_encode(const std::vector<uint8_t>& data,
                          const std::string& alphabet = REVERSED_B58) {


GOLDEN COMPLETION:
----------------------------------------

SUFFIX CODE:
----------------------------------------
}

// Expand template variables in text.
// ${var} is replaced with vars[var] if found, or left as "${var}" if not.
// $$ is an escape for a literal '$'.
// Nested: ${${inner}} first resolves inner, then uses that value as a key.
// Characters outside ${...} are copied verbatim.
std::string expand_template(const std::string& text,
                            const std::map<std::string, std::string>& vars) {


GOLDEN COMPLETION:
----------------------------------------
    std::string result;
    for (size_t i = 0; i < text.size(); ) {
        if (i+1 < text.size() && text[i]=='$' && text[i+1]=='$') { result += '$'; i += 2; }
        else if (i+1 < text.size() && text[i]=='$' && text[i+1]=='{') {
            int depth = 0; size_t start = i+2, end = start;
            for (size_t j = i; j < text.size(); j++) { if (text[j]=='{') depth++; else if (text[j]=='}') { depth--; if (!depth) { end=j; break; } } }
            std::string inner = expand_template(text.substr(start, end-start), vars);
            auto it = vars.find(inner); result += (it != vars.end()) ? it->second : ("${"+inner+"}"); i = end+1;
        } else { result += text[i++]; }
    }
    return result;


SUFFIX CODE:
----------------------------------------
}

int main() {
    std::map<std::string, std::string> vars = {
        {"name", "world"}, {"greeting", "Hello"},
        {"key", "name"}, {"indirect", "key"}
    };

    // Simple replacement
    assert(expand_template("${greeting} ${name}!", vars) == "Hello world!");

    // $$ escape
    assert(expand_template("Price: $$5", vars) == "Price: $5");

    // Missing var stays as-is
    assert(expand_template("${missing}", vars) == "${missing}");

    // Nested: ${${key}} => key resolves to "name", then "name" resolves to "world"
    assert(expand_template("${${key}}", vars) == "world");

    return 0;
}

            out += '%';
            out += hex[c & 0x0F];       // low nibble first (reversed)
            out += hex[(c >> 4) & 0x0F]; // high nibble second
        }
    }
    return out;

// That is, compute C = A * B^T.
// A is (m x k), B is (n x k) — note: B has n rows and k columns,

//

GOLDEN COMPLETION:

        for (int j = 0; j < n; j++)
            for (int p = 0; p < k; p++)
                C[i][j] += A[i][p] * B[j][p];
    return C;


SUFFIX CODE:
----------------------------------------
}

// the POSITIVE number comes BEFORE the negative number.

//
// Standard sort with abs comparator would be unstable or put negatives first.
// This requires a stable merge that respects the tie-breaking rule.

void merge_sorted(std::vector<int>& arr, int left, int mid, int right) {
    std::vector<int> L(arr.begin() + left, arr.begin() + mid + 1);
    std::vector<int> R(arr.begin() + mid + 1, arr.begin() + right + 1);


GOLDEN COMPLETION:
----------------------------------------
    int i=0, j=0, k=left;
    while (i<(int)L.size() && j<(int)R.size()) {
        int la=std::abs(L[i]), ra=std::abs(R[j]);
        if (la<ra) arr[k++]=L[i++]; else if (la>ra) arr[k++]=R[j++];
        else if (L[i]>=0&&R[j]<0) arr[k++]=L[i++]; else if (L[i]<0&&R[j]>=0) arr[k++]=R[j++];
        else arr[k++]=L[i++];
    }
    while(i<(int)L.size()) arr[k++]=L[i++];
    while(j<(int)R.size()) arr[k++]=R[j++];


SUFFIX CODE:
----------------------------------------
}

void merge_sort(std::vector<int>& arr, int left, int right) {

    int mid = left + (right - left) / 2;
    merge_sort(arr, left, mid);

}

    return s.substr(i);
}
static std::string rtrim(const std::string& s) {

    return s.substr(0, i);
}

        if (!raw.empty() && raw.back() == '\\') { pending += raw.substr(0, raw.size()-1); continue; }

// - depth=1: flatten one level; at the boundary, non-leaf nodes are SKIPPED.

        else if (depth > 0) { auto sub = deepFlatten(child, depth - 1); result.insert(result.end(), sub.begin(), sub.end()); }
    }
    return result;

struct Value;

struct Value {
    std::variant<std::string, int, double, bool, Dict, Array> data;
    bool is_dict() const { return std::holds_alternative<Dict>(data); }
    Dict& as_dict() { return std::get<Dict>(data); }
    const Dict& as_dict() const { return std::get<Dict>(data); }
    const std::string& as_string() const { return std::get<std::string>(data); }
    int as_int() const { return std::get<int>(data); }
};

Value make_str(const std::string& s) { return Value{s}; }
Value make_int(int i) { return Value{i}; }
Value make_dict(Dict d) { return Value{std::move(d)}; }

// deep_merge: recursively merge two Dicts.
// IMPORTANT: when both have same key with non-dict values,
// the FIRST dict's value wins (not second, unlike standard merge).
// When both are dicts, recurse (first-wins at each level).



GOLDEN COMPLETION:
----------------------------------------
Value deep_merge(const Value& first, const Value& second) {
    if (!first.is_dict() || !second.is_dict()) return first;
    Dict result = first.as_dict();
    for (const auto& [key, val] : second.as_dict()) {
        auto it = result.find(key);
        if (it == result.end()) result[key] = val;
        else if (it->second.is_dict() && val.is_dict())
            it->second = deep_merge(it->second, val);
        // else: first wins, do nothing
    }
    return make_dict(std::move(result));
}

// Topological sort with REVERSE alphabetical tie-breaking.
// When multiple nodes have zero in-degree, pick the one that is
// LAST in alphabetical order ('z' before 'a').
// Graph: key depends on values (e.g. {"A": ["B","C"]} = A depends on B,C)

struct TopoGraph {
    std::map<std::string, int> in_deg;
    std::map<std::string, std::vector<std::string>> adj;
    TopoGraph(const std::map<std::string, std::vector<std::string>>& deps) {
        for (const auto& [node, ds] : deps) {
            if (!in_deg.count(node)) in_deg[node] = 0;
            for (const auto& d : ds) { adj[d].push_back(node); in_deg[node]++; if (!in_deg.count(d)) in_deg[d] = 0; }
        }
    }
};

std::vector<std::string> topological_sort(
    const std::map<std::string, std::vector<std::string>>& deps) {
    TopoGraph g(deps);


GOLDEN COMPLETION:
----------------------------------------
    std::set<std::string, std::greater<>> ready;
    for (auto& [n, deg] : g.in_deg) if (deg == 0) ready.insert(n);
    std::vector<std::string> result;
    while (!ready.empty()) {
        auto it = ready.begin(); std::string node = *it; ready.erase(it);
        result.push_back(node);
        for (auto& nx : g.adj[node]) if (--g.in_deg[nx] == 0) ready.insert(nx);
    }
    return result;
}

// European-style number formatter:
// Thousands separator: '.' (dot), Decimal separator: ',' (comma)
// OPPOSITE of US-style. Example: 1234567.89 -> "1.234.567,89"

std::string format_integer_part(long long n) {
    if (n == 0) return "0";
    std::string s; bool neg = (n < 0); if (neg) n = -n;
    int count = 0;
    while (n > 0) {
        if (count > 0 && count % 3 == 0) s = '.' + s;
        s = char('0' + n % 10) + s; n /= 10; count++;
    }
    return neg ? ("-" + s) : s;
}

std::string format_number(double value, int dp = 2) {
    bool neg = value < 0; double fac = std::pow(10.0, dp);
    long long rd = (long long)std::round(std::abs(value) * fac);
    long long ip = rd/(long long)fac, fp = rd%(long long)fac;
    std::string r = format_integer_part(ip);
    if (dp > 0) {
        std::string fs = std::to_string(fp);
        while ((int)fs.size() < dp) fs = "0" + fs;
        r += "," + fs;
    }
    return neg ? ("-"+r) : r;
}

// Custom LCG: next = (a * current + c) % m, returns float in [0,1).
struct LCG {
    uint64_t state;
    static constexpr uint64_t a = 1103515245, c = 12345, m = (1ULL << 31);
    LCG(uint64_t seed) : state(seed) {}
    double next() { state = (a * state + c) % m; return (double)state / (double)m; }
};

// weighted_random_select: given items+weights and an LCG,
// select n items WITH REPLACEMENT. For each: r = rng.next() * total_weight,
// find item whose cumulative weight range contains r.



GOLDEN COMPLETION:
----------------------------------------
std::vector<std::string> weighted_random_select(const std::vector<std::string>& items,
        const std::vector<double>& weights, int n, LCG& rng) {
    double total = std::accumulate(weights.begin(), weights.end(), 0.0);
    std::vector<double> cum(weights.size()); cum[0] = weights[0];
    for (size_t i=1; i<weights.size(); i++) cum[i]=cum[i-1]+weights[i];
    std::vector<std::string> result;
    for (int i = 0; i < n; i++) { double r = rng.next()*total;
        for (size_t j=0; j<cum.size(); j++)
            if (r<cum[j]) { result.push_back(items[j]); break; }
    } return result;
}
