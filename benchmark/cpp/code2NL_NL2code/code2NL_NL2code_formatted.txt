/**
 * @brief Implements a fixed-capacity LRU cache.
 *
 * Uses a doubly-linked list (std::list) to track access order (most-recent
 * at front) and an unordered_map for O(1) key lookup. Supports:
 *   - get(key): returns value or -1 if absent; moves accessed entry to front.
 *   - put(key, value): inserts or updates; evicts the least-recently-used
 *     entry (back of list) when capacity is exceeded.
 */
class LRUCache {
public:
    explicit LRUCache(int capacity) : cap(capacity) {}



GOLDEN COMPLETION:
----------------------------------------
    int get(int key) {
        auto it = index.find(key); if (it == index.end()) return -1;
        items.splice(items.begin(), items, it->second); return it->second->second;
    }
    void put(int key, int value) {
        auto it = index.find(key);
        if (it != index.end()) { it->second->second = value; items.splice(items.begin(), items, it->second); return; }
        if ((int)items.size() == cap) { index.erase(items.back().first); items.pop_back(); }
        items.emplace_front(key, value); index[key] = items.begin();
    }

----------------------------------------

    int cap;
    std::list<std::pair<int,int>> items;
    std::unordered_map<int, std::list<std::pair<int,int>>::iterator> index;
};

/**

 */
struct TrieNode {
    std::unordered_map<char, TrieNode*> children;
    bool is_end = false;
    ~TrieNode() { for (auto& [c, child] : children) delete child; }

/**
 * @brief Trie data structure supporting insert and search.
 *
 * insert(word): Inserts a word into the trie, creating new nodes as needed.
 *   Each character of the word becomes a child node. The last node is marked
 *   as a word-end.
 *
 * search(word): Returns true only if the exact complete word was previously
 *   inserted. A prefix that was not inserted as a complete word returns false.
 */
class Trie {
public:
    TrieNode root;

    void insert(const std::string& word) {


GOLDEN COMPLETION:
----------------------------------------
        TrieNode* node = &root;
        for (char c : word) {
            if (node->children.find(c) == node->children.end()) {
                node->children[c] = new TrieNode();
            }
            node = node->children[c];
        }
        node->is_end = true;


SUFFIX CODE:

    }

        if (c == '(' || c == '[' || c == '{') { stk.push(c); continue; }
        if (c == ')' || c == ']' || c == '}') {
            if (stk.empty()) return false;
            char t = stk.top(); stk.pop();
            if ((c==')'&&t!='(') || (c==']'&&t!='[') || (c=='}'&&t!='{')) return false;
        }
    }
    return stk.empty();

}

    assert(is_balanced("()") == true);
    assert(is_balanced("([{}])") == true);
    assert(is_balanced("(]") == false);
    assert(is_balanced("") == true);
    assert(is_balanced("((())") == false);
    return 0;
