// FNV-1a (Fowler-Noll-Vo) hash function for compile-time string hashing.
//
// This implements the 32-bit variant of FNV-1a, which is widely used for
// hash tables, checksums, and compile-time string comparison.
//
// Algorithm:
//   1. Start with the FNV offset basis (2166136261 for 32-bit)
//   2. For each byte of input:
//      a. XOR the hash with the byte (cast to unsigned char first, then uint32_t)
//      b. Multiply by the FNV prime (16777619 for 32-bit)
//   3. Return the resulting hash
//
// Note: FNV-1a XORs first, then multiplies (unlike FNV-1 which multiplies
// first). The unsigned char cast is critical for portability — char may be
// signed, and sign extension would corrupt the hash for high bytes.

constexpr uint32_t fnv1a_hash(const char* str, size_t len) {
    uint32_t hash = 2166136261u;


GOLDEN COMPLETION:
----------------------------------------
    for (size_t i = 0; i < len; ++i) {
        hash ^= static_cast<uint32_t>(static_cast<unsigned char>(str[i]));
        hash *= 16777619u;
    }
    return hash;


SUFFIX CODE:
----------------------------------------
}

// A lazy pipeline that chains transformations on a vector.
// Transformations are stored as functions and applied only when evaluate() is called.

template <typename T>
class LazyPipeline {
    std::vector<T> data_;
    std::vector<std::function<std::vector<T>(const std::vector<T>&)>> transforms_;
public:
    explicit LazyPipeline(std::vector<T> data) : data_(std::move(data)) {}

    // Add a mapping transform (applies f to each element)
    LazyPipeline& map(std::function<T(const T&)> f) {
        transforms_.push_back([f](const std::vector<T>& v) {
            std::vector<T> result;
            result.reserve(v.size());
            for (const auto& x : v) result.push_back(f(x));
            return result;
        });
        return *this;
    }

    // Add a filter transform (keeps elements where pred is true)
    LazyPipeline& filter(std::function<bool(const T&)> pred) {


GOLDEN COMPLETION:
----------------------------------------
        transforms_.push_back([pred](const std::vector<T>& v) {
            std::vector<T> result;
            for (const auto& x : v) {
                if (pred(x)) result.push_back(x);
            }
            return result;
        });
        return *this;


SUFFIX CODE:
----------------------------------------
    }

    // Evaluate all deferred transforms in order
    std::vector<T> evaluate() const {
        std::vector<T> result = data_;
        for (const auto& transform : transforms_) {
            result = transform(result);
        }
        return result;
    }
};

//
