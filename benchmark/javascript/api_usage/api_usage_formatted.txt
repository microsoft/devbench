// Rules:
// 1. Base URL provides the origin, pathname, and initial params
// 2. Override URL provides additional params to merge
// 3. If a param exists in BOTH, use the OVERRIDE value(s)
// 4. Multi-value params (same key appears multiple times) are preserved as arrays
// 5. Return { href: string, params: Object } where params maps each key to
//    a single string (if one value) or array of strings (if multiple values)
// 6. The returned href must have params sorted alphabetically by key



GOLDEN COMPLETION:
----------------------------------------
function mergeURLParams(baseUrl, overrideUrl) {
    const base = new URL(baseUrl);
    const override = new URL(overrideUrl);
    const merged = new URLSearchParams();
    const overrideKeys = new Set();
    for (const [key] of override.searchParams) overrideKeys.add(key);
    for (const [key, value] of base.searchParams) {
        if (!overrideKeys.has(key)) merged.append(key, value);
    }
    for (const [key, value] of override.searchParams) merged.append(key, value);
    merged.sort();
    const result = new URL(base.origin + base.pathname);
    result.search = merged.toString();
    const params = {};
    for (const key of new Set(merged.keys())) {
        const vals = merged.getAll(key);
        params[key] = vals.length === 1 ? vals[0] : vals;
    }
    return { href: result.href, params };
}

//

GOLDEN COMPLETION:

async function waitForEvent(emitter, eventName, timeoutMs) {
    const ac = new AbortController();
    const timer = setTimeout(() => ac.abort(), timeoutMs);
    try {
        const args = await once(emitter, eventName, { signal: ac.signal });
        clearTimeout(timer);
        return { event: eventName, args, timedOut: false };
    } catch (err) {
        clearTimeout(timer);
        if (err.code === 'ABORT_ERR') {
            return { event: eventName, args: [], timedOut: true };
        }
        throw err;
    }
}

// Key derivation: HMAC-SHA256(salt, rawKey) -> derivedKey
// Then: HMAC-SHA256(derivedKey, message) -> signature

        }

        hostname: url.hostname,

// Capture a structured stack trace at the point where this function is called.
// Must use Error.captureStackTrace and Error.prepareStackTrace to get an array
// of CallSite objects, then map them to { function, file, line } objects.
// The constructorOpt parameter hides frames above the given function.
//
// Returns an array of { function: string, file: string, line: number }
function getStructuredStack(constructorOpt) {


GOLDEN COMPLETION:
----------------------------------------
    const orig = Error.prepareStackTrace;
    Error.prepareStackTrace = (_, stack) => stack;
    const obj = {};
    Error.captureStackTrace(obj, constructorOpt || getStructuredStack);
    const raw = obj.stack;
    Error.prepareStackTrace = orig;
    return raw.map(s => ({
        function: s.getFunctionName() || '<anonymous>',
        file: s.getFileName(),
        line: s.getLineNumber()
    }));


SUFFIX CODE:
----------------------------------------
}

    function innerFunction() {
        return getStructuredStack();
    }
    return innerFunction();

assert.strictEqual(typeof stack[0].file, 'string');
assert.strictEqual(typeof stack[0].line, 'number');

// Measure the execution time of a synchronous function using Performance marks.
// Must use performance.mark() and performance.measure() (NOT Date.now()).
// Returns { result, durationMs, markStart, markEnd, measureName }
function measureSync(name, fn) {


GOLDEN COMPLETION:
----------------------------------------
    const markStart = name + '-start';
    const markEnd = name + '-end';
    const measureName = name + '-duration';
    performance.mark(markStart);
    const result = fn();
    performance.mark(markEnd);
    performance.measure(measureName, markStart, markEnd);
    const entry = performance.getEntriesByName(measureName)[0];
    return { result, durationMs: entry.duration, markStart, markEnd, measureName };


SUFFIX CODE:
----------------------------------------
}
