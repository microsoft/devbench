function debounce(fn, delayMs, options) {
    options = options || {};
    const leading = !!options.leading;
    let timer = null;
    let leadingInvoked = false;
    return function (...args) {
        if (leading && !leadingInvoked) {
            leadingInvoked = true;
            fn.apply(this, args);
        }
        clearTimeout(timer);
        timer = setTimeout(() => {
            if (!leading) fn.apply(this, args);
            leadingInvoked = false;
        }, delayMs);
    };
}

function allSettled(promises) {
    return new Promise((resolve) => {
        if (promises.length === 0) { resolve([]); return; }
        const results = new Array(promises.length);
        let remaining = promises.length;
        promises.forEach((p, i) => {
            Promise.resolve(p)
                .then(value => { results[i] = { status: 'fulfilled', value }; })
                .catch(reason => { results[i] = { status: 'rejected', reason }; })
                .finally(() => { if (--remaining === 0) resolve(results); });
        });
    });
}

        if (key === null || (typeof key !== 'object' && typeof key !== 'function')) {

function parseConfigLines(text) {
    const result = {};
    const lines = text.split(/\r?\n/);
    lines.forEach((line, index) => {
        const trimmed = line.trim();
        if (trimmed === '' || trimmed.startsWith('#')) return;
        const eq = trimmed.indexOf('=');
        if (eq < 0) throw new SyntaxError('Missing = on line ' + (index + 1));
        const key = trimmed.slice(0, eq).trim();
        const value = trimmed.slice(eq + 1).trim();
        if (Object.prototype.hasOwnProperty.call(result, key)) {
            if (Array.isArray(result[key])) result[key].push(value);
            else result[key] = [result[key], value];
        } else {
            result[key] = value;
        }
    });

// Input: an object where keys are node names and values are arrays of
// dependencies (nodes that must come BEFORE the key node).
// Output: an array of node names in valid topological order.
//
// CYCLE HANDLING: If a cycle is detected, do NOT throw an error.
// Instead, return ONLY the nodes that are NOT part of any cycle,
// in valid topological order. Nodes involved in cycles are silently excluded.
//
// Example: { a: [], b: ['a'], c: ['b'] } => ['a', 'b', 'c']
// Example with cycle: { a: ['b'], b: ['a'], c: [] } => ['c']
//   (a and b form a cycle, so only c is returned)



GOLDEN COMPLETION:
----------------------------------------
function topoSort(graph) {
    const inDegree = {};
    const adj = {};
    for (const node of Object.keys(graph)) {
        if (!(node in inDegree)) inDegree[node] = 0;
        if (!(node in adj)) adj[node] = [];
        for (const dep of graph[node]) {
            if (!(dep in inDegree)) inDegree[dep] = 0;
            if (!(dep in adj)) adj[dep] = [];
            adj[dep].push(node);
            inDegree[node]++;
        }
    }
    const queue = Object.keys(inDegree).filter(n => inDegree[n] === 0);
    const result = [];
    while (queue.length > 0) {
        const node = queue.shift();
        result.push(node);
        for (const neighbor of adj[node]) {
            inDegree[neighbor]--;
            if (inDegree[neighbor] === 0) queue.push(neighbor);
        }
    }
    return result;
}

    }

            const val = ref.deref();
            if (val !== undefined) return val;

            return false;
        }
        return true;
    }

    }
}

class LRUCache {
    constructor(capacity) {
        this.capacity = capacity;
        this.map = new Map();
    }

    get(key) {
        if (!this.map.has(key)) return undefined;
        const val = this.map.get(key);
        this.map.delete(key);
        this.map.set(key, val);
        return val;
    }

    set(key, value) {
        if (this.map.has(key)) {
            this.map.delete(key);
            this.map.set(key, value);
            return null;

            this.map.delete(lruKey);
            evicted = { key: lruKey, value: lruVal };
        }
        this.map.set(key, value);
        return evicted;
    }

    get size() {
        return this.map.size;
    }
}

 * The set trap performs a strict equality check (===) between the old and new

 * all subscribers registered for that property by calling each subscriber function.
 * If the values are strictly equal, no notification occurs.
 * @param {Object} initial - The initial state object.
 * @returns {{ state: Proxy, subscribe: Function }} An object with the proxied
 *   state and a subscribe(fn) function. Calling subscribe(fn) runs fn immediately
 *   (to capture dependencies) and re-runs fn whenever any accessed property changes.

function createReactive(initial) {
    const data = { ...initial };
    const deps = {};  // property -> Set of subscriber fns
    let activeSub = null;

    const state = new Proxy(data, {
        get(target, prop) {
            if (activeSub && typeof prop === 'string') {
                if (!deps[prop]) deps[prop] = new Set();
                deps[prop].add(activeSub);
            }
            return target[prop];
        },
        set(target, prop, value) {
            if (target[prop] === value) return true;  // strict equality, no notify
            target[prop] = value;
            if (deps[prop]) {
                deps[prop].forEach(fn => fn());
            }
            return true;
        }
    });

    function subscribe(fn) {
        activeSub = fn;
        fn();  // run immediately to capture deps
        activeSub = null;
    }

    return { state, subscribe };
}

 * - on(event, fn): registers a listener; emits a 'newListener' event
 *   with (event, fn) BEFORE adding the listener to the list.
 * - once(event, fn): registers a listener that auto-removes after its
 *   first invocation; also emits 'newListener' before adding.
 * - off(event, fn): removes a specific listener by reference equality.
 * - emit(event, ...args): calls all listeners for the event synchronously
 *   in the order they were registered.
 * - setMaxListeners(n): sets the per-event listener cap (default: 10).
 *   When adding a listener would exceed this cap, a console.warn is issued
 *   with the message 'Possible memory leak: {count} {event} listeners'.
 *   The listener is still added despite the warning.
 * @returns {MiniEmitter} A new emitter instance.

class MiniEmitter {
    constructor() {
        this._events = {};
        this._maxListeners = 10;
        this._warnings = [];
    }
    setMaxListeners(n) { this._maxListeners = n; return this; }
    on(event, fn) {
        if (event !== 'newListener') {
            this.emit('newListener', event, fn);
        }
        if (!this._events[event]) this._events[event] = [];
        this._events[event].push(fn);
        if (this._events[event].length > this._maxListeners) {
            const msg = `Possible memory leak: ${this._events[event].length} ${event} listeners`;
            this._warnings.push(msg);
        }
        return this;
    }
    once(event, fn) {
        const wrapper = (...args) => {
            this.off(event, wrapper);
            fn(...args);
        };
        wrapper._original = fn;
        this.on(event, wrapper);
        return this;
    }
    off(event, fn) {
        if (!this._events[event]) return this;
        this._events[event] = this._events[event].filter(
            f => f !== fn && f._original !== fn
        );
        return this;
    }
    emit(event, ...args) {
        if (!this._events[event]) return false;
        const listeners = [...this._events[event]];
        listeners.forEach(fn => fn(...args));
        return true;
    }
}

 *   (after delay ms of inactivity)

function debounce(fn, delay, options = {}) {


GOLDEN COMPLETION:
----------------------------------------
    const { leading = false, trailing = true } = options;
    let timer = null;
    let lastArgs = null;
    let canLeading = true;

    function debounced(...args) {
        lastArgs = args;
        if (leading && canLeading) {
            fn(...args);
            canLeading = false;
        }
        if (timer) clearTimeout(timer);
        timer = setTimeout(() => {
            if (trailing && lastArgs) {
                if (!leading || !canLeading) {
                    fn(...lastArgs);
                }
            }
            lastArgs = null;
            canLeading = true;
            timer = null;
        }, delay);
    }
    debounced.cancel = () => {
        clearTimeout(timer);
        timer = null;
        lastArgs = null;
        canLeading = true;
    };
    return debounced;

 * - render(template, vars) replaces {varName} with vars[varName]
 * - Double braces {{text}} are literal escapes: output {text} as-is
 * - Pipe filters: {varName|filterName} applies a built-in filter

function render(template, vars = {}) {


GOLDEN COMPLETION:
----------------------------------------

        upper: (s) => String(s).toUpperCase(),
        lower: (s) => String(s).toLowerCase(),
        pad: (s, w) => String(s).padStart(Number(w), ' '),

                }

        }

/**
 * Creates a middleware chain runner.
 *
 * Each middleware is a function(ctx, next) where:
 * - ctx is a shared context object passed to all middleware
 * - next() invokes the next middleware in the chain; returns a Promise
 * - next() may be called AT MOST ONCE per middleware. Calling it a second
 *   time throws an Error with message 'next() called multiple times'
 * - If next() is not called, downstream middleware is skipped
 * - Middleware may be async (return a Promise)
 *
 * compose(middlewares) returns a function(ctx) => Promise that runs the chain.
 * The returned promise resolves when the chain completes.
 *
 * @param {Function[]} middlewares
 * @returns {Function} (ctx) => Promise<void>
 */
function compose(middlewares) {


GOLDEN COMPLETION:
----------------------------------------
  return function(ctx) {
    let index = -1;
    function dispatch(i) {
      if (i <= index) return Promise.reject(new Error('next() called multiple times'));
      index = i;
      if (!middlewares[i]) return Promise.resolve();
      try { return Promise.resolve(middlewares[i](ctx, () => dispatch(i + 1))); }
      catch (err) { return Promise.reject(err); }
    }
    return dispatch(0);
  };


SUFFIX CODE:
----------------------------------------
}

/**
 * A validator class that overrides instanceof behavior via Symbol.hasInstance.
 *
 * When code uses `value instanceof TypeValidator`, the static
 * Symbol.hasInstance method is called instead of the default prototype
 * chain check. This override returns true if the value is a non-null
 * object with a `__validated` property set to true, regardless of
 * whether the value was constructed by TypeValidator.
 *
 * Note: Symbol.hasInstance does NOT affect construction (`new TypeValidator()`
 * still creates a normal instance). It only intercepts the `instanceof` operator.
 * Subclasses inherit this behavior unless they define their own Symbol.hasInstance.
 *
 * @class
 */


SUFFIX CODE:

class TypeValidator {
  constructor(name) {
    this.name = name;
    this.__validated = true;
  }

  static [Symbol.hasInstance](value) {
    return value !== null && typeof value === 'object' && value.__validated === true;
  }
}

class StrictValidator extends TypeValidator {
  constructor(name) {
    super(name);
  }
}
