class RecordingNamespace(dict):
    def __init__(self):
        super().__init__()
        self.order = []

    def __setitem__(self, key, value):
        if key not in self:
            self.order.append(key)
        super().__setitem__(key, value)

class Field:
    def __init__(self, typ):
        self.typ = typ

class FieldMeta(type):
    """Metaclass that records Field declarations in source order.

    Use __prepare__ to return RecordingNamespace.
    __new__ receives keyword-only prefix from class declarations such as
    class User(metaclass=FieldMeta, prefix="db_").
    It must set cls.declared_fields to a tuple of prefixed names for values
    that are Field instances, preserving declaration order and ignoring helpers.
    """


GOLDEN COMPLETION:
----------------------------------------
    @classmethod
    def __prepare__(mcls, name, bases, **kwargs):
        return RecordingNamespace()

    def __new__(mcls, name, bases, namespace, *, prefix="", **kwargs):
        cls = super().__new__(mcls, name, bases, dict(namespace))
        cls.declared_fields = tuple(
            f"{prefix}{key}"
            for key in namespace.order
            if isinstance(namespace.get(key), Field)
        )
        return cls
