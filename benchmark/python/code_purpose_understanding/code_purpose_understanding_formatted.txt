class ClaimAccumulator:
    """Adjudicates simple outpatient visits against member accumulators.
    post_visit must be idempotent by visit_id. For a new visit it validates
    the code and billed amount before mutating state. Patient responsibility is
    applied in this order: optional facility fee, deductible, then coinsurance.
    The member out-of-pocket remaining caps all patient responsibility.
    Successful visits are stored, totals are updated, and an audit entry is added.
    Failed validations must leave all balances, totals, visits, and audit intact."""

    def __init__(self, allowed_by_code, deductible_remaining, oop_remaining, coinsurance=0.2):
        self.allowed_by_code = dict(allowed_by_code)
        self.deductible_remaining = float(deductible_remaining)
        self.oop_remaining = float(oop_remaining)
        self.coinsurance = coinsurance
        self.visits = {}
        self.audit = []
        self.patient_total = 0.0
        self.plan_total = 0.0

    def snapshot(self):
        return {
            'deductible_remaining': self.deductible_remaining,
            'oop_remaining': self.oop_remaining,
            'patient_total': self.patient_total,
            'plan_total': self.plan_total,
            'visits': dict(self.visits),
            'audit': list(self.audit),
        }



GOLDEN COMPLETION:
----------------------------------------
    def post_visit(self, visit_id, code, billed_amount, facility=False):
        if visit_id in self.visits:
            return dict(self.visits[visit_id])
        if code not in self.allowed_by_code:
            raise ValueError("unknown service code")
        billed_amount = float(billed_amount)
        if billed_amount < 0:
            raise ValueError("billed amount must be non-negative")

        allowed = round(min(billed_amount, float(self.allowed_by_code[code])), 2)
        remaining = allowed
        patient = 0.0
        facility_fee = 0.0
        if facility and remaining > 0 and self.oop_remaining > 0:
            facility_fee = min(25.0, remaining, self.oop_remaining)
            patient += facility_fee
            remaining = round(remaining - facility_fee, 2)

        oop_left = max(0.0, self.oop_remaining - patient)
        deductible = min(remaining, self.deductible_remaining, oop_left)
        patient += deductible
        remaining = round(remaining - deductible, 2)

        oop_left = max(0.0, self.oop_remaining - patient)
        coinsurance = min(round(remaining * self.coinsurance, 2), oop_left)
        patient = round(patient + coinsurance, 2)
        plan = round(allowed - patient, 2)

        self.deductible_remaining = round(self.deductible_remaining - deductible, 2)
        self.oop_remaining = round(self.oop_remaining - patient, 2)
        self.patient_total = round(self.patient_total + patient, 2)
        self.plan_total = round(self.plan_total + plan, 2)

        result = {
            'visit_id': visit_id, 'code': code, 'allowed': allowed,
            'patient': patient, 'plan': plan, 'facility_fee': round(facility_fee, 2),
            'deductible': round(deductible, 2), 'coinsurance': round(coinsurance, 2),
        }
        self.visits[visit_id] = dict(result)
        self.audit.append({'visit_id': visit_id, 'patient': patient, 'plan': plan, 'oop_remaining': self.oop_remaining})
        return dict(result)

class PayrollLedger:
    """Posts approved hourly timesheets into payroll.
    post_timesheet is idempotent by sheet_id. For a new sheet it must validate
    the employee and hours before mutating anything. Gross pay uses 40 regular
    hours then 1.5x overtime. Benefit deductions are pre-tax and capped by gross.
    Tax is withheld from taxable wages. Garnishment is taken after tax and is
    capped by both the remaining garnishment balance and 25% of disposable pay.
    Successful posts update employee YTD totals, remaining garnishment, payroll
    cash, stored timesheet result, and audit log."""

    def __init__(self):
        self.employees = {}
        self.timesheets = {}
        self.audit = []
        self.payroll_cash = 0.0

    def add_employee(self, emp_id, hourly_rate, tax_rate, benefit=0.0, garnishment_balance=0.0, active=True):
        self.employees[emp_id] = {
            'rate': float(hourly_rate), 'tax_rate': tax_rate,
            'benefit': float(benefit), 'garnishment_balance': float(garnishment_balance),
            'active': active, 'ytd_gross': 0.0, 'ytd_tax': 0.0, 'ytd_net': 0.0,
        }

    def snapshot(self):
        return {
            'employees': {k: dict(v) for k, v in self.employees.items()},
            'timesheets': dict(self.timesheets),
            'audit': list(self.audit),
            'payroll_cash': self.payroll_cash,
        }



GOLDEN COMPLETION:
----------------------------------------
    def post_timesheet(self, sheet_id, emp_id, hours, period):
        if sheet_id in self.timesheets:
            return dict(self.timesheets[sheet_id])
        if emp_id not in self.employees:
            raise ValueError("unknown employee")
        emp = self.employees[emp_id]
        if not emp['active']:
            raise ValueError("inactive employee")
        hours = float(hours)
        if hours < 0 or hours > 80:
            raise ValueError("invalid hours")

        regular_hours = min(hours, 40.0)
        overtime_hours = max(0.0, hours - 40.0)
        gross = round(regular_hours * emp['rate'] + overtime_hours * emp['rate'] * 1.5, 2)
        benefit = round(min(emp['benefit'], gross), 2)
        taxable = round(gross - benefit, 2)
        tax = round(taxable * emp['tax_rate'], 2)
        disposable = round(gross - benefit - tax, 2)
        garnishment_cap = round(disposable * 0.25, 2)
        garnishment = round(min(emp['garnishment_balance'], garnishment_cap), 2)
        net = round(disposable - garnishment, 2)

        emp['garnishment_balance'] = round(emp['garnishment_balance'] - garnishment, 2)
        emp['ytd_gross'] = round(emp['ytd_gross'] + gross, 2)
        emp['ytd_tax'] = round(emp['ytd_tax'] + tax, 2)
        emp['ytd_net'] = round(emp['ytd_net'] + net, 2)
        self.payroll_cash = round(self.payroll_cash + net, 2)

        result = {
            'sheet_id': sheet_id, 'emp_id': emp_id, 'period': period,
            'regular_hours': regular_hours, 'overtime_hours': overtime_hours,
            'gross': gross, 'benefit': benefit, 'tax': tax,
            'garnishment': garnishment, 'net': net,
        }
        self.timesheets[sheet_id] = dict(result)
        self.audit.append({'sheet_id': sheet_id, 'emp_id': emp_id, 'net': net, 'period': period})
        return dict(result)

class Subscription:
    VALID_TRANSITIONS = {
        'pending': ['active'],
        'active': ['paused', 'cancelled'],
        'paused': ['active', 'cancelled'],
        'cancelled': []
    }

    def __init__(self, user_id, plan, cooldown=0):
        self.user_id = user_id
        self.plan = plan
        self.state = 'pending'
        self.cooldown = cooldown
        self.audit_log = []
        self.paused_at = None

    def _transition(self, new_state, timestamp):
        if new_state not in self.VALID_TRANSITIONS[self.state]:
            raise ValueError(f"Cannot transition from {self.state} to {new_state}")
        old_state = self.state
        self.state = new_state
        self.audit_log.append({'from': old_state, 'to': new_state, 'ts': timestamp})

    def activate(self, timestamp):
        self._transition('active', timestamp)

    def pause(self, timestamp):
        self._transition('paused', timestamp)
        self.paused_at = timestamp



GOLDEN COMPLETION:
----------------------------------------
    def resume(self, timestamp):
        if self.state != 'paused':
            raise ValueError(f"Cannot resume from {self.state}")
        if self.paused_at is not None and (timestamp - self.paused_at) < self.cooldown:
            raise ValueError("Cooldown period has not elapsed")
        self._transition('active', timestamp)

class PurchaseOrderBook:
    """Receives purchase-order shipments into inventory.
    receive() is idempotent by receipt_id. It validates the purchase order,
    receipt lines, and over-receipt limits before mutating anything. Accepted
    units increase on-hand stock and weighted average cost. Defective units go
    to quarantine and count against the ordered quantity but do not create an
    inventory accrual. Line and order status close only when ordered quantities
    have all been received. Successful receipts update accruals, receipt storage,
    and audit entries atomically."""

    def __init__(self):
        self.orders = {}
        self.on_hand = {}
        self.avg_cost = {}
        self.quarantine = {}
        self.receipts = {}
        self.audit = []
        self.accrual_total = 0.0

    def create_order(self, po_id, vendor, lines):
        self.orders[po_id] = {
            'vendor': vendor, 'status': 'open',
            'lines': {line['sku']: {
                'ordered': line['qty'], 'received': 0,
                'unit_cost': float(line['unit_cost']), 'status': 'open'
            } for line in lines}
        }

    def seed_inventory(self, sku, qty, avg_cost):
        self.on_hand[sku] = qty
        self.avg_cost[sku] = float(avg_cost)

    def snapshot(self):
        return {
            'orders': {po: {'vendor': o['vendor'], 'status': o['status'], 'lines': {k: dict(v) for k, v in o['lines'].items()}} for po, o in self.orders.items()},
            'on_hand': dict(self.on_hand), 'avg_cost': dict(self.avg_cost),
            'quarantine': dict(self.quarantine), 'receipts': dict(self.receipts),
            'audit': list(self.audit), 'accrual_total': self.accrual_total,
        }



GOLDEN COMPLETION:
----------------------------------------
    def receive(self, po_id, receipt_id, lines, timestamp):
        if receipt_id in self.receipts:
            return dict(self.receipts[receipt_id])
        if po_id not in self.orders:
            raise ValueError("unknown purchase order")
        order = self.orders[po_id]
        if order['status'] == 'closed':
            raise ValueError("purchase order is closed")

        totals = {}
        for line in lines:
            sku = line['sku']
            accepted = int(line.get('accepted', 0))
            defective = int(line.get('defective', 0))
            if accepted < 0 or defective < 0:
                raise ValueError("negative receipt quantity")
            if sku not in order['lines']:
                raise ValueError("sku not on purchase order")
            totals.setdefault(sku, {'accepted': 0, 'defective': 0})
            totals[sku]['accepted'] += accepted
            totals[sku]['defective'] += defective

        for sku, qtys in totals.items():
            po_line = order['lines'][sku]
            remaining = po_line['ordered'] - po_line['received']
            if qtys['accepted'] + qtys['defective'] > remaining:
                raise ValueError("receipt exceeds remaining order quantity")

        accepted_units = 0
        defective_units = 0
        accrued = 0.0
        for sku, qtys in totals.items():
            po_line = order['lines'][sku]
            accepted = qtys['accepted']
            defective = qtys['defective']
            received = accepted + defective
            old_qty = self.on_hand.get(sku, 0)
            old_cost = self.avg_cost.get(sku, po_line['unit_cost'])
            if accepted:
                new_qty = old_qty + accepted
                new_cost = ((old_qty * old_cost) + (accepted * po_line['unit_cost'])) / new_qty
                self.on_hand[sku] = new_qty
                self.avg_cost[sku] = round(new_cost, 2)
            if defective:
                self.quarantine[sku] = self.quarantine.get(sku, 0) + defective
            po_line['received'] += received
            if po_line['received'] == po_line['ordered']:
                po_line['status'] = 'closed'
            accepted_units += accepted
            defective_units += defective
            accrued += accepted * po_line['unit_cost']

        order['status'] = 'closed' if all(line['status'] == 'closed' for line in order['lines'].values()) else 'open'
        accrued = round(accrued, 2)
        self.accrual_total = round(self.accrual_total + accrued, 2)
        result = {
            'receipt_id': receipt_id, 'po_id': po_id,
            'accepted_units': accepted_units, 'defective_units': defective_units,
            'accrued': accrued, 'order_status': order['status'],
        }
        self.receipts[receipt_id] = dict(result)
        self.audit.append({'receipt_id': receipt_id, 'po_id': po_id, 'ts': timestamp, 'accrued': accrued})
        return dict(result)

class Subscription:
    VALID_TRANSITIONS = {
        'pending': ['active'],
        'active': ['paused', 'cancelled', 'active'],
        'paused': ['active', 'cancelled'],
        'cancelled': []
    }

    def __init__(self, user_id, plan, plan_price, total_days):
        self.user_id = user_id
        self.plan = plan
        self.plan_price = plan_price
        self.total_days = total_days
        self.state = 'pending'
        self.days_used = 0
        self.audit_log = []
        self.balance_due = 0.0

    def _transition(self, new_state, timestamp):
        if new_state not in self.VALID_TRANSITIONS[self.state]:
            raise ValueError(f"Cannot transition from {self.state} to {new_state}")
        old_state = self.state
        self.state = new_state
        self.audit_log.append({'from': old_state, 'to': new_state, 'ts': timestamp})

    def activate(self, timestamp):
        self._transition('active', timestamp)

    def pause(self, timestamp):
        self._transition('paused', timestamp)

    def use_days(self, n):
        self.days_used += n

    def _compute_proration(self, price):
        remaining = self.total_days - self.days_used
        return (remaining / self.total_days) * price



GOLDEN COMPLETION:
----------------------------------------
    def upgrade(self, new_plan, new_price, timestamp):
        if self.state not in ('active', 'paused'):
            raise ValueError("Can only upgrade active or paused subscriptions")
        credit = self._compute_proration(self.plan_price)
        cost = self._compute_proration(new_price)
        self.balance_due = round(cost - credit, 2)
        old_plan = self.plan
        self.plan = new_plan
        self.plan_price = new_price
        if self.state == 'paused':
            self._transition('active', timestamp)
        self.audit_log.append({'action': 'upgrade', 'from_plan': old_plan, 'to_plan': new_plan, 'ts': timestamp})

class CardStatement:
    """Credit-card statement ledger with a nonstandard payment waterfall.
    apply_payment is idempotent by payment_id and validates amount before any
    mutation. Payments allocate in this exact order: fees, interest, cash advance,
    then purchases. Applied payment releases available credit up to the credit
    limit; remaining money becomes overpayment and does not increase available
    credit beyond the limit. A delinquent account returns to current only when
    all balances are paid. Successful payments are saved and audited."""

    WATERFALL = ('fees', 'interest', 'cash_advance', 'purchases')

    def __init__(self, credit_limit):
        self.credit_limit = float(credit_limit)
        self.available_credit = float(credit_limit)
        self.balances = {'fees': 0.0, 'interest': 0.0, 'cash_advance': 0.0, 'purchases': 0.0}
        self.status = 'current'
        self.payments = {}
        self.audit = []
        self.overpayment = 0.0

    def add_balance(self, kind, amount):
        if kind not in self.balances:
            raise ValueError('unknown balance kind')
        amount = float(amount)
        self.balances[kind] = round(self.balances[kind] + amount, 2)
        self.available_credit = round(max(0.0, self.available_credit - amount), 2)

    def mark_delinquent(self):
        self.status = 'delinquent'

    def total_balance(self):
        return round(sum(self.balances.values()), 2)

    def snapshot(self):
        return {
            'available_credit': self.available_credit, 'balances': dict(self.balances),
            'status': self.status, 'payments': dict(self.payments),
            'audit': list(self.audit), 'overpayment': self.overpayment,
        }



GOLDEN COMPLETION:
----------------------------------------
    def apply_payment(self, payment_id, amount, timestamp):
        if payment_id in self.payments:
            return dict(self.payments[payment_id])
        amount = float(amount)
        if amount <= 0:
            raise ValueError("payment amount must be positive")

        remaining = amount
        allocations = {}
        applied = 0.0
        for kind in self.WATERFALL:
            due = self.balances[kind]
            paid = min(due, remaining)
            paid = round(paid, 2)
            allocations[kind] = paid
            self.balances[kind] = round(due - paid, 2)
            remaining = round(remaining - paid, 2)
            applied = round(applied + paid, 2)

        over = round(remaining, 2)
        self.overpayment = round(self.overpayment + over, 2)
        self.available_credit = round(min(self.credit_limit, self.available_credit + applied), 2)
        if self.status == 'delinquent' and self.total_balance() == 0.0:
            self.status = 'current'

        result = {
            'payment_id': payment_id, 'applied': applied, 'overpayment': over,
            'allocations': allocations, 'status': self.status,
            'available_credit': self.available_credit,
        }
        self.payments[payment_id] = dict(result)
        self.audit.append({'payment_id': payment_id, 'ts': timestamp, 'applied': applied, 'overpayment': over})
        return dict(result)

class TaxCalculator:
    def __init__(self, brackets, std_deduction, phase_out_start):
        self.brackets = brackets  # [(upper_limit, rate), ...], last has None
        self.std_deduction = std_deduction
        self.phase_out_start = phase_out_start

    def compute_tax(self, income):
        tax = 0.0
        prev_limit = 0
        for upper, rate in self.brackets:
            if income <= prev_limit:
                break
            if upper is None:
                tax += (income - prev_limit) * rate
                break
            taxable = min(income, upper) - prev_limit
            tax += taxable * rate
            prev_limit = upper
        return round(tax, 2)



GOLDEN COMPLETION:
----------------------------------------
    def adjusted_deduction(self, gross_income):
        if gross_income <= self.phase_out_start:
            return self.std_deduction
        excess = gross_income - self.phase_out_start
        reduction = (excess // 1000) * 100
        return max(self.std_deduction - reduction, 0)

    def tax_owed(self, gross_income):
        deduction = self.adjusted_deduction(gross_income)
        taxable = max(gross_income - deduction, 0)
        return self.compute_tax(taxable)

class MarketplaceSettlement:
    """Settles captured marketplace orders to sellers.
    settle_order is idempotent by settlement_id. It validates the order exists,
    is captured, is not disputed, and has a delivery timestamp on or after the
    capture timestamp before mutating anything. Commission is charged on item
    subtotal only, reserve is withheld from subtotal only, and shipping passes
    through to the seller. Existing seller adjustment balances reduce payout
    after commission and reserve. Successful settlements update seller payable,
    seller reserve, platform fees, reserve liability, order state, saved result,
    and audit log."""

    def __init__(self, commission_rate=0.10, reserve_rate=0.05):
        self.commission_rate = commission_rate
        self.reserve_rate = reserve_rate
        self.sellers = {}
        self.orders = {}
        self.settlements = {}
        self.audit = []
        self.platform_fees = 0.0
        self.reserve_liability = 0.0

    def add_seller(self, seller_id, adjustment_balance=0.0):
        self.sellers[seller_id] = {
            'adjustment_balance': float(adjustment_balance),
            'payable': 0.0, 'reserve': 0.0, 'settled_count': 0,
        }

    def capture_order(self, order_id, seller_id, subtotal, shipping, captured_at, disputed=False):
        self.orders[order_id] = {
            'seller_id': seller_id, 'subtotal': float(subtotal), 'shipping': float(shipping),
            'captured_at': captured_at, 'disputed': disputed, 'state': 'captured',
        }

    def snapshot(self):
        return {
            'sellers': {k: dict(v) for k, v in self.sellers.items()},
            'orders': {k: dict(v) for k, v in self.orders.items()},
            'settlements': dict(self.settlements), 'audit': list(self.audit),
            'platform_fees': self.platform_fees, 'reserve_liability': self.reserve_liability,
        }



GOLDEN COMPLETION:
----------------------------------------
    def settle_order(self, settlement_id, order_id, delivered_at):
        if settlement_id in self.settlements:
            return dict(self.settlements[settlement_id])
        if order_id not in self.orders:
            raise ValueError("unknown order")
        order = self.orders[order_id]
        if order['seller_id'] not in self.sellers:
            raise ValueError("unknown seller")
        if order['state'] != 'captured':
            raise ValueError("order is not captured")
        if order['disputed']:
            raise ValueError("disputed order cannot settle")
        if delivered_at < order['captured_at']:
            raise ValueError("delivery precedes capture")

        seller = self.sellers[order['seller_id']]
        subtotal = order['subtotal']
        shipping = order['shipping']
        gross = round(subtotal + shipping, 2)
        commission = round(subtotal * self.commission_rate, 2)
        reserve = round(subtotal * self.reserve_rate, 2)
        before_adjustment = round(gross - commission - reserve, 2)
        adjustment = round(min(seller['adjustment_balance'], before_adjustment), 2)
        payout = round(before_adjustment - adjustment, 2)

        seller['adjustment_balance'] = round(seller['adjustment_balance'] - adjustment, 2)
        seller['payable'] = round(seller['payable'] + payout, 2)
        seller['reserve'] = round(seller['reserve'] + reserve, 2)
        seller['settled_count'] += 1
        self.platform_fees = round(self.platform_fees + commission, 2)
        self.reserve_liability = round(self.reserve_liability + reserve, 2)
        order['state'] = 'settled'

        result = {
            'settlement_id': settlement_id, 'order_id': order_id,
            'gross': gross, 'commission': commission, 'reserve': reserve,
            'adjustment': adjustment, 'payout': payout, 'seller_id': order['seller_id'],
        }
        self.settlements[settlement_id] = dict(result)
        self.audit.append({'settlement_id': settlement_id, 'order_id': order_id, 'payout': payout, 'delivered_at': delivered_at})
        return dict(result)

class WarrantyClaimLedger:
    """Approves appliance warranty repair claims.
    approve_repair is idempotent by approval_id. It validates the claim, labor,
    and all requested parts before mutating anything. Covered parts plus labor
    form the eligible amount. Uncovered parts and expedited fees are customer
    responsibility. Deductible is applied before warranty coverage, then warranty
    payment is capped by remaining coverage. Successful approvals update coverage
    and deductible balances, vendor payables, claim status, saved approval, and
    audit log."""

    def __init__(self, coverage_remaining, deductible_remaining, labor_rate=80):
        self.coverage_remaining = float(coverage_remaining)
        self.deductible_remaining = float(deductible_remaining)
        self.labor_rate = float(labor_rate)
        self.parts_catalog = {}
        self.claims = {}
        self.approvals = {}
        self.vendor_payables = {}
        self.audit = []
        self.reserve_used = 0.0

    def add_part(self, part_no, price, covered=True):
        self.parts_catalog[part_no] = {'price': float(price), 'covered': covered}

    def open_claim(self, claim_id, serial_no):
        self.claims[claim_id] = {'serial_no': serial_no, 'status': 'open'}

    def snapshot(self):
        return {
            'coverage_remaining': self.coverage_remaining,
            'deductible_remaining': self.deductible_remaining,
            'claims': {k: dict(v) for k, v in self.claims.items()},
            'approvals': dict(self.approvals),
            'vendor_payables': dict(self.vendor_payables),
            'audit': list(self.audit), 'reserve_used': self.reserve_used,
        }



GOLDEN COMPLETION:
----------------------------------------
    def approve_repair(self, approval_id, claim_id, part_lines, labor_hours, vendor, expedited=False):
        if approval_id in self.approvals:
            return dict(self.approvals[approval_id])
        if claim_id not in self.claims:
            raise ValueError("unknown claim")
        claim = self.claims[claim_id]
        if claim['status'] != 'open':
            raise ValueError("claim is not open")
        labor_hours = float(labor_hours)
        if labor_hours < 0:
            raise ValueError("negative labor hours")

        covered_parts = 0.0
        uncovered_parts = 0.0
        for line in part_lines:
            part_no = line['part_no']
            qty = int(line.get('qty', 0))
            if qty < 0:
                raise ValueError("negative part quantity")
            if part_no not in self.parts_catalog:
                raise ValueError("unknown part")
            part = self.parts_catalog[part_no]
            amount = qty * part['price']
            if part['covered']:
                covered_parts += amount
            else:
                uncovered_parts += amount

        labor = labor_hours * self.labor_rate
        expedited_fee = 35.0 if expedited else 0.0
        eligible = round(covered_parts + labor, 2)
        deductible = round(min(self.deductible_remaining, eligible), 2)
        after_deductible = round(eligible - deductible, 2)
        warranty_pay = round(min(self.coverage_remaining, after_deductible), 2)
        excess = round(after_deductible - warranty_pay, 2)
        customer_due = round(uncovered_parts + expedited_fee + deductible + excess, 2)
        vendor_payable = round(eligible + uncovered_parts + expedited_fee, 2)

        self.deductible_remaining = round(self.deductible_remaining - deductible, 2)
        self.coverage_remaining = round(self.coverage_remaining - warranty_pay, 2)
        self.reserve_used = round(self.reserve_used + warranty_pay, 2)
        self.vendor_payables[vendor] = round(self.vendor_payables.get(vendor, 0.0) + vendor_payable, 2)
        claim['status'] = 'approved'

        result = {
            'approval_id': approval_id, 'claim_id': claim_id, 'vendor': vendor,
            'eligible': eligible, 'deductible': deductible, 'warranty_pay': warranty_pay,
            'customer_due': customer_due, 'vendor_payable': vendor_payable,
        }
        self.approvals[approval_id] = dict(result)
        self.audit.append({'approval_id': approval_id, 'claim_id': claim_id, 'warranty_pay': warranty_pay, 'customer_due': customer_due})
        return dict(result)

class Amortizer:
    """Generates an amortization schedule. Each month:
    - interest = balance * monthly_rate (rounded to 2 decimals)
    - For months 1..N-1: payment is fixed, principal = payment - interest
    - For month N (last): principal = remaining balance, payment = principal + interest
    This ensures final balance is exactly 0 despite rounding.
    Returns list of dicts with keys: month, payment, interest, principal, balance."""

    def __init__(self, principal, annual_rate, num_payments):
        self.principal = principal
        self.annual_rate = annual_rate
        self.num_payments = num_payments
        self.monthly_rate = annual_rate / 12

    def monthly_payment(self):
        r = self.monthly_rate
        n = self.num_payments
        if r == 0:
            return round(self.principal / n, 2)
        payment = self.principal * (r * (1 + r) ** n) / ((1 + r) ** n - 1)
        return round(payment, 2)



GOLDEN COMPLETION:
----------------------------------------
    def schedule(self):
        payment = self.monthly_payment()
        balance = self.principal
        result = []
        for i in range(1, self.num_payments + 1):
            interest = round(balance * self.monthly_rate, 2)
            is_last = (i == self.num_payments)
            princ = balance if is_last else payment - interest
            pmt = princ + interest if is_last else payment
            balance = round(balance - princ, 2)
            result.append({'month': i, 'payment': round(pmt, 2), 'interest': interest, 'principal': round(princ, 2), 'balance': balance})
        return result

class PermissionSystem:
    """Role-based permission system with explicit deny support.
    Roles have hierarchy levels (higher = more permissions).
    Permissions have required levels.
    A user has a role, and can also have explicitly denied permissions.
    has_permission checks:
    1. If user doesn't exist -> False
    2. If permission is in user's denied set -> False (deny overrides role)
    3. If permission is unknown -> False
    4. If user's role level >= permission level -> True, else False"""

    ROLE_HIERARCHY = {
        'viewer': 0,
        'manager': 1,
        'admin': 2
    }

    PERMISSION_LEVEL = {
        'read': 0,
        'write': 1,
        'delete': 2
    }

    def __init__(self):
        self.users = {}  # user_id -> {'role': str, 'denied': set()}

    def add_user(self, user_id, role):
        if role not in self.ROLE_HIERARCHY:
            raise ValueError(f"Unknown role: {role}")
        self.users[user_id] = {'role': role, 'denied': set()}

    def deny_permission(self, user_id, permission):
        if user_id not in self.users:
            raise ValueError("User not found")
        self.users[user_id]['denied'].add(permission)



GOLDEN COMPLETION:
----------------------------------------
    def has_permission(self, user_id, permission):
        if user_id not in self.users:
            return False
        user = self.users[user_id]
        if permission in user['denied']:
            return False
        role_level = self.ROLE_HIERARCHY[user['role']]
        perm_level = self.PERMISSION_LEVEL.get(permission)
        if perm_level is None:
            return False
        return role_level >= perm_level

class LoanAccount:
    """Loan servicer with a non-standard payment waterfall.
    apply_payment(amount, timestamp) must allocate cash in this exact order:
    1. late_fees
    2. principal
    3. accrued_interest
    4. unapplied_cash
    This intentionally differs from the usual interest-before-principal order.
    Charged-off loans do not reduce balances; payments become recovery_cash.
    Every successful payment appends one audit entry with the allocation dict."""

    def __init__(self, principal):
        self.principal = float(principal)
        self.accrued_interest = 0.0
        self.late_fees = 0.0
        self.unapplied_cash = 0.0
        self.recovery_cash = 0.0
        self.status = 'current'
        self.audit_log = []

    def accrue_interest(self, amount):
        self.accrued_interest = round(self.accrued_interest + amount, 2)

    def add_late_fee(self, amount):
        self.late_fees = round(self.late_fees + amount, 2)

    def charge_off(self):
        self.status = 'charged_off'

    def _take(self, field, remaining):
        amount = min(getattr(self, field), remaining)
        setattr(self, field, round(getattr(self, field) - amount, 2))
        return round(amount, 2)



GOLDEN COMPLETION:
----------------------------------------
    def apply_payment(self, amount, timestamp):
        if amount <= 0:
            raise ValueError("payment amount must be positive")
        remaining = round(float(amount), 2)
        if self.status == 'charged_off':
            self.recovery_cash = round(self.recovery_cash + remaining, 2)
            allocation = {'late_fees': 0.0, 'principal': 0.0, 'accrued_interest': 0.0, 'unapplied_cash': 0.0, 'recovery_cash': remaining}
            self.audit_log.append({'action': 'payment', 'amount': remaining, 'ts': timestamp, 'allocation': allocation})
            return allocation
        allocation = {}
        for field in ('late_fees', 'principal', 'accrued_interest'):
            paid = self._take(field, remaining)
            allocation[field] = paid
            remaining = round(remaining - paid, 2)
        allocation['unapplied_cash'] = remaining
        allocation['recovery_cash'] = 0.0
        self.unapplied_cash = round(self.unapplied_cash + remaining, 2)
        if self.principal == 0 and self.accrued_interest == 0 and self.late_fees == 0:
            self.status = 'paid_off'
        self.audit_log.append({'action': 'payment', 'amount': round(float(amount), 2), 'ts': timestamp, 'allocation': allocation})
        return allocation

class TemporalPermissions:
    """Time-bounded permission system.
    Grants and denials have valid_from (inclusive) and valid_until (exclusive, or None=forever).
    _is_active checks if an entry is active at a given time.
    has_permission: deny-overrides-grant.
    - If any active denial matches -> False
    - If any active grant matches -> True
    - Otherwise -> False"""

    def __init__(self):
        self.grants = []  # list of {'user','perm','valid_from','valid_until'}
        self.denials = []

    def grant(self, user, perm, valid_from, valid_until=None):
        self.grants.append({
            'user': user, 'perm': perm,
            'valid_from': valid_from, 'valid_until': valid_until
        })

    def deny(self, user, perm, valid_from, valid_until=None):
        self.denials.append({
            'user': user, 'perm': perm,
            'valid_from': valid_from, 'valid_until': valid_until
        })

class ClaimAdjudicator:
    """Medical claim adjudication with plan accumulators.
    adjudicate_line(code, billed_amount, prior_auth=False) must:
    - Denied codes return status 'denied' and do not change accumulators.
    - Codes requiring prior auth return status 'auth_required' unless prior_auth=True.
    - allowed amount is min(billed_amount, allowed_rates.get(code, billed_amount)).
    - Patient responsibility order is copay, then deductible, then coinsurance.
    - copay applies only to codes in copay_codes and is capped by allowed amount.
    - deductible applies to remaining allowed amount and reduces deductible_remaining.
    - coinsurance applies to amount left after copay and deductible.
    - oop_remaining caps total patient responsibility; plan pays the rest.
    - oop_remaining is reduced by the final patient amount.
    - Return a report and append the same report to audit_log."""

    def __init__(self, deductible_remaining, oop_remaining, coinsurance, copay):
        self.deductible_remaining = float(deductible_remaining)
        self.oop_remaining = float(oop_remaining)
        self.coinsurance = float(coinsurance)
        self.copay = float(copay)
        self.allowed_rates = {}
        self.denied_codes = set()
        self.prior_auth_codes = set()
        self.copay_codes = set()
        self.audit_log = []

    def set_allowed_rate(self, code, amount):
        self.allowed_rates[code] = float(amount)

    def deny_code(self, code):
        self.denied_codes.add(code)

    def require_prior_auth(self, code):
        self.prior_auth_codes.add(code)

    def set_copay_code(self, code):
        self.copay_codes.add(code)

    def _round_money(self, value):
        return round(value + 0.0, 2)



GOLDEN COMPLETION:
----------------------------------------
    def adjudicate_line(self, code, billed_amount, prior_auth=False):
        billed_amount = float(billed_amount)
        if code in self.denied_codes:
            report = {'code': code, 'status': 'denied', 'allowed': 0.0, 'patient': 0.0, 'plan': 0.0}
            self.audit_log.append(report)
            return report
        if code in self.prior_auth_codes and not prior_auth:
            report = {'code': code, 'status': 'auth_required', 'allowed': 0.0, 'patient': 0.0, 'plan': 0.0}
            self.audit_log.append(report)
            return report
        allowed = min(billed_amount, self.allowed_rates.get(code, billed_amount))
        remaining_allowed = allowed
        copay_paid = min(self.copay, remaining_allowed) if code in self.copay_codes else 0.0
        remaining_allowed -= copay_paid
        deductible_paid = min(self.deductible_remaining, remaining_allowed)
        self.deductible_remaining = self._round_money(self.deductible_remaining - deductible_paid)
        remaining_allowed -= deductible_paid
        coinsurance_paid = remaining_allowed * self.coinsurance
        raw_patient = copay_paid + deductible_paid + coinsurance_paid
        patient = min(raw_patient, self.oop_remaining)
        self.oop_remaining = self._round_money(self.oop_remaining - patient)
        plan = self._round_money(allowed - patient)
        patient = self._round_money(patient)
        report = {'code': code, 'status': 'paid', 'allowed': self._round_money(allowed), 'patient': patient, 'plan': plan}
        self.audit_log.append(report)
        return report

class PaymentLedger:
    """Authorization/capture ledger.
    capture(auth_id, capture_id, amount, timestamp) must:
    - Treat capture_id as idempotency key: repeated capture_id returns the
      original report and does not append another audit entry.
    - Validate before mutating: auth must exist, amount must be positive, and
      amount cannot exceed the uncaptured authorization remainder.
    - Partial captures leave auth status 'partially_captured'; exact final
      capture sets status 'captured'.
    - For each new capture, fee = round(amount * fee_rate, 2) and
      net = amount - fee. gross_captured, fees_collected, and net_settled are
      cumulative totals.
    - Store and audit the exact report returned."""

    def __init__(self, fee_rate):
        self.fee_rate = float(fee_rate)
        self.authorizations = {}
        self.captures = {}
        self.gross_captured = 0.0
        self.fees_collected = 0.0
        self.net_settled = 0.0
        self.audit_log = []

    def authorize(self, auth_id, amount):
        if auth_id in self.authorizations:
            raise ValueError("duplicate authorization")
        self.authorizations[auth_id] = {
            'amount': float(amount),
            'captured': 0.0,
            'status': 'authorized',
        }

    def _remaining_auth(self, auth_id):
        auth = self.authorizations[auth_id]
        return round(auth['amount'] - auth['captured'], 2)

    def _money(self, value):
        return round(value + 0.0, 2)



GOLDEN COMPLETION:
----------------------------------------
    def capture(self, auth_id, capture_id, amount, timestamp):
        if capture_id in self.captures:
            return self.captures[capture_id]
        if auth_id not in self.authorizations:
            raise ValueError("unknown authorization")
        amount = self._money(float(amount))
        if amount <= 0:
            raise ValueError("capture amount must be positive")
        remaining = self._remaining_auth(auth_id)
        if amount > remaining:
            raise ValueError("capture exceeds authorization remainder")
        auth = self.authorizations[auth_id]
        fee = self._money(amount * self.fee_rate)
        net = self._money(amount - fee)
        auth['captured'] = self._money(auth['captured'] + amount)
        remaining_after = self._remaining_auth(auth_id)
        auth['status'] = 'captured' if remaining_after == 0 else 'partially_captured'
        self.gross_captured = self._money(self.gross_captured + amount)
        self.fees_collected = self._money(self.fees_collected + fee)
        self.net_settled = self._money(self.net_settled + net)
        report = {'auth_id': auth_id, 'capture_id': capture_id, 'amount': amount, 'fee': fee, 'net': net, 'remaining_auth': remaining_after, 'status': auth['status'], 'ts': timestamp}
        self.captures[capture_id] = report
        self.audit_log.append({'action': 'capture', 'report': report})
        return report

class TransferPicker:
    """Warehouse transfer picking.
    pick(sku, qty, pick_id, ship_date) must:
    - Treat pick_id as idempotency key: replay returns the original report and
      does not reserve more stock or append another audit entry.
    - Ignore expired batches where expiry <= ship_date.
    - Allocate by earliest expiry first (FEFO) using available qty
      batch['qty'] - batch['reserved'].
    - Validate enough available stock before mutating any batch.
    - On success, increase each batch's reserved quantity, return exact
      allocations, and append one audit entry.
    - On insufficient stock or invalid qty, raise ValueError before mutation."""

    def __init__(self):
        self.stock = {}
        self.picks = {}
        self.audit_log = []

    def add_batch(self, sku, batch_id, qty, expiry):
        self.stock.setdefault(sku, []).append({
            'batch_id': batch_id,
            'qty': qty,
            'reserved': 0,
            'expiry': expiry,
        })

    def available_quantity(self, sku, ship_date):
        total = 0
        for batch in self.stock.get(sku, []):
            if batch['expiry'] > ship_date:
                total += batch['qty'] - batch['reserved']
        return total

    def _snapshot(self, sku):
        return [dict(batch) for batch in self.stock.get(sku, [])]



GOLDEN COMPLETION:
----------------------------------------
    def pick(self, sku, qty, pick_id, ship_date):
        if pick_id in self.picks:
            return self.picks[pick_id]
        if qty <= 0:
            raise ValueError("pick quantity must be positive")
        candidates = []
        for batch in self.stock.get(sku, []):
            available = batch['qty'] - batch['reserved']
            if batch['expiry'] > ship_date and available > 0:
                candidates.append((batch['expiry'], batch, available))
        candidates.sort(key=lambda item: item[0])
        if sum(available for _, _, available in candidates) < qty:
            raise ValueError("insufficient available stock")
        remaining = qty
        allocations = []
        for _, batch, available in candidates:
            if remaining == 0:
                break
            take = min(available, remaining)
            batch['reserved'] += take
            remaining -= take
            allocations.append({'batch_id': batch['batch_id'], 'qty': take, 'expiry': batch['expiry']})
        report = {'pick_id': pick_id, 'sku': sku, 'qty': qty, 'ship_date': ship_date, 'allocations': allocations}
        self.picks[pick_id] = report
        self.audit_log.append({'action': 'pick', 'report': report})
        return report

class UsageBilling:
    """Usage invoice closing.
    close_period(period_id) must:
    - Be idempotent: if the period was already invoiced, return the same
      invoice object and do not append audit entries again.
    - Sum only events for period_id whose billed flag is False.
    - subtotal = base_fee + max(units - included_units, 0) * overage_rate.
    - If subtotal is below minimum_commit, invoice subtotal is minimum_commit.
    - Apply account credits in FIFO order after the minimum commit adjustment.
      Credits reduce amount_due but never below 0; partially used credits keep
      their remaining amount.
    - Mark the period's included events billed only after the invoice is built.
    - Return and audit an invoice dict with exact credit application details."""

    def __init__(self, base_fee, included_units, overage_rate, minimum_commit):
        self.base_fee = float(base_fee)
        self.included_units = included_units
        self.overage_rate = float(overage_rate)
        self.minimum_commit = float(minimum_commit)
        self.events = []
        self.credits = []
        self.invoices = {}
        self.audit_log = []

    def record_usage(self, event_id, period_id, units):
        self.events.append({'event_id': event_id, 'period_id': period_id, 'units': units, 'billed': False})

    def add_credit(self, credit_id, amount):
        self.credits.append({'credit_id': credit_id, 'remaining': float(amount)})

    def _money(self, value):
        return round(value + 0.0, 2)



GOLDEN COMPLETION:
----------------------------------------
    def close_period(self, period_id):
        if period_id in self.invoices:
            return self.invoices[period_id]
        period_events = [event for event in self.events if event['period_id'] == period_id and not event['billed']]
        units = sum(event['units'] for event in period_events)
        overage_units = max(units - self.included_units, 0)
        raw_subtotal = self.base_fee + overage_units * self.overage_rate
        subtotal = self._money(max(raw_subtotal, self.minimum_commit))
        remaining_due = subtotal
        applied = []
        for credit in self.credits:
            if remaining_due == 0:
                break
            if credit['remaining'] <= 0:
                continue
            use = min(credit['remaining'], remaining_due)
            credit['remaining'] = self._money(credit['remaining'] - use)
            remaining_due = self._money(remaining_due - use)
            applied.append({'credit_id': credit['credit_id'], 'amount': self._money(use)})
        invoice = {'period_id': period_id, 'units': units, 'overage_units': overage_units, 'subtotal': subtotal, 'credits_applied': applied, 'amount_due': remaining_due}
        self.invoices[period_id] = invoice
        for event in period_events:
            event['billed'] = True
        self.audit_log.append({'action': 'close_period', 'invoice': invoice})
        return invoice

    - Each date costs 1.0 unit except dates in half_days cost 0.5.
    - Vacation requests require enough vacation balance and never create unpaid.
    - Sick requests consume sick balance first; any remainder becomes unpaid
      leave and increments unpaid_used.
    - Store a request report and append one audit entry on success."""

    def __init__(self):
        self.employees = {}
        self.blackout_dates = set()
        self.requests = {}
        self.audit_log = []

    def add_employee(self, employee, vacation, sick):
        self.employees[employee] = {
            'vacation': float(vacation),
            'sick': float(sick),
            'unpaid_used': 0.0,
        }

    def add_blackout(self, date):
        self.blackout_dates.add(date)

    def _units(self, dates, half_days):
        half = set(half_days)
        return sum(0.5 if date in half else 1.0 for date in dates)

    def _snapshot(self, employee):
        return dict(self.employees[employee])



GOLDEN COMPLETION:
----------------------------------------
    def request_leave(self, employee, request_id, dates, leave_type, half_days=()):
        if request_id in self.requests:
            return self.requests[request_id]
        if employee not in self.employees:
            raise ValueError("unknown employee")
        if not dates:
            raise ValueError("leave dates required")
        if leave_type not in ('vacation', 'sick'):
            raise ValueError("unknown leave type")
        if any(date in self.blackout_dates for date in dates):
            raise ValueError("blackout date requested")
        units = self._units(dates, half_days)
        balance = self.employees[employee]
        used_vacation = used_sick = unpaid = 0.0
        if leave_type == 'vacation':
            if units > balance['vacation']:
                raise ValueError("insufficient vacation balance")
            balance['vacation'] = round(balance['vacation'] - units, 2)
            used_vacation = units
        else:
            used_sick = min(balance['sick'], units)
            unpaid = round(units - used_sick, 2)
            balance['sick'] = round(balance['sick'] - used_sick, 2)
            balance['unpaid_used'] = round(balance['unpaid_used'] + unpaid, 2)
        report = {'employee': employee, 'request_id': request_id, 'dates': list(dates), 'leave_type': leave_type, 'units': units, 'used_vacation': used_vacation, 'used_sick': used_sick, 'unpaid': unpaid}
        self.requests[request_id] = report
        self.audit_log.append({'action': 'request_leave', 'report': report})
        return report

class EntitlementLedger:
    """Feature entitlement consumption.
    consume(user, feature, units, request_id, timestamp) must:
    - Be idempotent by request_id: replay returns the original result object and
      does not consume more units or append another audit entry.
    - Active denials override grants. Denials are active for
      valid_from <= timestamp < valid_until, with valid_until=None meaning forever.
    - Use only active grants for the same user and feature.
    - Validate enough active remaining units before mutating any grant.
    - Consume from grants expiring soonest first; grants with valid_until=None
      are consumed after dated grants.
    - Return exact allocations and remaining active units after consumption."""

    def __init__(self):
        self.grants = []
        self.denials = []
        self.consumptions = {}
        self.audit_log = []

    def grant(self, grant_id, user, feature, units, valid_from, valid_until=None):
        self.grants.append({
            'grant_id': grant_id, 'user': user, 'feature': feature,
            'remaining': units, 'valid_from': valid_from, 'valid_until': valid_until,
        })

    def deny(self, user, feature, valid_from, valid_until=None):
        self.denials.append({'user': user, 'feature': feature, 'valid_from': valid_from, 'valid_until': valid_until})

    def _active(self, row, timestamp):
        if timestamp < row['valid_from']:
            return False
        if row['valid_until'] is not None and timestamp >= row['valid_until']:
            return False
        return True

    def _snapshot(self):
        return [dict(grant) for grant in self.grants]



GOLDEN COMPLETION:
----------------------------------------
    def consume(self, user, feature, units, request_id, timestamp):
        if request_id in self.consumptions:
            return self.consumptions[request_id]
        if units <= 0:
            raise ValueError("units must be positive")
        for denial in self.denials:
            if denial['user'] == user and denial['feature'] == feature and self._active(denial, timestamp):
                raise ValueError("active denial")
        candidates = [grant for grant in self.grants if grant['user'] == user and grant['feature'] == feature and self._active(grant, timestamp) and grant['remaining'] > 0]
        candidates.sort(key=lambda grant: (grant['valid_until'] is None, grant['valid_until'] if grant['valid_until'] is not None else 0))
        if sum(grant['remaining'] for grant in candidates) < units:
            raise ValueError("insufficient entitlement")
        remaining_to_consume = units
        allocations = []
        for grant in candidates:
            if remaining_to_consume == 0:
                break
            take = min(grant['remaining'], remaining_to_consume)
            grant['remaining'] -= take
            remaining_to_consume -= take
            allocations.append({'grant_id': grant['grant_id'], 'units': take})
        remaining_active = sum(grant['remaining'] for grant in candidates)
        result = {'request_id': request_id, 'user': user, 'feature': feature, 'units': units, 'allocations': allocations, 'remaining_active': remaining_active}
        self.consumptions[request_id] = result
        self.audit_log.append({'action': 'consume', 'result': result})
        return result

class LWWProjection:
    """Last-Writer-Wins projection. Each key tracks the timestamp
    of the last write. A write with an older timestamp is ignored.
    Deletes are tracked as tombstones with timestamps.
    apply_event handles 'set' and 'delete' actions with timestamp 'ts'.
    For 'set': ignore if tombstone >= ts or existing write >= ts.
    For 'delete': ignore if existing write > ts (strictly greater).
    If delete ts >= write ts, the key is removed and tombstone recorded.
    project(events) resets all state and applies events in order."""

    def __init__(self):
        self.data = {}
        self.timestamps = {}     # key -> timestamp of last set
        self.tombstones = {}     # key -> timestamp of delete

    def _is_stale_set(self, key, ts):
        if key in self.tombstones and self.tombstones[key] >= ts:
            return True
        if key in self.timestamps and self.timestamps[key] >= ts:
            return True
        return False

    def project(self, events):
        self.data, self.timestamps, self.tombstones = {}, {}, {}
        for event in events:
            self.apply_event(event)



GOLDEN COMPLETION:
----------------------------------------
    def apply_event(self, event):
        key, ts, action = event['key'], event['ts'], event['action']
        if action == 'set':
            if self._is_stale_set(key, ts):
                return
            self.data[key], self.timestamps[key] = event['value'], ts
        elif action == 'delete':
            if self.timestamps.get(key, -1) > ts:
                return
            self.data.pop(key, None)
            self.tombstones[key] = ts
            self.timestamps.pop(key, None)

class MilestoneEscrow:
    """Construction milestone escrow.
    record_progress(milestone_id, release_id, percent_complete, defects_open, timestamp)
    must:
    - Be idempotent by release_id.
    - Validate before mutating: milestone exists and percent_complete is between
      the current percent and 100 inclusive.
    - Pay only the newly completed delta percent, not cumulative percent.
    - Withhold holdback_rate from each newly earned amount.
    - If defects_open is True, the non-holdback payable is blocked_payable
      instead of paid_now; progress and holdback still update.
    - If defects_open is False, non-holdback payable increases paid_to_vendor.
    - Return and audit the exact release report."""

    def __init__(self):
        self.milestones = {}
        self.releases = {}
        self.audit_log = []

    def add_milestone(self, milestone_id, amount, holdback_rate):
        self.milestones[milestone_id] = {
            'amount': float(amount),
            'holdback_rate': float(holdback_rate),
            'percent_complete': 0.0,
            'earned': 0.0,
            'holdback_held': 0.0,
            'paid_to_vendor': 0.0,
            'blocked_payable': 0.0,
        }

    def _money(self, value):
        return round(value + 0.0, 2)

    def _snapshot(self, milestone_id):
        return dict(self.milestones[milestone_id])



GOLDEN COMPLETION:
----------------------------------------
    def record_progress(self, milestone_id, release_id, percent_complete, defects_open, timestamp):
        if release_id in self.releases:
            return self.releases[release_id]
        if milestone_id not in self.milestones:
            raise ValueError("unknown milestone")
        milestone = self.milestones[milestone_id]
        percent_complete = float(percent_complete)
        if percent_complete < milestone['percent_complete'] or percent_complete > 100:
            raise ValueError("invalid progress percent")
        delta_percent = percent_complete - milestone['percent_complete']
        gross = self._money(milestone['amount'] * delta_percent / 100)
        holdback = self._money(gross * milestone['holdback_rate'])
        payable = self._money(gross - holdback)
        paid_now = 0.0 if defects_open else payable
        blocked = payable if defects_open else 0.0
        milestone['percent_complete'] = percent_complete
        milestone['earned'] = self._money(milestone['earned'] + gross)
        milestone['holdback_held'] = self._money(milestone['holdback_held'] + holdback)
        milestone['paid_to_vendor'] = self._money(milestone['paid_to_vendor'] + paid_now)
        milestone['blocked_payable'] = self._money(milestone['blocked_payable'] + blocked)
        report = {'milestone_id': milestone_id, 'release_id': release_id, 'delta_percent': delta_percent, 'gross_earned': gross, 'holdback': holdback, 'paid_now': paid_now, 'blocked_payable': blocked, 'percent_complete': percent_complete, 'ts': timestamp}
        self.releases[release_id] = report
        self.audit_log.append({'action': 'record_progress', 'report': report})
        return report

class PharmacyRefill:
    """Prescription refill policy using integer service days.
    fill(rx_id, fill_id, requested_days, service_day, expedited=False) must:
    - Be idempotent by fill_id.
    - Validate before mutating: rx exists, requested_days > 0, prescription is active.
    - First fill does not consume a refill; later fills decrement refills_remaining.
    - Standard fills cannot exceed plan_max_days. Expedited fills may exceed
      plan_max_days, but controlled substances can never be expedited.
    - Too-soon rule: after any fill, next eligible day is
      last_fill_day + int(last_days_supply * refill_threshold).
    - Refill attempts before next eligible day fail before mutation.
    - Copay is per fill, not per day, and depends on the prescription tier.
    - Return and audit the exact fill report."""

    TIER_COPAY = {'generic': 5.0, 'preferred': 20.0, 'specialty': 75.0}

    def __init__(self, plan_max_days=30, refill_threshold=0.8):
        self.plan_max_days = plan_max_days
        self.refill_threshold = refill_threshold
        self.prescriptions = {}
        self.fills = {}
        self.audit_log = []

    def add_prescription(self, rx_id, daily_qty, refills, tier='generic', controlled=False):
        self.prescriptions[rx_id] = {
            'daily_qty': daily_qty,
            'refills_remaining': refills,
            'tier': tier,
            'controlled': controlled,
            'active': True,
            'last_fill_day': None,
            'last_days_supply': None,
        }

    def deactivate(self, rx_id):
        self.prescriptions[rx_id]['active'] = False

    def _next_eligible_day(self, rx):
        if rx['last_fill_day'] is None:
            return None
        return rx['last_fill_day'] + int(rx['last_days_supply'] * self.refill_threshold)

    def _snapshot(self, rx_id):
        return dict(self.prescriptions[rx_id])



GOLDEN COMPLETION:
----------------------------------------
    def fill(self, rx_id, fill_id, requested_days, service_day, expedited=False):
        if fill_id in self.fills:
            return self.fills[fill_id]
        if rx_id not in self.prescriptions:
            raise ValueError("unknown prescription")
        rx = self.prescriptions[rx_id]
        if not rx['active']:
            raise ValueError("inactive prescription")
        if requested_days <= 0:
            raise ValueError("requested days must be positive")
        if requested_days > self.plan_max_days and not expedited:
            raise ValueError("days supply exceeds plan maximum")
        if expedited and rx['controlled']:
            raise ValueError("controlled prescriptions cannot be expedited")
        next_eligible = self._next_eligible_day(rx)
        is_refill = rx['last_fill_day'] is not None
        if next_eligible is not None and service_day < next_eligible:
            raise ValueError("refill too soon")
        if is_refill and rx['refills_remaining'] <= 0:
            raise ValueError("no refills remaining")
        if is_refill:
            rx['refills_remaining'] -= 1
        qty = rx['daily_qty'] * requested_days
        copay = self.TIER_COPAY[rx['tier']]
        rx['last_fill_day'] = service_day
        rx['last_days_supply'] = requested_days
        report = {'rx_id': rx_id, 'fill_id': fill_id, 'days_supply': requested_days, 'quantity': qty, 'copay': copay, 'refills_remaining': rx['refills_remaining'], 'next_eligible_day': self._next_eligible_day(rx), 'expedited': expedited}
        self.fills[fill_id] = report
        self.audit_log.append({'action': 'fill', 'report': report})
        return report

class IncidentSLA:
    """Incident SLA state machine using integer minute timestamps.
    transition(event_id, new_state, timestamp, reason='') must:
    - Be idempotent by event_id.
    - Enforce transitions:
      open -> acknowledged
      acknowledged -> paused or mitigated
      paused -> acknowledged
      mitigated -> resolved
    - While paused, elapsed SLA time stops. Resuming from paused adds the pause
      duration to paused_total before changing state.
    - acknowledged_at is set only on the first acknowledged transition.
    - resolved_at is set only on resolved.
    - ack_breached is computed when first acknowledged:
      timestamp - created_at - paused_total > ack_target.
    - resolve_breached is computed when resolved:
      timestamp - created_at - paused_total > resolve_target.
    - Return and audit an exact transition report."""

    ALLOWED = {
        'open': {'acknowledged'},
        'acknowledged': {'paused', 'mitigated'},
        'paused': {'acknowledged'},
        'mitigated': {'resolved'},
        'resolved': set(),
    }

    def __init__(self, incident_id, created_at, ack_target, resolve_target):
        self.incident_id = incident_id
        self.created_at = created_at
        self.ack_target = ack_target
        self.resolve_target = resolve_target
        self.state = 'open'
        self.acknowledged_at = None
        self.resolved_at = None
        self.paused_at = None
        self.paused_total = 0
        self.ack_breached = False
        self.resolve_breached = False
        self.events = {}
        self.audit_log = []

    def _elapsed(self, timestamp):
        return timestamp - self.created_at - self.paused_total

    def _snapshot(self):
        return {
            'state': self.state,
            'acknowledged_at': self.acknowledged_at,
            'resolved_at': self.resolved_at,
            'paused_at': self.paused_at,
            'paused_total': self.paused_total,
            'ack_breached': self.ack_breached,
            'resolve_breached': self.resolve_breached,
        }



GOLDEN COMPLETION:
----------------------------------------
    def transition(self, event_id, new_state, timestamp, reason=''):
        if event_id in self.events:
            return self.events[event_id]
        if new_state not in self.ALLOWED[self.state]:
            raise ValueError(f"Cannot transition from {self.state} to {new_state}")
        old_state = self.state
        if old_state == 'paused' and new_state == 'acknowledged':
            self.paused_total += timestamp - self.paused_at
            self.paused_at = None
        if new_state == 'paused':
            self.paused_at = timestamp
        if new_state == 'acknowledged' and self.acknowledged_at is None:
            self.acknowledged_at = timestamp
            self.ack_breached = self._elapsed(timestamp) > self.ack_target
        if new_state == 'resolved':
            self.resolved_at = timestamp
            self.resolve_breached = self._elapsed(timestamp) > self.resolve_target
        self.state = new_state
        report = {'event_id': event_id, 'from': old_state, 'to': new_state, 'ts': timestamp, 'reason': reason, 'elapsed': self._elapsed(timestamp), 'paused_total': self.paused_total, 'ack_breached': self.ack_breached, 'resolve_breached': self.resolve_breached}
        self.events[event_id] = report
        self.audit_log.append({'action': 'transition', 'report': report})
        return report

class LRUCache:
    def __init__(self, max_size):
        self.max_size = max_size
        self.cache = {}
        self.access_order = []  # most-recently-used at the END

    def _touch(self, key):
        if key in self.access_order:
            self.access_order.remove(key)
        self.access_order.append(key)

    def get(self, key):
        if key not in self.cache:
            return None
        self._touch(key)
        return self.cache[key]



GOLDEN COMPLETION:
----------------------------------------
    def set(self, key, value):
        if key in self.cache:
            self.cache[key] = value
            self._touch(key)
            return
        if len(self.cache) >= self.max_size:
            evicted = self.access_order.pop(0)
            del self.cache[evicted]
        self.cache[key] = value
        self._touch(key)

class DependentValidator:
    """Validators can depend on others. Only runs if all deps passed.
    Skipped validators do not appear in errors."""

    def __init__(self):
        self.validators = {}
        self.order = []

    def add(self, name, fn, depends_on=None):
        self.validators[name] = {'fn': fn, 'deps': depends_on or []}
        self.order.append(name)



GOLDEN COMPLETION:
----------------------------------------
    def validate(self, data):
        errors, passed = {}, set()
        for name in self.order:
            v = self.validators[name]
            if not all(d in passed for d in v['deps']):
                continue
            ok, msg = v['fn'](data)
            if ok:
                passed.add(name)
            else:
                errors[name] = msg
        return errors, passed

GOLDEN COMPLETION:

        for field, fn in self.normalizers.items():

class EscalatingWorkflow:
    def __init__(self, approver_chain, clock=None):
        self.chain = list(approver_chain)
        self.idx = 0
        self.status = 'pending'
        self.history = []
        self.clock = clock or (lambda: 0)

    def current_approver(self):
        return self.chain[self.idx] if self.idx < len(self.chain) else None

    def decide(self, approver, decision):
        if self.status != 'pending':
            return self.status
        if approver != self.current_approver():
            return 'wrong_approver'
        self.history.append((approver, decision, self.clock()))
        if decision == 'approve':
            self.status = 'approved'
        elif decision == 'reject':
            self.status = 'rejected'
        return self.status



GOLDEN COMPLETION:
----------------------------------------
    def escalate(self):
        if self.status != 'pending':
            return self.status
        cur = self.current_approver()
        self.history.append((cur, 'escalated', self.clock()))
        self.idx += 1
        if self.idx >= len(self.chain):
            self.status = 'rejected'
        return self.status

class TransactionalWorkflow:
    def __init__(self):
        self.steps = []
        self.completed = []
        self.rolled_back = []
        self.status = 'idle'

    def add_step(self, name, do_fn, undo_fn):
        self.steps.append((name, do_fn, undo_fn))



GOLDEN COMPLETION:
----------------------------------------
    def execute(self):
        self.status, self.completed, self.rolled_back, done = 'running', [], [], []
        for name, do_fn, undo_fn in self.steps:
            if not do_fn():
                for n, u in reversed(done):
                    u(); self.rolled_back.append(n)
                self.status = 'rolled_back'
                return False
            self.completed.append(name)
            done.append((name, undo_fn))
        self.status = 'completed'
        return True

class WorkflowStateMachine:
    def __init__(self, initial_state):
        self.state = initial_state
        self.transitions = {}
        self.history = [initial_state]

    def allow_transition(self, from_state, to_state):
        if from_state not in self.transitions:
            self.transitions[from_state] = set()
        self.transitions[from_state].add(to_state)



GOLDEN COMPLETION:
----------------------------------------
    def advance(self, to_state):
        allowed = self.transitions.get(self.state, set())
        if to_state not in allowed:
            return False
        self.state = to_state
        self.history.append(to_state)
        return True

    def can_advance(self, to_state):
        return to_state in self.transitions.get(self.state, set())

class QuorumApproval:
    """Requires N approvals from M approvers.
    Auto-approved when threshold met.
    Auto-rejected when impossible to reach threshold."""

    def __init__(self, approvers, threshold):
        self.approvers = set(approvers)
        self.threshold = threshold
        self.votes = {}
        self.status = 'pending'



GOLDEN COMPLETION:
----------------------------------------
    def vote(self, approver, decision):
        if self.status != 'pending': return self.status
        if approver not in self.approvers: return 'invalid_approver'
        if approver in self.votes: return 'already_voted'
        self.votes[approver] = decision

        if approvals >= self.threshold:
            self.status = 'approved'
        elif approvals + (len(self.approvers) - len(self.votes)) < self.threshold:
            self.status = 'rejected'
        return self.status

class PercentileCalculator:
    """Nearest-rank method. For percentile P and N values:
    rank = ceil(P / 100 * N), 1-indexed.
    P=0 returns minimum. P=100 returns maximum."""

    def __init__(self, values):
        self.sorted_values = sorted(values)
        self.n = len(self.sorted_values)



GOLDEN COMPLETION:
----------------------------------------
    def percentile(self, p):
        if self.n == 0:
            return None
        if p <= 0:
            return self.sorted_values[0]
        if p >= 100:
            return self.sorted_values[-1]
        rank = math.ceil(p / 100 * self.n)
        return self.sorted_values[rank - 1]

        return self.percentile(50)

class Histogram:
    """Fixed-width histogram. Buckets cover [lower, lower+width).
    Values below start -> underflow. Values >= end -> overflow."""

    def __init__(self, start, width, num_buckets):
        self.start = start
        self.width = width
        self.num_buckets = num_buckets
        self.buckets = [0] * num_buckets
        self.underflow = 0
        self.overflow = 0



GOLDEN COMPLETION:
----------------------------------------
    def add(self, value):
        if value < self.start:
            self.underflow += 1
        elif value >= self.start + self.width * self.num_buckets:
            self.overflow += 1
        else:
            idx = int((value - self.start) / self.width)
            self.buckets[idx] += 1

    def get_buckets(self):
        return [(self.start + i * self.width, self.buckets[i])
                for i in range(self.num_buckets)]

class ExponentialMovingStats:
    """EMA mean and variance with custom alpha.
    mean = alpha * value + (1-alpha) * mean
    var = alpha * (value - OLD_mean)^2 + (1-alpha) * var
    Note: variance uses PREVIOUS mean (before update).
    First value initializes mean; variance starts at 0."""

    def __init__(self, alpha):
        self.alpha = alpha
        self.mean = None
        self.variance = 0.0
        self.count = 0



GOLDEN COMPLETION:
----------------------------------------
    def update(self, value):
        self.count += 1
        if self.mean is None:
            self.mean = float(value)
            self.variance = 0.0
            return
        old_mean = self.mean
        self.mean = self.alpha * value + (1 - self.alpha) * old_mean
        diff = value - old_mean
        self.variance = self.alpha * diff * diff + (1 - self.alpha) * self.variance

    def std_dev(self):
        return self.variance ** 0.5



SUFFIX CODE:
----------------------------------------

assert ema.mean == 10.0
assert ema.variance == 0.0

assert ema.mean == 15.0
# var = 0.5 * (20-10)^2 + 0.5*0 = 50.0 (uses OLD mean=10)
assert ema.variance == 50.0

ema.update(15)
assert ema.mean == 15.0
# var = 0.5 * (15-15)^2 + 0.5*50 = 25.0

class DecayingCounter:
    """Counter decays exponentially: count *= decay_factor^elapsed.
    increment() adds to the current decayed count."""

    def __init__(self, decay_factor, clock=None):
        self.decay_factor = decay_factor
        self.clock = clock or (lambda: __import__('time').monotonic())
        self.counters = {}   # key -> (count, last_update)



GOLDEN COMPLETION:
----------------------------------------
    def _decay(self, key):
        if key not in self.counters: return 0.0
        count, last = self.counters[key]
        now = self.clock()
        decayed = count * (self.decay_factor ** (now - last))
        self.counters[key] = (decayed, now)
        return decayed

    def increment(self, key, amount=1):
        self.counters[key] = (self._decay(key) + amount, self.clock())

    def get_count(self, key):
        return self._decay(key) if key in self.counters else 0.0

class TimeWindowStats:
    """Statistics over a time-based sliding window.
    Values older than window_seconds are evicted."""

    def __init__(self, window_seconds, clock=None):
        self.window = window_seconds
        self.clock = clock or (lambda: __import__('time').monotonic())
        self.values = []  # (timestamp, value)

    def _evict(self):
        cutoff = self.clock() - self.window
        self.values = [(t, v) for t, v in self.values if t > cutoff]



GOLDEN COMPLETION:
----------------------------------------
    def add(self, value):
        self._evict()
        self.values.append((self.clock(), value))

    def mean(self):
        self._evict()
        if not self.values:
            return None
        return sum(v for _, v in self.values) / len(self.values)

    def count(self):
        self._evict()
        return len(self.values)
