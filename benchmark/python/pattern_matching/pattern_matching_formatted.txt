class CustomSerializer:
    def serialize_int(self, value):
        return str(value)

    def serialize_float(self, value):
        if value == int(value):
            return f'{int(value)}.0'
        return str(value)

    def serialize_str(self, value):
        escaped = value.replace("\\", "\\\\").replace("'", "\\'")
        return f"'{escaped}'"

    def serialize_bool(self, value):
        return 'true' if value else 'false'

    def serialize_none(self, value):
        return 'null'

    def serialize_list(self, value):
        items = ', '.join(self.serialize(v) for v in value)
        return f'[{items}]'

    def serialize(self, value):
        if value is None:
            return self.serialize_none(value)
        if isinstance(value, bool):
            return self.serialize_bool(value)
        if isinstance(value, int):
            return self.serialize_int(value)
        if isinstance(value, float):
            return self.serialize_float(value)
        if isinstance(value, str):
            return self.serialize_str(value)
        if isinstance(value, list):
            return self.serialize_list(value)
        if isinstance(value, dict):
            return self.serialize_dict(value)
        raise TypeError(f'Cannot serialize {type(value)}')



GOLDEN COMPLETION:

            k = self.serialize(key)
            v = self.serialize(value[key])
            pairs.append(f'{k}: {v}')
        return '{' + ', '.join(pairs) + '}'

    if isinstance(data, str):
        data = data.encode()

        chunk = data[i:i + 3]

def url_encode(text, safe=''):
    out = []
    for b in text.encode('utf-8'):
        ch = chr(b)
        if b < 128 and (is_unreserved(ch) or ch in safe):
            out.append(ch)
        else:
            out.append('%' + hex_byte(b))
    return ''.join(out)

def html_escape(text, quote=True, slash=True):
    out = []
    for ch in text:
        if ch in BASIC_HTML:
            out.append(BASIC_HTML[ch])
        elif quote and ch in QUOTE_HTML:
            out.append(QUOTE_HTML[ch])

            out.append('&#47;')

    return ''.join(out)

def is_balanced_parens(text):
    count = 0
    for ch in text:
        if ch == '(':
            count += 1
        elif ch == ')':
            count -= 1
        if count < 0:
            return False
    return count == 0

def is_balanced_brackets(text):
    stack = []
    pairs = {'(': ')', '[': ']', '{': '}'}
    for ch in text:
        if ch in pairs:
            stack.append(pairs[ch])
        elif ch in pairs.values():
            if not stack or stack[-1] != ch:
                return False
            stack.pop()
    return len(stack) == 0



GOLDEN COMPLETION:
----------------------------------------
def is_balanced_code(text):
    stack = []
    pairs = {'(': ')', '[': ']', '{': '}'}
    in_string = False
    string_char = None
    i = 0
    while i < len(text):
        ch = text[i]
        if in_string:
            if ch == '\\' and i + 1 < len(text):
                i += 2
                continue
            if ch == string_char:
                in_string = False
        else:
            if ch in ('"', "'"):
                in_string = True
                string_char = ch
            elif ch in pairs:
                stack.append(pairs[ch])
            elif ch in pairs.values():
                if not stack or stack[-1] != ch:
                    return False
                stack.pop()
        i += 1
    return len(stack) == 0 and not in_string

----------------------------------------

assert is_balanced_parens('(()') == False
assert is_balanced_brackets('{[()]}') == True

def format_integer(n):
    negative = n < 0
    s = str(abs(n))
    groups = []
    while s:
        groups.append(s[-3:])
        s = s[:-3]
    result = ','.join(reversed(groups))
    return f'-{result}' if negative else result

def format_decimal(n, places=2):
    negative = n < 0
    n = abs(n)
    integer_part = int(n)
    frac = round(n - integer_part, places)
    frac_str = str(frac)[2:].ljust(places, '0')[:places]
    int_str = format_integer(integer_part).lstrip('-')
    result = f'{int_str}.{frac_str}'
    return f'-{result}' if negative else result



GOLDEN COMPLETION:
----------------------------------------
def format_currency(amount, symbol='$', places=2):
    negative = amount < 0
    formatted = format_decimal(abs(amount), places)
    if negative:
        return f'-{symbol}{formatted}'
    return f'{symbol}{formatted}'

def format_percentage(value, places=1):
    pct = value * 100
    formatted = format_decimal(abs(pct), places)
    if pct < 0:
        return f'-{formatted}%'
    return f'{formatted}%'

def hamming_distance(s1, s2):
    if len(s1) != len(s2):
        raise ValueError('Strings must be equal length')
    return sum(c1 != c2 for c1, c2 in zip(s1, s2))

def longest_common_prefix(s1, s2):
    i = 0
    while i < len(s1) and i < len(s2) and s1[i] == s2[i]:
        i += 1
    return s1[:i]



GOLDEN COMPLETION:
----------------------------------------
def weighted_edit_distance(s1, s2, insert_cost=1, delete_cost=1, replace_cost=2):
    m, n = len(s1), len(s2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        dp[i][0] = i * delete_cost
    for j in range(n + 1):
        dp[0][j] = j * insert_cost
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = min(
                    dp[i-1][j] + delete_cost,
                    dp[i][j-1] + insert_cost,
                    dp[i-1][j-1] + replace_cost
                )
    return dp[m][n]



SUFFIX CODE:

assert longest_common_prefix('abcdef', 'abcxyz') == 'abc'

# Weighted edit distance: insert=1, delete=1, but REPLACE=2
# So it's cheaper to delete+insert (cost 2) than replace (cost 2) — equal!
# But for 'abc' → 'axc': replace b→x costs 2
assert weighted_edit_distance('abc', 'axc') == 2
# For 'abc' → 'axxc': delete b (1) + insert x (1) + insert x... no.
# 'abc' → 'axxc': replace b→x (2) + insert x (1) = 3
assert weighted_edit_distance('abc', 'axxc') == 3
# Standard Levenshtein('abc','axc') would be 1, ours is 2
assert weighted_edit_distance('abc', 'axc') != 1, 'Should not equal standard Levenshtein!'

def schedule_sequential(tasks):
    return sorted(tasks.keys())

def schedule_parallel(tasks):
    groups = {}
    for name, info in tasks.items():
        priority = info.get('priority', 0)
        groups.setdefault(priority, []).append(name)
    result = []
    for priority in sorted(groups.keys()):
        result.append(sorted(groups[priority]))
    return result



GOLDEN COMPLETION:
----------------------------------------
def schedule_with_deps(tasks):
    order, done, active = [], set(), set()
    def visit(n):
        if n in done: return
        if n in active: raise ValueError(f'Circular dependency: {n}')
        active.add(n)
        for d in sorted(tasks[n].get('deps', [])): visit(d)
        active.discard(n); done.add(n); order.append(n)
    for n in sorted(tasks): visit(n)
    return order



SUFFIX CODE:

def template_dollar(text, variables):
    result = []
    i = 0
    while i < len(text):
        if i + 1 < len(text) and text[i] == '\\' and text[i + 1] == '$':
            result.append('$')
            i += 2

def parse_inherited_config(text):
    config = {}
    current_section = 'default'
    parents = {}
    for line in text.strip().split('\n'):
        line = line.strip()
        if not line or line.startswith('#'):
            continue
        if line.startswith('[') and line.endswith(']'):
            header = line[1:-1].strip()
            if ':' in header:
                section, _, parent = header.partition(':')
                current_section = section.strip()
                parents[current_section] = parent.strip()
            else:
                current_section = header
            if current_section not in config:
                config[current_section] = {}
            continue
        key, _, value = line.partition('=')
        if current_section not in config:
            config[current_section] = {}
        config[current_section][key.strip()] = value.strip()
    # Apply inheritance
    for section, parent in parents.items():
        if parent in config:
            merged = dict(config[parent])
            merged.update(config[section])
            config[section] = merged
    return config

            sig = inspect.signature(func)

def normalize_dots(path):
    # Remove single-dot components from a path.
    parts = path.split('/')
    is_absolute = parts[0] == ''
    trailing_slash = path.endswith('/') and len(path) > 1
    filtered = [p for p in parts if p != '.']
    if is_absolute and (not filtered or filtered[0] != ''):
        filtered.insert(0, '')
    result = '/'.join(filtered) or ('/' if is_absolute else '.')
    if trailing_slash and not result.endswith('/'):
        result += '/'
    return result

def normalize_slashes(path):

    parts = path.split('/')

def normalize_path(path):
    # Full path normalization: collapse slashes, resolve '..', keep trailing slash, keep leading '..'.
    if not path:
        return '.'
    is_absolute = path.startswith('/')
    trailing_slash = path.endswith('/') and len(path) > 1
    parts = path.split('/')
    cleaned = [p for p in parts if p != '' and p != '.']
    result = []
    for part in cleaned:
        if part == '..':
            if result and result[-1] != '..':
                result.pop()
            elif not is_absolute:
                result.append('..')
        else:
            result.append(part)
    if is_absolute:
        out = '/' + '/'.join(result)
    else:
        out = '/'.join(result) or '.'
    if trailing_slash and not out.endswith('/'):
        out += '/'
    return out

def tokenize_simple(text):
    # Tokenize into numbers, identifiers, and single-char operators: + - * /
    tokens = []
    i = 0
    while i < len(text):
        if text[i].isspace():
            i += 1
        elif text[i].isdigit():
            start = i
            while i < len(text) and text[i].isdigit():
                i += 1
            tokens.append(('NUM', text[start:i]))
        elif text[i].isalpha() or text[i] == '_':
            start = i
            while i < len(text) and (text[i].isalnum() or text[i] == '_'):
                i += 1
            tokens.append(('ID', text[start:i]))
        elif text[i] in '+-*/':
            tokens.append(('OP', text[i]))
            i += 1
        else:
            tokens.append(('UNKNOWN', text[i]))
            i += 1
    return tokens

def tokenize_with_parens(text):

    # Serialize a tree to a preorder list with None for missing children.

        val = next(it, None)

        node = TNode(val)
        node.left = build()
        node.right = build()
        return node
    return build()

def tree_to_bracket(node):
    # Serialize tree to bracket notation: val(left,right) with '-' for None.
    if node is None:
        return '-'
    left_str = tree_to_bracket(node.left)
    right_str = tree_to_bracket(node.right)
    if node.left is None and node.right is None:
        return str(node.val)
    return f'{node.val}({left_str},{right_str})'

def bracket_to_tree(s):
    # Deserialize bracket notation back to a tree.
    pos = [0]
    def parse():
        if pos[0] >= len(s):
            return None
        if s[pos[0]] == '-':
            pos[0] += 1
            return None
        start = pos[0]
        while pos[0] < len(s) and s[pos[0]] not in '(,)':
            pos[0] += 1
        val = int(s[start:pos[0]])
        if pos[0] >= len(s) or s[pos[0]] != '(':
            return TNode(val)
        pos[0] += 1  # skip '('
        left = parse()
        pos[0] += 1  # skip ','
        right = parse()
        pos[0] += 1  # skip ')'
        return TNode(val, left, right)
    return parse()

class RingBuffer:
    def __init__(self, capacity):
        self.capacity = capacity
        self.buffer = [None] * capacity
        self.write_pos = 0
        self.count = 0

    def push(self, val):
        self.buffer[self.write_pos] = val
        self.write_pos = (self.write_pos + 1) % self.capacity
        if self.count < self.capacity:
            self.count += 1

    def to_list(self):
        if self.count < self.capacity:
            return self.buffer[:self.count]
        start = self.write_pos
        return [self.buffer[(start + i) % self.capacity] for i in range(self.capacity)]

    def peek(self):
        if self.count == 0:
            return None
        if self.count < self.capacity:
            return self.buffer[0]
        return self.buffer[self.write_pos % self.capacity]

def char_diff(s1, s2):
    # Return list of (char, status) where status is '=', '+', or '-'.
    # Simple: compares position by position, marks extras as '+' or '-'.
    result = []
    for i in range(max(len(s1), len(s2))):
        if i < len(s1) and i < len(s2):
            if s1[i] == s2[i]:
                result.append((s1[i], '='))
            else:
                result.append((s1[i], '-'))
                result.append((s2[i], '+'))
        elif i < len(s1):
            result.append((s1[i], '-'))
        else:
            result.append((s2[i], '+'))
    return result

def format_diff(diff_list):
    # Format a diff list into a readable string.
    # '=' chars shown as-is, '-' wrapped in [-x-], '+' wrapped in [+x+].
    parts = []
    for char, status in diff_list:
        if status == '=':
            parts.append(char)
        elif status == '-':
            parts.append(f'[-{char}-]')
        elif status == '+':
            parts.append(f'[+{char}+]')
    return ''.join(parts)

def smart_diff(s1, s2):
    # LCS-based diff: finds longest common subsequence, marks insertions/deletions.
    m, n = len(s1), len(s2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    result = []
    i, j = m, n
    while i > 0 or j > 0:
        if i > 0 and j > 0 and s1[i-1] == s2[j-1]:
            result.append((s1[i-1], '='))
            i -= 1
            j -= 1
        elif j > 0 and (i == 0 or dp[i][j-1] >= dp[i-1][j]):
            result.append((s2[j-1], '+'))
            j -= 1
        else:
            result.append((s1[i-1], '-'))
            i -= 1
    return list(reversed(result))

from collections import defaultdict

def topo_sort_min(edges):
    # Topological sort breaking ties by choosing the smallest node first.
    graph = defaultdict(list)
    in_degree = defaultdict(int)
    nodes = set()
    for u, v in edges:
        graph[u].append(v)
        in_degree[v] += 1
        nodes.add(u)
        nodes.add(v)
    for n in nodes:
        if n not in in_degree:
            in_degree[n] = 0
    result = []
    while True:
        ready = sorted([n for n in nodes if in_degree.get(n, 0) == 0 and n not in result])
        candidates = [n for n in ready if n not in result]
        if not candidates:
            break
        pick = candidates[0]  # smallest
        result.append(pick)
        for neighbor in graph[pick]:
            in_degree[neighbor] -= 1
    return result

def eval_left_to_right(expr):
    # Evaluate expression left to right, no precedence. Supports + - * /
    tokens = expr.split()
    result = int(tokens[0])
    i = 1
    while i < len(tokens):
        op = tokens[i]
        num = int(tokens[i + 1])
        if op == '+':
            result += num
        elif op == '-':
            result -= num
        elif op == '*':
            result *= num
        elif op == '/':
            result = result // num
        i += 2
    return result

def eval_with_precedence(expr):
    # Evaluate expression with standard precedence: * / before + -
    tokens = expr.split()
    nums = []
    ops = []
    for t in tokens:
        if t in '+-*/':
            ops.append(t)
        else:
            nums.append(int(t))
    # First pass: resolve * and /
    new_nums = [nums[0]]
    new_ops = []
    for i, op in enumerate(ops):
        if op in ('*', '/'):
            prev = new_nums.pop()
            if op == '*':
                new_nums.append(prev * nums[i + 1])
            else:
                new_nums.append(prev // nums[i + 1])
        else:
            new_nums.append(nums[i + 1])
            new_ops.append(op)
    # Second pass: resolve + and -
    result = new_nums[0]
    for i, op in enumerate(new_ops):
        if op == '+':
            result += new_nums[i + 1]
        else:
            result -= new_nums[i + 1]
    return result

class WeightedCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}
        self.weights = {}
        self.access_count = {}
        self.last_access = {}
        self.time = 0

    def get(self, key):
        if key not in self.cache:
            return -1
        self.access_count[key] += 1
        self.time += 1
        self.last_access[key] = self.time
        return self.cache[key]

    def put(self, key, value, weight=1):
        if self.capacity <= 0:
            return
        self.time += 1
        if key in self.cache:
            self.cache[key] = value
            self.weights[key] = weight
            self.access_count[key] += 1
            self.last_access[key] = self.time
            return
        if len(self.cache) >= self.capacity:
            min_score = None
            evict_key = None
            for k in self.cache:
                score = self.access_count[k] * self.weights[k]
                if min_score is None or score < min_score or (score == min_score and self.last_access[k] < self.last_access[evict_key]):
                    min_score = score
                    evict_key = k
            del self.cache[evict_key]
            del self.weights[evict_key]
            del self.access_count[evict_key]
            del self.last_access[evict_key]
        self.cache[key] = value
        self.weights[key] = weight
        self.access_count[key] = 1
        self.last_access[key] = self.time

def interpolate_simple(template, variables):
    # Replace ${name} with variables[name]. No nesting.
    result = []
    i = 0
    while i < len(template):
        if template[i] == '$' and i + 1 < len(template) and template[i+1] == '{':
            j = template.index('}', i + 2)
            key = template[i+2:j]
            result.append(str(variables.get(key, '')))
            i = j + 1
        else:
            result.append(template[i])
            i += 1
    return ''.join(result)

def interpolate_with_default(template, variables):
    # Replace ${name:default} with variables[name] or default if missing.
    result = []
    i = 0
    while i < len(template):
        if template[i] == '$' and i + 1 < len(template) and template[i+1] == '{':
            j = template.index('}', i + 2)
            spec = template[i+2:j]
            if ':' in spec:

            result.append(str(variables.get(key, default)))
            i = j + 1
        else:
            result.append(template[i])
            i += 1
    return ''.join(result)

def interpolate_nested(template, variables):
    # Replace ${...} with nesting support. Inner ${} resolved first.
    result = []
    i = 0
    while i < len(template):
        if template[i] == '$' and i + 1 < len(template) and template[i+1] == '{':
            # Find matching closing brace, handling nested ${}
            depth = 0
            j = i + 1
            while j < len(template):
                if template[j] == '{' and j > 0 and template[j-1] == '$':
                    depth += 1
                elif template[j] == '}':
                    depth -= 1
                    if depth == 0:
                        break
                j += 1
            inner = template[i+2:j]
            # Recursively resolve the inner content first
            resolved_key = interpolate_nested(inner, variables)
            result.append(str(variables.get(resolved_key, '')))
            i = j + 1
        else:
            result.append(template[i])
            i += 1
    return ''.join(result)

def fill_row_major(n, start=1):
    matrix = [[0] * n for _ in range(n)]
    val = start
    for i in range(n):
        for j in range(n):
            matrix[i][j] = val
            val += 1
    return matrix

def fill_diagonal(n, start=1):
    matrix = [[0] * n for _ in range(n)]
    val = start
    for d in range(2 * n - 1):
        if d % 2 == 0:
            r = min(d, n - 1)
            c = d - r
            while r >= 0 and c < n:
                matrix[r][c] = val
                val += 1
                r -= 1
                c += 1
        else:
            c = min(d, n - 1)
            r = d - c
            while c >= 0 and r < n:
                matrix[r][c] = val
                val += 1
                r += 1
                c -= 1
    return matrix

def spiral_fill(n, start=1):
    m = [[0]*n for _ in range(n)]
    val, t, b, l, r = start, 0, n-1, 0, n-1
    while t <= b and l <= r:
        for j in range(l, r+1): m[t][j] = val; val += 1
        t += 1
        for i in range(t, b+1): m[i][r] = val; val += 1
        r -= 1
        for j in range(r, l-1, -1): m[b][j] = val; val += 1
        b -= 1
        for i in range(b, t-1, -1): m[i][l] = val; val += 1
        l += 1
    return m

def to_base2_lsf(n):
    """Convert to base 2, digits in least-significant-first order."""
    if n == 0:
        return [0]
    digits = []
    while n > 0:
        digits.append(n % 2)
        n //= 2
    return digits

def from_base2_lsf(digits):
    """Convert from base 2 LSF digits back to integer."""
    result = 0
    for i, d in enumerate(digits):
        result += d * (2 ** i)
    return result

def greedy_color_sequential(adj, nodes):
    """Color nodes in the given order, assigning smallest available color."""
    color = {}
    for node in nodes:
        used = set()
        for neighbor in adj.get(node, []):
            if neighbor in color:
                used.add(color[neighbor])
        c = 0
        while c in used:
            c += 1
        color[node] = c
    return color

def greedy_color_by_id(adj):
    """Color nodes in ascending node-id order."""
    nodes = sorted(adj.keys())
    return greedy_color_sequential(adj, nodes)

    Uses smallest available color for each node."""
    nodes = sorted(adj.keys(), key=lambda n: (-len(adj.get(n, [])), n))
    return greedy_color_sequential(adj, nodes)

class WindowedAggregator:
    def __init__(self, reducer, initial, window_size):
        self.reducer, self.initial = reducer, initial
        self.window_size, self.events = window_size, []

    def add(self, value, timestamp):
        self.events.append((timestamp, value))
        cutoff = timestamp - self.window_size
        self.events = [(t, v) for t, v in self.events if t > cutoff]

    def result(self):
        acc = self.initial
        for _, v in self.events:
            acc = self.reducer(acc, v)
        return acc

def build_failure(pattern):
    """Build KMP failure/prefix function."""
    m = len(pattern)
    fail = [0] * m
    k = 0
    for i in range(1, m):
        while k > 0 and pattern[k] != pattern[i]:
            k = fail[k - 1]
        if pattern[k] == pattern[i]:
            k += 1
        fail[i] = k
    return fail

def kmp_search(text, pattern):
    """Find all occurrences of pattern in text. Returns list of start indices."""
    if not pattern:
        return []
    fail = build_failure(pattern)
    results = []
    k = 0
    for i in range(len(text)):
        while k > 0 and pattern[k] != text[i]:
            k = fail[k - 1]
        if pattern[k] == text[i]:
            k += 1
        if k == len(pattern):
            results.append(i - len(pattern) + 1)
            k = fail[k - 1]
    return results

    """Return days in month (1-indexed, 1-12)."""
    return MONTH_LENGTHS[month - 1]

    return sum(MONTH_LENGTHS)

def date_to_day_of_year(year, month, day):
    """Convert (year, month, day) to day-of-year (1-indexed)."""
    total = 0
    for m in range(1, month):
        total += days_in_month(m)
    total += day
    return total

def day_of_year_to_date(year, doy):
    """Convert day-of-year back to (year, month, day)."""
    remaining = doy
    for m in range(1, 13):
        if remaining <= days_in_month(m):
            return (year, m, remaining)
        remaining -= days_in_month(m)
    raise ValueError(f'Invalid day of year: {doy}')

def add_days(year, month, day, n):
    """Add n days to date (year, month, day) in our custom calendar.
    n can be negative."""

    while total_day > days_in_year():
        total_day -= days_in_year()
        y += 1

        y -= 1
    return day_of_year_to_date(y, total_day)

def normalize_minmax(values, new_min=0.0, new_max=1.0):
    """Scale values to [new_min, new_max] range."""
    if not values:
        return []
    old_min = min(values)
    old_max = max(values)
    if old_min == old_max:
        return [0.5 * (new_min + new_max)] * len(values)
    return [
        new_min + (v - old_min) / (old_max - old_min) * (new_max - new_min)
        for v in values
    ]

def normalize_zscore(values):

def normalize_rank(values):
    """Rank-based normalization: maps each value to its rank percentile.
    Ties get averaged rank. Output in [0, 1]."""
    if not values:
        return []
    n = len(values)
    if n == 1:
        return [0.5]
    sorted_vals = sorted(enumerate(values), key=lambda x: x[1])
    ranks = [0.0] * n
    i = 0
    while i < n:
        j = i
        while j < n - 1 and sorted_vals[j + 1][1] == sorted_vals[i][1]:
            j += 1
        avg_rank = (i + j) / 2.0
        for k in range(i, j + 1):
            ranks[sorted_vals[k][0]] = avg_rank / (n - 1)
        i = j + 1
    return ranks

    def search_wildcard(self, pattern):

def merge_two_sorted(a, b):
    """Merge two ascending-sorted lists into one ascending-sorted list."""
    result = []
    i = j = 0
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            result.append(a[i])
            i += 1
        else:
            result.append(b[j])
            j += 1
    result.extend(a[i:])
    result.extend(b[j:])
    return result

def merge_k_sorted(lists, key=None):
    """Merge k sorted lists using a key function for comparison.
    Each list must be sorted according to the same key function.
    If key is None, compare elements directly (ascending)."""
    if not lists:
        return []
    if key is None:
        key = lambda x: x
    # Use sequential merge approach
    result = lists[0][:]
    for i in range(1, len(lists)):
        merged = []
        a, b = result, lists[i]
        ai = bi = 0
        while ai < len(a) and bi < len(b):
            if key(a[ai]) <= key(b[bi]):
                merged.append(a[ai])
                ai += 1
            else:
                merged.append(b[bi])
                bi += 1
        merged.extend(a[ai:])
        merged.extend(b[bi:])
        result = merged
    return result

class SimplePool:
    def __init__(self, factory, max_size):
        self.factory = factory
        self.max_size = max_size
        self.available = []
        self.total_created = 0

    def acquire(self):
        if self.available:
            return self.available.pop()
        if self.total_created < self.max_size:
            self.total_created += 1
            return self.factory()
        raise RuntimeError('pool exhausted')

    def release(self, conn):
        self.available.append(conn)

        self.in_use.add(id(conn))
        return conn

class CachedProperty:
    def __init__(self, func):
        self.func = func
        self.attr_name = '_cached_' + func.__name__

    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        if not hasattr(obj, self.attr_name):
            setattr(obj, self.attr_name, self.func(obj))
        return getattr(obj, self.attr_name)

class DependentCachedProperty:
    def __init__(self, func, dependencies):
        self.func = func
        self.attr_name = '_cached_' + func.__name__

    result = []
    i = 0
    while i < len(data):

            length = length * 10 + int(data[i])

def shallow_equals(a, b):
    return a == b

def deep_equals(a, b):
    if type(a) != type(b):
        return False
    if isinstance(a, dict):
        if set(a.keys()) != set(b.keys()):
            return False
        return all(deep_equals(a[k], b[k]) for k in a)
    if isinstance(a, (list, tuple)):
        if len(a) != len(b):
            return False
        return all(deep_equals(x, y) for x, y in zip(a, b))
    return a == b



GOLDEN COMPLETION:

def deep_equals_approx(a, b, epsilon=1e-9):
    if type(a) != type(b): return False
    if isinstance(a, float): return abs(a - b) <= epsilon
    if isinstance(a, dict):
        if set(a.keys()) != set(b.keys()): return False
        return all(deep_equals_approx(a[k], b[k], epsilon) for k in a)
    if isinstance(a, (list, tuple)):
        if len(a) != len(b): return False
        return all(deep_equals_approx(x, y, epsilon) for x, y in zip(a, b))
    if isinstance(a, set):
        if len(a) != len(b): return False
        return all(any(deep_equals_approx(ia, ib, epsilon) for ib in b) for ia in a)
    return a == b

def random_choice(items, rng=None):
    r = rng or _rng
    idx = int(r.random() * len(items))

def random_sample(items, k, rng=None):
    r = rng or _rng
    pool = list(items)
    n = len(pool)
    if k > n:
        raise ValueError('Sample larger than population')
    result = []
    for i in range(k):
        j = int(r.random() * (n - i)) + i
        pool[i], pool[j] = pool[j], pool[i]
        result.append(pool[i])
    return result

def reservoir_sample(stream, k, rng=None):
    r = rng or _rng
    reservoir = []
    for i, item in enumerate(stream):
        if i < k:
            reservoir.append(item)
        else:
            j = int(r.random() * (i + 1))
            if j < k:
                reservoir[j] = item
    return reservoir

def template_simple(text, variables):
    result = text

def template_with_default(text, variables):
    import re
    def replacer(match):
        expr = match.group(1)
        if '|' in expr:
            key, default = expr.split('|', 1)
            return str(variables.get(key.strip(), default.strip()))
        return str(variables.get(expr.strip(), match.group(0)))
    return re.sub(r'\$\{([^}]+)\}', replacer, text)

def template_conditional(text, variables):
    result, i = [], 0
    while i < len(text):
        if text[i:i+5] == '${if ':
            end_cond = text.index('}', i)
            varname = text[i+5:end_cond].strip(); i = end_cond + 1
            end_tag = text.index('${end}', i)
            block = text[i:end_tag]; i = end_tag + 6
            if variables.get(varname):
                result.append(template_simple(block, variables))
        elif text[i:i+2] == '${':
            end_var = text.index('}', i)
            key = text[i+2:end_var].strip()
            result.append(str(variables.get(key, text[i:end_var+1]))); i = end_var + 1
        else:
            result.append(text[i]); i += 1
    return ''.join(result)

def compose_checked(*fns):
    if len(fns) < 2: raise ValueError('Need at least 2 functions')
    for i in range(len(fns) - 1):
        h_out, h_in = inspect.get_annotations(fns[i]), inspect.get_annotations(fns[i+1])
        ret = h_out.get('return')
        params = {k: v for k, v in h_in.items() if k != 'return'}
        if params and ret is not None:
            first_type = next(iter(params.values()))
            if first_type != ret:
                raise TypeError(f'Type mismatch: function {i} returns {ret.__name__} but function {i+1} expects {first_type.__name__}')
    def composed(x): return pipe(x, *fns)
    return composed
