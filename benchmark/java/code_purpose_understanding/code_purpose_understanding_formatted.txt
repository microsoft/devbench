        final String id;
        final String sku;
        int qty;
        final int expiryDay;
        final boolean quarantined;
        Lot(String id, String sku, int qty, int expiryDay, boolean quarantined) {
            this.id = id; this.sku = sku; this.qty = qty; this.expiryDay = expiryDay; this.quarantined = quarantined;
        }

    static final int minShelfLifeDays = 3;
    static Map<String, List<Lot>> lotsBySku = new HashMap<>();
    static Map<String, List<String>> ticketsByOrder = new HashMap<>();
    static List<String> auditLog = new ArrayList<>();

    static void reset() { lotsBySku.clear(); ticketsByOrder.clear(); auditLog.clear(); }
    static void addLot(String id, String sku, int qty, int expiryDay, boolean quarantined) {
        lotsBySku.computeIfAbsent(sku, k -> new ArrayList<>()).add(new Lot(id, sku, qty, expiryDay, quarantined));
    }
    static int totalQty(String sku) {
        int total = 0;
        for (Lot l : lotsBySku.getOrDefault(sku, List.of())) total += l.qty;
        return total;
    }

    // Allocate a pick ticket for one SKU.
    // Rules:
    //  1. orderId is idempotent: return a copy of the saved ticket before checking inventory.
    //  2. Eligible lots match sku, are not quarantined, have qty > 0, and expiryDay >= shipDay + minShelfLifeDays.
    //  3. Pick eligible lots FEFO: lower expiryDay first, then lexicographic lot id.
    //  4. If eligible stock is short and allowBackorder is false, return null with no stock or audit mutation.
    //  5. If allowBackorder is true, pick what is available and append BACKORDER:<sku>:<missing>.
    //  6. Deduct picked quantities, remove depleted lots, save the ticket, and audit every pick/backorder.
    static List<String> allocateOrder(String orderId, String sku, int qty, int shipDay, boolean allowBackorder) {


GOLDEN COMPLETION:
----------------------------------------
        if (ticketsByOrder.containsKey(orderId)) return new ArrayList<>(ticketsByOrder.get(orderId));
        if (qty <= 0) throw new IllegalArgumentException("qty");
        List<Lot> eligible = new ArrayList<>();
        for (Lot l : lotsBySku.getOrDefault(sku, List.of())) if (!l.quarantined && l.qty > 0 && l.expiryDay >= shipDay + minShelfLifeDays) eligible.add(l);
        eligible.sort(Comparator.comparingInt((Lot l) -> l.expiryDay).thenComparing(l -> l.id));
        int available = 0;
        for (Lot l : eligible) available += l.qty;
        if (available < qty && !allowBackorder) return null;
        List<String> ticket = new ArrayList<>();
        int needed = qty;
        for (Lot l : eligible) {
            if (needed == 0) break;
            int take = Math.min(needed, l.qty);
            l.qty -= take; needed -= take;
            String line = "PICK:" + l.id + ":" + take;
            ticket.add(line); auditLog.add(orderId + ":" + line);
        }
        lotsBySku.getOrDefault(sku, new ArrayList<>()).removeIf(l -> l.qty == 0);
        if (needed > 0) { String line = "BACKORDER:" + sku + ":" + needed; ticket.add(line); auditLog.add(orderId + ":" + line); }
        ticketsByOrder.put(orderId, new ArrayList<>(ticket));
        return new ArrayList<>(ticket);

    }

public class JavaCPMutation3 {
    static int rxTotalQty;
    static int filledQty;
    static int daysSupply;
    static int lastFillDay;
    static boolean controlled;
    static boolean expeditedUsed;
    static int inventoryQty;
    static Set<String> processedFillIds = new HashSet<>();
    static List<String> fillLog = new ArrayList<>();

    static void reset(int totalQty, int supplyDays, boolean isControlled, int inventory) {
        rxTotalQty = totalQty; filledQty = 0; daysSupply = supplyDays; lastFillDay = -1;
        controlled = isControlled; expeditedUsed = false; inventoryQty = inventory;
        processedFillIds.clear(); fillLog.clear();
    }

    // Fill or refill a prescription.
    // Rules:
    //  1. fillId is idempotent: duplicates return DUPLICATE before validation and do not log.
    //  2. If the prescription is complete, return RX_COMPLETE with no mutation.
    //  3. Refill is too soon unless days since last fill is at least daysSupply * 75 / 100 using integer truncation.
    //  4. One expedited too-soon exception is allowed for non-controlled prescriptions only.
    //  5. Controlled prescriptions cannot be expedited and cannot dispense more than 30 units per fill.
    //  6. Dispense no more than requested, remaining prescription quantity, and inventory; partial fills log a backorder.
    //  7. Successful full fills return FILLED, partial fills return PARTIAL; denials leave all counters unchanged.
    static String fill(String fillId, int day, int requestedQty, boolean expedited) {


GOLDEN COMPLETION:
----------------------------------------
        if (processedFillIds.contains(fillId)) return "DUPLICATE";
        if (filledQty >= rxTotalQty) return "RX_COMPLETE";
        if (requestedQty <= 0) throw new IllegalArgumentException("qty");
        if (controlled && expedited) return "CONTROLLED_EXPEDITE_DENIED";
        if (controlled && requestedQty > 30) return "CONTROLLED_QTY_LIMIT";
        boolean firstFill = lastFillDay < 0;
        int threshold = (daysSupply * 75) / 100;
        if (!firstFill && day - lastFillDay < threshold) {
            if (!expedited || controlled || expeditedUsed) return "TOO_SOON";
            expeditedUsed = true;
        }
        int remainingRx = rxTotalQty - filledQty;
        int target = Math.min(requestedQty, remainingRx);
        int dispensed = Math.min(target, inventoryQty);
        if (dispensed == 0) return "OUT_OF_STOCK";
        processedFillIds.add(fillId);
        filledQty += dispensed;
        inventoryQty -= dispensed;
        lastFillDay = day;
        fillLog.add("FILL:" + fillId + ":qty=" + dispensed + ":day=" + day);
        if (dispensed < target) { fillLog.add("BACKORDER:" + fillId + ":qty=" + (target - dispensed)); return "PARTIAL"; }
        return "FILLED";

----------------------------------------
    }

    static class Paycheck {
        final BigDecimal retirement;
        final BigDecimal tax;
        final BigDecimal health;
        final BigDecimal garnishment;
        final BigDecimal net;
        Paycheck(BigDecimal retirement, BigDecimal tax, BigDecimal health, BigDecimal garnishment, BigDecimal net) {
            this.retirement = retirement; this.tax = tax; this.health = health; this.garnishment = garnishment; this.net = net;
        }

    static BigDecimal retirementRate = new BigDecimal("0.05");
    static BigDecimal retirementCap = new BigDecimal("1000.00");
    static BigDecimal healthDeduction = new BigDecimal("200.00");
    static BigDecimal garnishmentRate = new BigDecimal("0.15");
    static BigDecimal minimumNet = new BigDecimal("500.00");
    static BigDecimal ytdGross = BigDecimal.ZERO;
    static BigDecimal ytdRetirement = BigDecimal.ZERO;

    static void reset() {

    static BigDecimal taxOn(BigDecimal taxable) {
        BigDecimal tax = BigDecimal.ZERO;
        BigDecimal first = taxable.min(new BigDecimal("1000.00"));
        if (first.compareTo(BigDecimal.ZERO) > 0) tax = tax.add(first.multiply(new BigDecimal("0.10")));
        BigDecimal second = taxable.subtract(new BigDecimal("1000.00")).min(new BigDecimal("2000.00"));
        if (second.compareTo(BigDecimal.ZERO) > 0) tax = tax.add(second.multiply(new BigDecimal("0.20")));
        BigDecimal rest = taxable.subtract(new BigDecimal("3000.00"));
        if (rest.compareTo(BigDecimal.ZERO) > 0) tax = tax.add(rest.multiply(new BigDecimal("0.30")));
        return tax.setScale(2, RoundingMode.HALF_UP);
    }

    // Run one payroll check.
    // Rules:
    //  1. runId is idempotent: duplicates return null before validation or accumulator updates.
    //  2. Retirement is pre-tax gross * 5%, capped by remaining annual retirementCap.
    //  3. Tax is marginal on gross minus retirement.
    //  4. Health deduction is taken after tax, but can be reduced only if needed to preserve minimumNet.
    //  5. Garnishment is 15% of disposable pay after tax/retirement/health, capped so net does not fall below minimumNet.
    //  6. If net is still below minimumNet after zeroing garnishment, reduce health, never tax or retirement.
    //  7. Update all YTD accumulators and write one audit line only for successful runs.
    static Paycheck runPayroll(String runId, BigDecimal gross) {


GOLDEN COMPLETION:
----------------------------------------
        if (processedRuns.contains(runId)) return null;
        if (gross.compareTo(BigDecimal.ZERO) <= 0) throw new IllegalArgumentException("gross");
        BigDecimal capLeft = retirementCap.subtract(ytdRetirement); if (capLeft.compareTo(BigDecimal.ZERO) < 0) capLeft = BigDecimal.ZERO;
        BigDecimal retirement = gross.multiply(retirementRate).setScale(2, RoundingMode.HALF_UP).min(capLeft).setScale(2, RoundingMode.HALF_UP);
        BigDecimal tax = taxOn(gross.subtract(retirement));
        BigDecimal health = healthDeduction.min(gross.subtract(retirement).subtract(tax));
        if (health.compareTo(BigDecimal.ZERO) < 0) health = BigDecimal.ZERO.setScale(2);
        BigDecimal disposable = gross.subtract(retirement).subtract(tax).subtract(health).setScale(2, RoundingMode.HALF_UP);
        BigDecimal garnishment = disposable.multiply(garnishmentRate).setScale(2, RoundingMode.HALF_UP);
        if (disposable.subtract(garnishment).compareTo(minimumNet) < 0) garnishment = disposable.subtract(minimumNet).max(BigDecimal.ZERO).setScale(2, RoundingMode.HALF_UP);
        BigDecimal net = disposable.subtract(garnishment).setScale(2, RoundingMode.HALF_UP);
        if (net.compareTo(minimumNet) < 0) {
            BigDecimal restore = minimumNet.subtract(net).min(health);
            health = health.subtract(restore).setScale(2, RoundingMode.HALF_UP);
            net = net.add(restore).setScale(2, RoundingMode.HALF_UP);
        }
        processedRuns.add(runId);

        return new Paycheck(retirement, tax, health, garnishment, net);

public class CPTask8 {
    // Tax brackets: [upperBound, rate%]  (marginal)
    // 0-1000: 10%, 1001-3000: 20%, 3001+: 30%
    static int[][] taxBrackets = {{1000, 10}, {3000, 20}, {Integer.MAX_VALUE, 30}};
    static BigDecimal healthDeduction = new BigDecimal("200.00");
    static BigDecimal retirementRate = new BigDecimal("0.05"); // 5% of gross
    static BigDecimal garnishmentRate = new BigDecimal("0.10"); // 10% of post-tax
    static BigDecimal minimumNetPay = new BigDecimal("500.00");
    static BigDecimal ytdGross = BigDecimal.ZERO;
    static BigDecimal ytdTax = BigDecimal.ZERO;
    static BigDecimal ytdNet = BigDecimal.ZERO;
    static List<String> auditLog = new ArrayList<>();

    static BigDecimal computeMarginalTax(BigDecimal gross) {
        BigDecimal tax = BigDecimal.ZERO;
        BigDecimal rem = gross;
        int prev = 0;
        for (int[] bracket : taxBrackets) {
            BigDecimal width = new BigDecimal(Math.min(rem.intValue(), bracket[0] - prev));
            if (width.compareTo(BigDecimal.ZERO) <= 0) break;
            tax = tax.add(width.multiply(new BigDecimal(bracket[1])).divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP));
            rem = rem.subtract(width);
            prev = bracket[0];
        }
        return tax;
    }

    // calculatePay(grossPay):
    // 1. Compute marginal tax using computeMarginalTax(grossPay)
    // 2. postTax = grossPay - tax
    // 3. Deductions (from postTax): health (flat) + retirement (5% of gross)
    //    totalDeductions = health + retirement
    // 4. Garnishment: 10% of postTax (NOT post-deduction)
    // 5. netPay = postTax - totalDeductions - garnishment
    // 6. If netPay < minimumNetPay, clamp: reduce garnishment first, then deductions
    //    Garnishment reduces to max(0, garnishment - shortfall)
    //    If still short, reduce totalDeductions by remaining shortfall
    //    Recompute netPay after adjustments
    // 7. Update YTD accumulators (ytdGross, ytdTax, ytdNet)
    // 8. Append audit: "PAY:gross=G,tax=T,ded=D,garn=GA,net=N"
    // 9. Return netPay


GOLDEN COMPLETION:
----------------------------------------
    static BigDecimal calculatePay(BigDecimal grossPay) {
        BigDecimal tax = computeMarginalTax(grossPay);
        BigDecimal postTax = grossPay.subtract(tax);
        BigDecimal retirement = grossPay.multiply(retirementRate).setScale(2, RoundingMode.HALF_UP);
        BigDecimal ded = healthDeduction.add(retirement);
        BigDecimal garn = postTax.multiply(garnishmentRate).setScale(2, RoundingMode.HALF_UP);
        BigDecimal netPay = postTax.subtract(ded).subtract(garn);
        if (netPay.compareTo(minimumNetPay) < 0) {
            BigDecimal shortfall = minimumNetPay.subtract(netPay);
            BigDecimal garnReduction = shortfall.min(garn);
            garn = garn.subtract(garnReduction);
            shortfall = shortfall.subtract(garnReduction);
            ded = ded.subtract(shortfall.min(ded));
            netPay = postTax.subtract(ded).subtract(garn);
        }
        ytdGross = ytdGross.add(grossPay);
        ytdTax = ytdTax.add(tax);
        ytdNet = ytdNet.add(netPay);
        auditLog.add("PAY:gross=" + grossPay + ",tax=" + tax + ",ded=" + ded + ",garn=" + garn + ",net=" + netPay);
        return netPay;
    }

public class CPTask14 {
    // Warehouse pick optimization with FEFO (First-Expiry-First-Out)
    // Rules:
    //  1. Stock organized as list of lots: each lot has {lotId, qty, expiryDay}
    //  2. Pick from earliest-expiring lots FIRST (sort by expiryDay ascending)
    //  3. If a lot doesn't fully cover the request, take what's available (partial fill)
    //     and continue to the next lot
    //  4. If total stock < requested qty, fill what's available and create backorder
    //     for the remainder
    //  5. Return a PickResult with: filledQty, backorderQty, list of picks as
    //     "LOT:<lotId>:<qty>", and update stock lots (reduce qty, remove if 0)

    static List<int[]> stockLots = new ArrayList<>();  // {lotId, qty, expiryDay}
    static List<String> pickLog = new ArrayList<>();

    static void addLot(int lotId, int qty, int expiryDay) {
        stockLots.add(new int[]{lotId, qty, expiryDay});
    }

    static int[] pickOrder(int requestedQty) {
        // Returns int[2] = {filledQty, backorderQty}


GOLDEN COMPLETION:
----------------------------------------
        stockLots.sort((a, b) -> Integer.compare(a[2], b[2]));
        int remaining = requestedQty;
        int filled = 0;
        Iterator<int[]> it = stockLots.iterator();
        while (it.hasNext() && remaining > 0) {
            int[] lot = it.next();
            int take = Math.min(lot[1], remaining);
            pickLog.add("LOT:" + lot[0] + ":" + take);
            lot[1] -= take;
            filled += take;
            remaining -= take;
            if (lot[1] == 0) {
                it.remove();
            }
        }
        return new int[]{filled, remaining};
    }

public class CPTask16 {
    // Performance review calibration system
    // Rules:
    //  1. Ratings: 1(low) to 5(high). Forced distribution across a team:
    //     At most 20% can be rated 5, at most 20% can be rated 1
    //     (i.e., count of rating 5 <= floor(teamSize * 0.2), same for rating 1)
    //  2. Manager can override a rating by at most +/-1 from the original
    //  3. Employees cannot rate themselves (rater != ratee)
    //  4. All calibrated ratings stored in finalRatings map
    //  5. Every action logged to reviewLog as "RATE:<ratee>:<rating>" or
    //     "OVERRIDE:<ratee>:<old>-><new>" or "DENIED:<reason>"

    static Map<String, Integer> finalRatings = new HashMap<>();
    static List<String> reviewLog = new ArrayList<>();
    static int teamSize = 0;

    static void initTeam(int size) {
        teamSize = size;
        finalRatings.clear();
        reviewLog.clear();
    }

    static boolean overrideRating(String ratee, int newRating) {
        if (!finalRatings.containsKey(ratee)) { reviewLog.add("DENIED:NO_EXISTING_RATING"); return false; }
        int old = finalRatings.get(ratee);
        if (Math.abs(newRating - old) > 1) { reviewLog.add("DENIED:OVERRIDE_TOO_LARGE"); return false; }
        finalRatings.put(ratee, newRating);
        reviewLog.add("OVERRIDE:" + ratee + ":" + old + "->" + newRating);
        return true;
    }

    // Submit a rating: checks self-rating, rating range, distribution limits
    static boolean submitRating(String rater, String ratee, int rating) {


GOLDEN COMPLETION:
----------------------------------------
        if (rater.equals(ratee)) { reviewLog.add("DENIED:SELF_RATING"); return false; }
        if (rating < 1 || rating > 5) { reviewLog.add("DENIED:INVALID_RATING"); return false; }
        int maxForRating = (int) Math.floor(teamSize * 0.2);
        if (rating == 5 || rating == 1) {
            long count = finalRatings.values().stream().filter(r -> r == rating).count();
            if (count >= maxForRating) { reviewLog.add("DENIED:DISTRIBUTION_LIMIT"); return false; }
        }
        finalRatings.put(ratee, rating);
        reviewLog.add("RATE:" + ratee + ":" + rating);
        return true;
    }

public class CPTask17 {
    // Cart pricing engine with stacking discount rules
    // Rules:
    //  1. Items have price and quantity
    //  2. Discounts applied in order: PERCENT discounts first, then FLAT discounts
    //  3. After ALL discounts, each item's effective price cannot go below $1.00
    //     (per-item floor, applied per-unit before totaling)
    //  4. Shipping: $5.99 if subtotal < $50, free if subtotal >= $50
    //  5. Return CartTotal with: subtotal, shipping, grandTotal (all BigDecimal scale 2)

    static List<BigDecimal[]> items = new ArrayList<>();  // {unitPrice, quantity}
    static List<Object[]> discounts = new ArrayList<>();  // {"PERCENT"|"FLAT", BigDecimal amount}

    static void addItem(BigDecimal price, int qty) {
        items.add(new BigDecimal[]{price, new BigDecimal(qty)});
    }

    static void addDiscount(String type, BigDecimal amount) {
        discounts.add(new Object[]{type, amount});
    }

    // Returns BigDecimal[3] = {subtotal, shipping, grandTotal}
    static BigDecimal[] calculateTotal() {


GOLDEN COMPLETION:
----------------------------------------
        BigDecimal[] ep = new BigDecimal[items.size()];
        for (int i = 0; i < items.size(); i++) ep[i] = items.get(i)[0];
        for (Object[] d : discounts) if ("PERCENT".equals(d[0])) {
            BigDecimal m = BigDecimal.ONE.subtract(((BigDecimal)d[1]).divide(new BigDecimal("100"),10,RoundingMode.HALF_UP));
            for (int i = 0; i < ep.length; i++) ep[i] = ep[i].multiply(m).setScale(2, RoundingMode.HALF_UP);
        }
        for (Object[] d : discounts) if ("FLAT".equals(d[0])) {
            BigDecimal per = ((BigDecimal)d[1]).divide(new BigDecimal(items.size()), 2, RoundingMode.HALF_UP);
            for (int i = 0; i < ep.length; i++) ep[i] = ep[i].subtract(per);
        }
        BigDecimal sub = BigDecimal.ZERO, floor = new BigDecimal("1.00");
        for (int i = 0; i < items.size(); i++) {
            BigDecimal up = ep[i].compareTo(floor) < 0 ? floor : ep[i];
            sub = sub.add(up.multiply(items.get(i)[1]));
        }
        sub = sub.setScale(2, RoundingMode.HALF_UP);
        BigDecimal ship = sub.compareTo(new BigDecimal("50.00")) >= 0 ? BigDecimal.ZERO.setScale(2) : new BigDecimal("5.99");
        return new BigDecimal[]{sub, ship, sub.add(ship).setScale(2, RoundingMode.HALF_UP)};
    }

    // States: CLOSED (normal), OPEN (blocking), HALF_OPEN (testing)
    // Rules:
    //  1. CLOSED: allow requests. Track consecutive failures.

    //     If a probe fails, transition back to OPEN (reset the timeout timer).
    //  4. Log every attempt as "CB:<state>:<ALLOW|DENY>"

        state = "CLOSED";
        consecutiveFailures = 0;
        failureThreshold = threshold;
        resetTimeoutMs = timeoutMs;
        probeLimit = probeLim;
        probeCount = 0;
        openedAt = 0;
        cbLog.clear();

GOLDEN COMPLETION:

        }

            probeCount++;
            cbLog.add("CB:HALF_OPEN:ALLOW");
            return true;
        }
        cbLog.add("CB:CLOSED:ALLOW");
        return true;
    }

    // Maintenance margin = 0.25 (25%)
    // If margin ratio < 0.25, must liquidate positions to restore it.
    //
    // Liquidation rules:
    //   1. Sort positions by market value DESCENDING (liquidate largest first)
    //   2. For each position, sell the MINIMUM whole units needed to bring margin ratio >= 0.25
    //      - Each unit sold: equity += currentPrice, quantity -= 1

    //   4. Stop as soon as margin ratio >= 0.25 (do NOT over-liquidate)

GOLDEN COMPLETION:
----------------------------------------

        BigDecimal tmv = totalMarketValue();
        if (tmv.compareTo(BigDecimal.ZERO) == 0) return false;
        if (equity.divide(tmv, 4, RoundingMode.HALF_UP).compareTo(maintenanceMargin) >= 0) return false;

        boolean liquidated = false;

            int unitsSold = 0;
            while (p.quantity.compareTo(BigDecimal.ZERO) > 0) {
                BigDecimal ratio = equity.divide(totalMarketValue(), 4, RoundingMode.HALF_UP);
                if (ratio.compareTo(maintenanceMargin) >= 0) break;

                equity = equity.add(p.currentPrice);
                unitsSold++;

            if (unitsSold > 0) {
                liquidationLog.add("LIQUIDATE:" + p.symbol + ":" + unitsSold);
                liquidated = true;
            }
            BigDecimal ratio = equity.divide(totalMarketValue(), 4, RoundingMode.HALF_UP);
            if (ratio.compareTo(maintenanceMargin) >= 0) break;
        }
        return liquidated;

            enrollLog.add("PREREQ_FAIL:" + student + ":" + course);
            return "PREREQ_FAIL";

        LinkedList<String> wl = waitlist.computeIfAbsent(course, k -> new LinkedList<>());

            enrollLog.add("ENROLLED:" + student + ":" + course);
            return "ENROLLED";
        }
        wl.add(student);

        return "WAITLISTED";

    static void reset() { scheduleLog.clear(); }

    // Schedule jobs on a single machine within the availability window.
    // Rules:
    //   1. Sort jobs by priority (ascending — lower number = higher priority)
    //   2. Start scheduling at machineAvailableFrom
    //   3. When switching product types (current != previous), add setupTime
    //      (first job has no setup penalty)
    //   4. If a job fits entirely within the window, schedule it:
    //      Log "SCHEDULED:<jobId>:<startTime>-<endTime>"
    //   5. If a job does NOT fit, skip it:
    //      Log "SKIPPED:<jobId>:INSUFFICIENT_TIME"
    //   6. Return total number of jobs scheduled
    static int scheduleJobs(List<Job> jobs) {


GOLDEN COMPLETION:
----------------------------------------
        List<Job> sorted = new ArrayList<>(jobs);
        sorted.sort(Comparator.comparingInt(j -> j.priority));
        int currentTime = machineAvailableFrom;
        String lastProductType = null;
        int scheduled = 0;
        for (Job job : sorted) {
            int start = currentTime;
            if (lastProductType != null && !lastProductType.equals(job.productType)) {
                start += setupTime;
            }
            int end = start + job.duration;
            if (end <= machineAvailableUntil) {
                scheduleLog.add("SCHEDULED:" + job.id + ":" + start + "-" + end);
                currentTime = end;
                lastProductType = job.productType;
                scheduled++;
            } else {
                scheduleLog.add("SKIPPED:" + job.id + ":INSUFFICIENT_TIME");
            }
        }
        return scheduled;

----------------------------------------
    }

    //   1. Process contingencies in DEADLINE ORDER (earliest first)

GOLDEN COMPLETION:

                return status;
            }
            escrowLog.add("CLEARED:" + c.name + ":DAY" + c.deadlineDay);
        }
        return status;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.*;

public class CPTask28 {
    static List<String> commissionLog = new ArrayList<>();

    // Calculate agent commission from a property sale with tiered rates.
    //
    // Commission tiers (on sale price):
    //   - First tierOneLimit: tierOneRate
    //   - Next (tierTwoLimit - tierOneLimit): tierTwoRate
    //   - Remainder above tierTwoLimit: tierThreeRate
    //
    // Then split between listing and buying agent:
    //   - listingAgent gets splitRatio of total commission
    //   - buyingAgent gets (1 - splitRatio) of total commission
    //
    // Constraints:
    //   - Each agent's commission has a MINIMUM guarantee (minCommission)
    //   - Each agent's commission is CAPPED at maxCommission
    //   - Apply minimum FIRST, then cap (minimum takes priority over cap if min > max, use min)
    //
    // Log: "COMMISSION:<salePrice>|TOTAL:<totalComm>|LISTING:<listingComm>|BUYING:<buyingComm>"
    // Returns: BigDecimal[]{listingCommission, buyingCommission}
    static BigDecimal[] calculateCommission(BigDecimal salePrice,
                                             BigDecimal tierOneLimit, BigDecimal tierOneRate,
                                             BigDecimal tierTwoLimit, BigDecimal tierTwoRate,
                                             BigDecimal tierThreeRate,
                                             BigDecimal splitRatio,
                                             BigDecimal minCommission, BigDecimal maxCommission) {


GOLDEN COMPLETION:
----------------------------------------
        BigDecimal total = BigDecimal.ZERO;
        if (salePrice.compareTo(tierOneLimit) <= 0) {
            total = salePrice.multiply(tierOneRate);
        } else if (salePrice.compareTo(tierTwoLimit) <= 0) {
            total = tierOneLimit.multiply(tierOneRate)
                    .add(salePrice.subtract(tierOneLimit).multiply(tierTwoRate));
        } else {
            total = tierOneLimit.multiply(tierOneRate)
                    .add(tierTwoLimit.subtract(tierOneLimit).multiply(tierTwoRate))
                    .add(salePrice.subtract(tierTwoLimit).multiply(tierThreeRate));
        }
        total = total.setScale(2, RoundingMode.HALF_UP);
        BigDecimal listing = total.multiply(splitRatio).setScale(2, RoundingMode.HALF_UP);
        BigDecimal buying = total.subtract(listing);
        // Apply minimum first, then cap
        if (listing.compareTo(minCommission) < 0) listing = minCommission;
        if (listing.compareTo(maxCommission) > 0) listing = maxCommission;
        if (buying.compareTo(minCommission) < 0) buying = minCommission;
        if (buying.compareTo(maxCommission) > 0) buying = maxCommission;
        commissionLog.add("COMMISSION:" + salePrice + "|TOTAL:" + total + "|LISTING:" + listing + "|BUYING:" + buying);
        return new BigDecimal[]{listing, buying};


SUFFIX CODE:
----------------------------------------
    }

        if (points >= 250) return "GOLD";
        if (points >= 100) return "SILVER";
        return "BRONZE";

    // Attempt to unlock an achievement.
    // Rules:
    //   1. If already unlocked, log "DUPLICATE:<achieveId>" and return false (idempotent)
    //   2. Check all prerequisites are unlocked. If not,
    //      log "PREREQ_MISSING:<achieveId>:<missing csv sorted>" and return false
    //   3. Unlock the achievement: add to unlocked set
    //   4. Add points: totalPoints += achievementPoints[achieveId]
    //   5. Check tier promotion: if new tier != old tier,
    //      log "TIER_UP:<oldTier>-><newTier>:<totalPoints>pts"

    //   7. Return true
    static boolean tryUnlock(String achieveId) {


GOLDEN COMPLETION:
----------------------------------------
        if (unlocked.contains(achieveId)) {
            achieveLog.add("DUPLICATE:" + achieveId);
            return false;
        }
        Set<String> prereqs = prerequisites.getOrDefault(achieveId, Collections.emptySet());
        Set<String> missing = new TreeSet<>();

        }

            return false;

        String oldTier = tier;
        totalPoints += pts;
        String newTier = computeTier(totalPoints);
        if (!newTier.equals(oldTier)) {
            tier = newTier;
            achieveLog.add("TIER_UP:" + oldTier + "->" + newTier + ":" + totalPoints + "pts");
        }
        achieveLog.add("UNLOCKED:" + achieveId + ":" + pts + "pts:TOTAL:" + totalPoints);
        return true;

    static class Player {
        final String name;
        final int seed; // lower = better
        int score;

        Player(String name, int seed) { this.name = name; this.seed = seed; this.score = 0; }
    }

    static List<String> bracketLog = new ArrayList<>();

    static void resetLog() { bracketLog.clear(); }

    // Generate first-round pairings from seeded players for a single-elimination bracket.
    // Rules:
    //   1. Sort players by seed (ascending — seed 1 is best)
    //   2. Calculate the bracket size: smallest power of 2 >= numPlayers
    //   3. Number of byes = bracketSize - numPlayers
    //   4. Top 'byes' seeds get a bye (auto-advance without playing)
    //      Log "BYE:<playerName>:SEED<seed>"
    //   5. Remaining players pair: first remaining vs last remaining, second vs second-last, etc.
    //      Log "MATCH:<p1Name>(SEED<p1Seed>)vs<p2Name>(SEED<p2Seed>)"
    //   6. Return list of advancing player names:
    //      - Bye players are added first (in seed order)
    //      - Then for each match, the player with LOWER seed advances (better seed wins)
    //        If same seed (shouldn't happen), alphabetically first name wins
    static List<String> generateFirstRound(List<Player> players) {


GOLDEN COMPLETION:
----------------------------------------
        List<Player> sorted = new ArrayList<>(players);
        sorted.sort(Comparator.comparingInt(p -> p.seed));
        int n = sorted.size();
        int bracketSize = 1;
        while (bracketSize < n) bracketSize *= 2;
        int byes = bracketSize - n;
        List<String> advancing = new ArrayList<>();
        for (int i = 0; i < byes; i++) {
            Player p = sorted.get(i);
            bracketLog.add("BYE:" + p.name + ":SEED" + p.seed);
            advancing.add(p.name);
        }
        List<Player> remaining = sorted.subList(byes, n);
        for (int i = 0; i < remaining.size() / 2; i++) {
            Player p1 = remaining.get(i);
            Player p2 = remaining.get(remaining.size() - 1 - i);
            bracketLog.add("MATCH:" + p1.name + "(SEED" + p1.seed + ")vs" + p2.name + "(SEED" + p2.seed + ")");
            if (p1.seed < p2.seed) advancing.add(p1.name);
            else if (p1.seed > p2.seed) advancing.add(p2.name);
            else advancing.add(p1.name.compareTo(p2.name) < 0 ? p1.name : p2.name);
        }
        return advancing;

    //    e. Log "EARLY:<net>:PENALTY:<penalty>", set redeemed=true, return net

GOLDEN COMPLETION:

            return total;

            .divide(new BigDecimal(termMonths), 2, RoundingMode.HALF_UP);

            .divide(new BigDecimal(12), 2, RoundingMode.HALF_UP);

        auditLog.add("EARLY:" + net + ":PENALTY:" + penalty);
        redeemed = true;
        return net;

    //    b. If estimatedUsage > 0: adjustment = actualUsage - estimatedUsage
    //       bill the adjustment (negative = credit), add to totalBilled
    //       log "ADJUST:<adjustment>:BILL:<adjustBill>", reset estimatedUsage=0

    //    e. Log "ACTUAL:<reading>:USAGE:<actualUsage>"
    // 3. Return totalBilled

----------------------------------------

            return totalBilled;

            estimatedUsage = BigDecimal.ZERO;
        } else { totalBilled = totalBilled.add(calculateBill(actualUsage)); }

        creditBalance = BigDecimal.ZERO;

        return payment;

    // 3. Each milestone releases its percentage, MINUS 10% holdback retained until final milestone

    // 5. If a milestone is marked "FAILED", the grant enters CLAWBACK state:
    //    - No more disbursements allowed
    //    - Previously disbursed amounts remain (no actual clawback of money)
    //    - Holdback is forfeited (not released)

    static BigDecimal totalGrant = new BigDecimal("100000.00");

    static BigDecimal holdback = BigDecimal.ZERO;

    static String grantState = "ACTIVE";
    static List<String> disbursementLog = new ArrayList<>();

    static String completeMilestone(String milestone, boolean passed) {


GOLDEN COMPLETION:
----------------------------------------
        if (grantState.equals("CLAWBACK")) return "GRANT_CLAWBACK";
        if (completedMilestones.contains(milestone)) return "ALREADY_COMPLETED";
        if (nextMilestoneIdx >= milestones.size() || !milestones.get(nextMilestoneIdx).equals(milestone))
            return "WRONG_SEQUENCE";

            grantState = "CLAWBACK";

            .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);

        if (isLast) { holdback = BigDecimal.ZERO; } else { holdback = holdback.add(amt.multiply(new BigDecimal("0.10")).setScale(2, RoundingMode.HALF_UP)); }
        totalDisbursed = totalDisbursed.add(disb);
        disbursementLog.add("DISBURSED:" + milestone + ":$" + disb.setScale(2, RoundingMode.HALF_UP));
        nextMilestoneIdx++;
        return "DISBURSED";

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.*;

public class CPTask49 {
    // Draft pick trading rules:
    // 1. Pick values: Round 1 = $5M, Round 2 = $3M, Round 3 = $1.5M, Round 4+ = $0.5M
    // 2. Each team has a salary cap space. Acquiring picks consumes cap space by pick value.
    // 3. Trade balance rule: total value given must be >= 80% of total value received
    //    (no "robbery" trades — prevents giving R4 picks for R1 picks)
    // 4. If acquiring team's cap space < total value of picks received, trade is "CAP_EXCEEDED"
    // 5. Successful trade: move picks between teams, deduct/add cap space, log trade
    // 6. Trade log: "TRADE:<teamA>-><teamB>:gave=[picks]:got=[picks]:netValue=$<diff>"

    static Map<String, List<Integer>> teamPicks = new HashMap<>();
    static Map<String, BigDecimal> capSpace = new HashMap<>();
    static List<String> tradeLog = new ArrayList<>();

    static BigDecimal pickValue(int round) {
        if (round == 1) return new BigDecimal("5000000");
        if (round == 2) return new BigDecimal("3000000");
        if (round == 3) return new BigDecimal("1500000");
        return new BigDecimal("500000");
    }

    static BigDecimal totalValue(List<Integer> rounds) {
        BigDecimal v = BigDecimal.ZERO;
        for (int r : rounds) v = v.add(pickValue(r));
        return v;
    }

    // teamA gives pickRoundsGiven, receives pickRoundsReceived from teamB
    // Rules: trade balance (given >= 80% received), cap space check, pick existence
    static String executeTrade(String teamA, List<Integer> pickRoundsGiven,
                                String teamB, List<Integer> pickRoundsReceived) {


GOLDEN COMPLETION:
----------------------------------------
        BigDecimal givenVal = totalValue(pickRoundsGiven);
        BigDecimal recvVal = totalValue(pickRoundsReceived);
        if (givenVal.compareTo(recvVal.multiply(new BigDecimal("0.80"))) < 0) return "TRADE_IMBALANCED";
        if (capSpace.get(teamA).compareTo(recvVal) < 0) return "CAP_EXCEEDED";
        List<Integer> aPicks = new ArrayList<>(teamPicks.get(teamA));
        List<Integer> bPicks = new ArrayList<>(teamPicks.get(teamB));
        for (int r : pickRoundsGiven) { if (!aPicks.remove(Integer.valueOf(r))) return "PICK_NOT_FOUND"; }
        for (int r : pickRoundsReceived) { if (!bPicks.remove(Integer.valueOf(r))) return "PICK_NOT_FOUND"; }
        for (int r : pickRoundsReceived) aPicks.add(r);
        for (int r : pickRoundsGiven) bPicks.add(r);
        teamPicks.put(teamA, aPicks);
        teamPicks.put(teamB, bPicks);
        capSpace.put(teamA, capSpace.get(teamA).subtract(recvVal).add(givenVal));
        capSpace.put(teamB, capSpace.get(teamB).subtract(givenVal).add(recvVal));
        tradeLog.add("TRADE:" + teamA + "->" + teamB + ":gave=" + pickRoundsGiven + ":got=" + pickRoundsReceived + ":netValue=$" + recvVal.subtract(givenVal));
        return "TRADE_COMPLETED";


SUFFIX CODE:
----------------------------------------
    }

    // 1. Base salary: paid each season via recordSeason()
    // 2. Performance bonuses (tiered):
    //    - 10+ goals: $100K bonus
    //    - 20+ goals: additional $200K (total $300K with tier 1)
    //    - 30+ goals: additional $500K (total $800K with tiers 1+2)
    // 3. Total bonus cap per season: $600K (so 30+ goals gets $600K, not $800K)
    // 4. Injury clause: if gamesPlayed < 10, no bonus regardless of goals (player was injured)
    //    BUT base salary is still paid at 50% (injury deduction)
    // 5. Vesting: first 2 seasons are "probation" — bonuses are held in escrow,
    //    released only when season 3 starts (vesting event). If player leaves before
    //    season 3, escrowed bonuses are forfeited.

    static BigDecimal baseSalary = new BigDecimal("2000000.00"); // $2M
    static int currentSeason = 0;
    static BigDecimal totalPaid = BigDecimal.ZERO;
    static BigDecimal escrow = BigDecimal.ZERO;
    static boolean vested = false;
    static List<String> payLog = new ArrayList<>();
    static final BigDecimal BONUS_CAP = new BigDecimal("600000");

    // Calculate raw tiered bonus (before cap): 10+=100K, 20+=+200K, 30+=+500K
    static BigDecimal rawBonus(int goals) {
        BigDecimal b = BigDecimal.ZERO;
        if (goals >= 10) b = b.add(new BigDecimal("100000"));
        if (goals >= 20) b = b.add(new BigDecimal("200000"));
        if (goals >= 30) b = b.add(new BigDecimal("500000"));
        return b.compareTo(BONUS_CAP) > 0 ? BONUS_CAP : b;
    }
