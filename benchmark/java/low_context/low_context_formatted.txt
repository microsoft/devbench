import java.util.*;
public class LCTask5 {
    /** Update map: if key absent, set to defaultVal.
     *  If present and current > threshold, REMOVE it (via null return).
     *  Otherwise increment current by 1. Uses Map.compute. */
    public static void update(Map<String, Integer> m, String k, int defaultVal, int threshold) {


GOLDEN COMPLETION:
----------------------------------------
        m.compute(k, (key, cur) -> {
            if (cur == null) return defaultVal;
            return cur > threshold ? null : cur + 1;
        });


SUFFIX CODE:
----------------------------------------
    }
    public static void main(String[] args) {
        var m = new HashMap<String, Integer>();
        update(m, "a", 5, 10);  // absent -> 5
        assert m.get("a") == 5;
        update(m, "a", 5, 10);  // 5 <= 10 -> 6
        assert m.get("a") == 6;
        var m2 = new HashMap<>(Map.of("x", 11));
        update(m2, "x", 0, 10); // 11 > 10 -> removed
        assert !m2.containsKey("x");
    }
}

    }
