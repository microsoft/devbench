        }

import java.util.*;

public class C2NTask20 {

    /**
     * Merge two sorted int arrays into a single sorted int array.
     * STABILITY REQUIREMENT: when elements are equal, the element
     * from 'left' must appear before the element from 'right'.
     * Both input arrays are sorted in ascending order.
     * Must handle empty arrays (either or both may be empty).
     * Returns a new array — inputs are not modified.
     */
    public static int[] stableMerge(int[] left, int[] right) {


GOLDEN COMPLETION:
----------------------------------------
        int[] result = new int[left.length + right.length];
        int i = 0, j = 0, k = 0;
        while (i < left.length && j < right.length) {
            if (left[i] <= right[j]) {
                result[k++] = left[i++];
            } else {
                result[k++] = right[j++];
            }
        }
        while (i < left.length) result[k++] = left[i++];
        while (j < right.length) result[k++] = right[j++];
        return result;


SUFFIX CODE:
----------------------------------------
    }

    public static void main(String[] args) {
        int[] a = {1, 3, 5, 7};
        int[] b = {2, 3, 6, 8};
        int[] merged = stableMerge(a, b);
        assert Arrays.equals(merged, new int[]{1, 2, 3, 3, 5, 6, 7, 8}) : Arrays.toString(merged);
        // empty cases
        assert Arrays.equals(stableMerge(new int[]{}, new int[]{1}), new int[]{1});
        assert Arrays.equals(stableMerge(new int[]{2}, new int[]{}), new int[]{2});
        assert Arrays.equals(stableMerge(new int[]{}, new int[]{}), new int[]{});
        System.out.println("Task 20 PASS");
    }
}

    // Keys are node IDs, values are lists of nodes that the key points to (outgoing edges).

    // Throws IllegalArgumentException if the graph contains a cycle.

    // Returns a list of all nodes in topological order.
    public static List<Integer> topologicalSort(Map<Integer, List<Integer>> graph) {


GOLDEN COMPLETION:
----------------------------------------

        for (int n : allNodes) inDegree.put(n, 0);

            for (int nb : entry.getValue()) inDegree.merge(nb, 1, Integer::sum);
