class Program {
    static readonly string HEX = "0123456789ABCDEF";

    static string HexByte(byte v) {
        return "" + HEX[v >> 4] + HEX[v & 0xF];
    }

    static bool IsUnreserved(char ch) {
        return char.IsLetterOrDigit(ch);
    }

    // URL encoder: ONLY alphanumerics are unescaped by default.
    // Unlike Uri.EscapeDataString, this encodes '~', '_', '-', '.'.
    // The 'safe' parameter can whitelist additional characters.
    static string UrlEncode(string text, string safe = "") {


GOLDEN COMPLETION:
----------------------------------------
        var sb = new StringBuilder();
        foreach (byte b in Encoding.UTF8.GetBytes(text)) {
            char ch = (char)b;
            if (b < 128 && (IsUnreserved(ch) || safe.Contains(ch)))
                sb.Append(ch);
            else
                sb.Append('%').Append(HexByte(b));
        }
        return sb.ToString();


SUFFIX CODE:
----------------------------------------
    }

    static void Main(string[] args) {
        if (HexByte(32) != "20") throw new Exception("hexbyte");
        if (!IsUnreserved('A')) throw new Exception("unreserved A");
        if (IsUnreserved('_')) throw new Exception("_ should be reserved");

        // Stricter than standard: encodes ~, _, -, .
        if (UrlEncode("a b") != "a%20b") throw new Exception("space");
        if (UrlEncode("/~") != "%2F%7E") throw new Exception("slash tilde");
        if (UrlEncode("/~") == "/~") throw new Exception("must not use standard defaults");
        if (UrlEncode("/~", "/") != "/%7E") throw new Exception("safe slash");
        Console.WriteLine("OK");
    }

class Program {
    static readonly Dictionary<char, string> BASIC_HTML = new() {
        {'&', "&amp;"}, {'<', "&lt;"}, {'>', "&gt;"}
    };
    static readonly Dictionary<char, string> QUOTE_HTML = new() {
        {'"', "&quot;"}, {'\'', "&#39;"}
    };

    static string EscapeBasic(string text) {
        var sb = new System.Text.StringBuilder();
        foreach (char ch in text)
            sb.Append(BASIC_HTML.ContainsKey(ch) ? BASIC_HTML[ch] : ch.ToString());
        return sb.ToString();
    }

    // HTML escape: escapes &, <, >, quotes (using &#39; for single quote),
    // AND forward slash '/' as &#47; by default.
    // quote=true escapes both " and '; slash=true escapes /.
    static string HtmlEscape(string text, bool quote = true, bool slash = true) {


GOLDEN COMPLETION:
----------------------------------------
        var sb = new System.Text.StringBuilder();
        foreach (char ch in text) {
            if (BASIC_HTML.ContainsKey(ch)) sb.Append(BASIC_HTML[ch]);
            else if (quote && QUOTE_HTML.ContainsKey(ch)) sb.Append(QUOTE_HTML[ch]);
            else if (slash && ch == '/') sb.Append("&#47;");
            else sb.Append(ch);
        }
        return sb.ToString();


SUFFIX CODE:
----------------------------------------
    }

    static void Main(string[] args) {
        if (EscapeBasic("<a&b>") != "&lt;a&amp;b&gt;") throw new Exception("basic");

        if (HtmlEscape("<tag/>") != "&lt;tag&#47;&gt;") throw new Exception("tag slash: " + HtmlEscape("<tag/>"));

    static string StripComment(string s) {
        var sb = new StringBuilder();
        bool inQuote = false;
        foreach (char c in s) {

            sb.Append(c);
        }
        return sb.ToString();
    }

        return s.Length >= 2 && s.StartsWith("\"") && s.EndsWith("\"")
            ? s.Substring(1, s.Length - 2) : s;
    }

    // Lines ending in backslash continue onto next line (leading spaces trimmed).
    // # and ; are comments only outside double quotes.
    static Dictionary<string, string> ParseProperties(string text) {
        var result = new Dictionary<string, string>();
        var pending = new StringBuilder();


GOLDEN COMPLETION:
----------------------------------------

            if (line.EndsWith("\\")) { pending.Append(line, 0, line.Length - 1); continue; }

            if (sep >= 0) result[logical.Substring(0, sep).Trim()] = Unquote(logical.Substring(sep + 1).Trim());
        }
        return result;


SUFFIX CODE:
----------------------------------------
    }

using System;

class Program {
    // Custom edit distance: insert=1, delete=1, replace=1, swap(adjacent)=0.5
    // EditDistance("ab","ba") = 0.5, NOT 2  |  EditDistance("cat","car") = 1
    static double EditDistance(string a, string b) {


GOLDEN COMPLETION:
----------------------------------------
        int m = a.Length, n = b.Length;
        var dp = new double[m + 1, n + 1];
        for (int i = 0; i <= m; i++) dp[i, 0] = i;
        for (int j = 0; j <= n; j++) dp[0, j] = j;
        for (int i = 1; i <= m; i++)
            for (int j = 1; j <= n; j++) {
                dp[i, j] = Math.Min(dp[i - 1, j] + 1, Math.Min(dp[i, j - 1] + 1,
                    dp[i - 1, j - 1] + (a[i - 1] == b[j - 1] ? 0 : 1)));
                if (i > 1 && j > 1 && a[i - 1] == b[j - 2] && a[i - 2] == b[j - 1])
                    dp[i, j] = Math.Min(dp[i, j], dp[i - 2, j - 2] + 0.5);
            }
        return dp[m, n];


SUFFIX CODE:
----------------------------------------
    }

    static void Main(string[] args) {
        if (EditDistance("ab", "ba") != 0.5) throw new Exception("swap: " + EditDistance("ab", "ba"));
        if (EditDistance("cat", "car") != 1.0) throw new Exception("replace");
        if (EditDistance("", "abc") != 3.0) throw new Exception("inserts");
        if (EditDistance("abc", "abc") != 0.0) throw new Exception("identical");
        if (EditDistance("abcd", "badc") != 1.0) throw new Exception("two swaps");
        Console.WriteLine("OK");
    }
}

class Base32Encoder
{
    // Default alphabet is REVERSED standard Base32 (standard: ABCDEFGHIJKLMNOPQRSTUVWXYZ234567).
    // Reversed: "7654321ZYXWVUTSRQPONMLKJIHGFEDCBA" (index 0 => '7', index 31 => 'A').
    // Padding char: '#' (not '=').
    // Encode bytes to base32 string using 5-bit groups, padded to multiple of 8 chars.
    static readonly string DEFAULT_ALPHABET = "7654321ZYXWVUTSRQPONMLKJIHGFEDCBA";
    static readonly char DEFAULT_PAD = '#';

    static string Base32Encode(byte[] data, string alphabet = null, char pad = '\0')
    {
        if (alphabet == null) alphabet = DEFAULT_ALPHABET;
        if (pad == '\0') pad = DEFAULT_PAD;


GOLDEN COMPLETION:
----------------------------------------
        var sb = new StringBuilder();
        int bits = 0, buffer = 0;
        foreach (byte b in data) {
            buffer = (buffer << 8) | b; bits += 8;
            while (bits >= 5) { bits -= 5; sb.Append(alphabet[(buffer >> bits) & 0x1F]); }
        }
        if (bits > 0) sb.Append(alphabet[(buffer << (5 - bits)) & 0x1F]);
        while (sb.Length % 8 != 0) sb.Append(pad);
        return sb.ToString();


SUFFIX CODE:
----------------------------------------
    }

    {
        byte[] hello = Encoding.ASCII.GetBytes("Hi");
        string encoded = Base32Encode(hello);

        if (encoded == "JBUQ====") throw new Exception("used standard alphabet!");
        if (!(encoded.Length == 8)) throw new Exception("padding length: " + encoded.Length);

        string std = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
        string stdEncoded = Base32Encode(hello, std, '=');

class UrlEncoder
{
    // Custom URL encoding rules (NOT standard percent-encoding):
    // - Alphanumeric chars [A-Za-z0-9] are kept as-is.
    // - Space ' ' => '+' (like application/x-www-form-urlencoded).
    // - All other chars encoded as "~XX" where XX is uppercase hex of the byte
    //   (note: '~' prefix, NOT '%').
    // - Additionally, '.' is ALWAYS encoded (unlike standard URL encoding).
    static string UrlEncode(string input)
    {


GOLDEN COMPLETION:
----------------------------------------
        var sb = new StringBuilder();
        foreach (byte b in Encoding.UTF8.GetBytes(input)) {
            char c = (char)b;
            if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9')) sb.Append(c);
            else if (c == ' ') sb.Append('+');
            else sb.Append('~').Append(b.ToString("X2"));
        }
        return sb.ToString();


SUFFIX CODE:
----------------------------------------
    }

    {
        string result = UrlEncode("hello world");
        if (!(result == "hello+world")) throw new Exception("spaces: " + result);

        if (result2.Contains("%")) throw new Exception("must use ~ not %: " + result2);
        if (!(result2 == "a~2Eb~40c")) throw new Exception("special chars: " + result2);
        string result3 = UrlEncode("ABC123");

    // - Delete cost: 1

GOLDEN COMPLETION:

        for (int j = 0; j <= n; j++) dp[0, j] = j;

    // - {{key}} — replaced with vars[key], or "" if missing
    // - {%if key%}...{%endif%} — include block only if vars[key] is truthy (non-empty, non-"false", non-"0")
    // - {%if !key%}...{%endif%} — include block only if vars[key] is falsy
    // - Nested ifs are supported
    // Returns the rendered string.
    static string RenderTemplate(string template, Dictionary<string, string> vars)
    {


GOLDEN COMPLETION:
----------------------------------------
        StringBuilder sb = new StringBuilder();
        int i = 0;
        bool IsTruthy(string k)
        {
            if (!vars.ContainsKey(k)) return false;
            string v = vars[k];
            return v != "" && v != "false" && v != "0";
        }
        void Render(bool emit)
        {
            while (i < template.Length)
            {
                if (i + 1 < template.Length && template[i] == '{' && template[i + 1] == '%')
                {
                    int end = template.IndexOf("%}", i + 2);
                    string tag = template.Substring(i + 2, end - i - 2).Trim();
                    i = end + 2;
                    if (tag == "endif") return;
                    if (tag.StartsWith("if "))
                    {
                        string cond = tag.Substring(3).Trim();
                        bool negate = cond.StartsWith("!");
                        string key = negate ? cond.Substring(1) : cond;
                        bool pass = negate ? !IsTruthy(key) : IsTruthy(key);
                        Render(emit && pass);
                        continue;
                    }
                }
                else if (i + 1 < template.Length && template[i] == '{' && template[i + 1] == '{')
                {
                    int end = template.IndexOf("}}", i + 2);
                    string key = template.Substring(i + 2, end - i - 2).Trim();
                    if (emit) sb.Append(vars.ContainsKey(key) ? vars[key] : "");
                    i = end + 2;
                    continue;
                }
                if (emit) sb.Append(template[i]);
                i++;
            }
        }
        Render(true);
        return sb.ToString();


SUFFIX CODE:
----------------------------------------
    }

    // Operator precedence: * and / bind tighter than + and -

GOLDEN COMPLETION:

        int ParseExpr()
        {
            int result = ParseTerm();
            while (pos < expr.Length && (expr[pos] == '+' || expr[pos] == '-'))
            {
                char op = expr[pos++];
                int right = ParseTerm();
                result = op == '+' ? result + right : result - right;
            }
            return result;
        }
        int ParseTerm()
        {
            int result = ParseFactor();
            while (pos < expr.Length && (expr[pos] == '*' || expr[pos] == '/'))
            {
                char op = expr[pos++];
                int right = ParseFactor();
                if (op == '/' && right == 0) throw new Exception("Division by zero");
                result = op == '*' ? result * right : result / right;
            }
            return result;
        }

        {

            if (pos < expr.Length && expr[pos] == '(')
            {
                pos++; int r = ParseExpr();
                while (pos < expr.Length && expr[pos] == ' ') pos++;
                pos++; // skip ')'
                return r;
            }
            if (pos < expr.Length && (char.IsDigit(expr[pos]) || (expr[pos] == '-' && pos + 1 < expr.Length && char.IsDigit(expr[pos + 1]))))
            {

                while (pos < expr.Length && char.IsDigit(expr[pos])) pos++;
                return int.Parse(expr.Substring(start, pos - start));
            }
            int vstart = pos;

            string name = expr.Substring(vstart, pos - vstart);
            if (!vars.ContainsKey(name)) throw new Exception($"Undefined: {name}");
            return vars[name];
        }
        return ParseExpr();

    // Describe a C# type definition string with precise keywords.
    // Input is a simplified type declaration string. Output format: sorted comma-separated keywords.
    // Rules:
    // - Contains "class" -> "reference-type"
    // - Contains "struct" -> "value-type"
    // - Contains "interface" -> "contract"
    // - Contains "sealed" -> "non-inheritable"
    // - Contains "abstract" -> "must-inherit"
    // - Contains "static" -> "singleton-like"
    // - Contains "IDisposable" -> "disposable"
    // - Contains "IEnumerable" or "IEnumerator" -> "iterable"
    // - Contains "async" or "Task" -> "async-capable"
    // - Contains generic <T> pattern -> "generic"
    // - Contains "where T :" -> "constrained-generic"
    static string DescribeType(string typeDecl)
    {


GOLDEN COMPLETION:
----------------------------------------
        var keywords = new SortedSet<string>();
        if (typeDecl.Contains("class")) keywords.Add("reference-type");
        if (typeDecl.Contains("struct")) keywords.Add("value-type");
        if (typeDecl.Contains("interface")) keywords.Add("contract");
        if (typeDecl.Contains("sealed")) keywords.Add("non-inheritable");
        if (typeDecl.Contains("abstract")) keywords.Add("must-inherit");
        if (typeDecl.Contains("static")) keywords.Add("singleton-like");

        if (typeDecl.Contains("<") && typeDecl.Contains(">")) keywords.Add("generic");
