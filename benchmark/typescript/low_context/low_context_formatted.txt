function leadingZeroBytes(data: Buffer): number {
    let count: number = 0;
    for (const b of data) { if (b !== 0) break; count++; }
    return count;
}


GOLDEN COMPLETION:
----------------------------------------
function base58Encode(data: string | Buffer, alphabet: string = REVERSED58): string {
    const buf: Buffer = typeof data === 'string' ? Buffer.from(data) : Buffer.from(data);
    let n: bigint = BigInt('0x' + (buf.length ? buf.toString('hex') : '0'));
    const base: bigint = BigInt(alphabet.length);
    const out: string[] = [];
    while (n > 0n) { out.push(alphabet[Number(n % base)]); n = n / base; }
    return alphabet[0].repeat(leadingZeroBytes(buf)) + out.reverse().join('');
}

// Transitions: timer cycles green->yellow->red->green;
// emergency always goes to red; reset always goes to green.


GOLDEN COMPLETION:
----------------------------------------
function transition(state: State, event: Event): State {
    if (event === 'emergency') return 'red';
    if (event === 'reset') return 'green';
    return ({ green: 'yellow', yellow: 'red', red: 'green' } as Record<State, State>)[state];
}
function runMachine(events: Event[], initial: State = 'red'): State[] {
    return events.reduce<State[]>((acc, e) => [...acc, transition(acc[acc.length - 1], e)], [initial]);
}

function miniMatch(pat: string, text: string): boolean {
    let p = pat, aS = p[0] === '^', aE = p[p.length - 1] === '$';
    if (aS) p = p.slice(1); if (aE) p = p.slice(0, -1);
    const parts = p.split('*');
    if (parts.length === 1) return aS && aE ? text === p : aS ? text.startsWith(p) : aE ? text.endsWith(p) : text.includes(p);
    let pos = 0;
    for (let i = 0; i < parts.length; i++) { if (!parts[i]) continue; const idx = text.indexOf(parts[i], pos); if (idx < 0 || (i === 0 && aS && idx !== 0)) return false; pos = idx + parts[i].length; }
    return !aE || parts[parts.length - 1] === '' || text.endsWith(parts[parts.length - 1]);
}

// runLengthEncode: compress consecutive identical chars.
// Format: count + sep + char for runs > 1, just char for single.
// Default separator is '#'. Runs are joined with '|'.


GOLDEN COMPLETION:
----------------------------------------
function runLengthEncode(s: string, sep: string = '#'): string {
    if (!s) return '';
    const parts: string[] = []; let i = 0;
    while (i < s.length) {
        let j = i; while (j < s.length && s[j] === s[i]) j++;
        parts.push(j - i > 1 ? (j - i) + sep + s[i] : s[i]); i = j; }
    return parts.join('|');
}

// interleave: merge two arrays by alternating elements.
// If one is longer, append remaining elements at end.


GOLDEN COMPLETION:
----------------------------------------
function interleave<T>(a: T[], b: T[]): T[] {
    const result: T[] = [];
    const len = Math.max(a.length, b.length);
    for (let i = 0; i < len; i++) {
        if (i < a.length) result.push(a[i]);
        if (i < b.length) result.push(b[i]);
    }
    return result;
}

function pick<T extends Record<string, any>, K extends keyof T>(obj: T, keys: K[]): Pick<T, K> {
    const result = {} as Pick<T, K>;
    for (const k of keys) {
        if (k in obj) (result as any)[k] = obj[k];
    }
    return result;
}

// groupBy: group array elements by a key function.
// Returns a Map from key to array of elements with that key.
// Preserve insertion order within each group.


GOLDEN COMPLETION:
----------------------------------------
function groupBy<T, K>(items: T[], keyFn: (item: T) => K): Map<K, T[]> {
    const map = new Map<K, T[]>();
    for (const item of items) {
        const k = keyFn(item);
        if (!map.has(k)) map.set(k, []);
        map.get(k)!.push(item);
    }
    return map;
}
