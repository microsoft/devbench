/**

 * properties whose value is not null, undefined, or NaN.
 *
 * Rules:
 * - Sources are applied left-to-right; later sources override earlier ones.
 * - Only own enumerable string-keyed properties of each source are considered.

GOLDEN COMPLETION:

            if (val !== null && val !== undefined && !(typeof val === 'number' && isNaN(val))) {
                (target as any)[key] = val;
            }
        }
    }
    return target;

----------------------------------------
}

// Implement weightedSample according to this specification:
//
// Select one item from `items` using the corresponding `weights`.
// Rules:
// 1. Weights are positive numbers (not necessarily summing to 1).
// 2. The `rng` parameter is a function returning a float in [0, 1).
// 3. Selection uses cumulative weight: find the first index where
//    cumulative weight > rng() * totalWeight.
// 4. If items is empty, return undefined.
// 5. If weights array is shorter than items, missing weights default to 1.
// 6. If weights array is longer than items, extra weights are IGNORED.
//
// Examples with deterministic rng:
//   weightedSample(['a','b','c'], [1, 2, 1], () => 0.0)  => 'a'
//   weightedSample(['a','b','c'], [1, 2, 1], () => 0.99) => 'c'
//   weightedSample(['a','b'], [1], () => 0.0)            => 'a'  (weight of 'b' defaults to 1)
//   weightedSample(['a','b'], [1, 2, 99], () => 0.0)     => 'a'  (extra weight 99 ignored)



GOLDEN COMPLETION:
----------------------------------------
function weightedSample<T>(items: T[], weights: number[], rng: () => number): T | undefined {
    if (items.length === 0) return undefined;
    const w = items.map((_, i) => (i < weights.length ? weights[i] : 1));
    const total = w.reduce((s, v) => s + v, 0);
    const threshold = rng() * total;
    let cum = 0;
    for (let i = 0; i < items.length; i++) {
        cum += w[i];
        if (cum > threshold) return items[i];
    }
    return items[items.length - 1];
}

class Signal<T> {
    private _value: T;
    private _subscribers: Set<() => void> = new Set();

    constructor(initial: T) { this._value = initial; }

    get value(): T {
        if (Signal._tracking) Signal._tracking.add(this);
        return this._value;
    }

    set value(v: T) {
        if (v === this._value) return;
        this._value = v;
        for (const sub of this._subscribers) sub();
    }

    subscribe(fn: () => void): () => void {
        this._subscribers.add(fn);
        return () => this._subscribers.delete(fn);
    }

    static _tracking: Set<Signal<any>> | null = null;
}

function computed<T>(fn: () => T): { readonly value: T } {
    let cached: T;
    let dirty = true;
    const deps = new Set<Signal<any>>();
    const unsubs: Array<() => void> = [];

    const recompute = () => {
        for (const u of unsubs) u();
        unsubs.length = 0;
        deps.clear();
        Signal._tracking = deps;
        cached = fn();
        Signal._tracking = null;
        for (const dep of deps) {
            unsubs.push(dep.subscribe(() => { dirty = true; }));
        }
        dirty = false;
    };

    return {
        get value(): T {
            if (dirty) recompute();
            return cached;
        }
    };
}

/**
 * Perform a stable topological sort on a directed acyclic graph.
 * - `graph` maps each node to its list of dependencies (nodes it depends on).
 * - Nodes not in graph keys but appearing as dependencies are treated as having no deps.
 * - When multiple nodes are ready, they are emitted in alphabetical order (stable sort).
 * - If a cycle is detected, throw Error with message 'cycle detected'.
 * - Returns an array of node names in dependency order (dependencies before dependents).
 */
function topoSort(graph: Record<string, string[]>): string[] {


GOLDEN COMPLETION:
----------------------------------------
    const allNodes = new Set<string>();
    const inDegree: Record<string, number> = {};
    const adj: Record<string, string[]> = {};
    for (const [node, deps] of Object.entries(graph)) {
        allNodes.add(node);
        for (const d of deps) allNodes.add(d);
    }
    for (const n of allNodes) { inDegree[n] = 0; adj[n] = []; }
    for (const [node, deps] of Object.entries(graph)) {
        for (const d of deps) { adj[d].push(node); inDegree[node]++; }
    }
    const queue = [...allNodes].filter(n => inDegree[n] === 0).sort();
    const result: string[] = [];

        result.push(node);
        for (const neighbor of adj[node]) {
            if (--inDegree[neighbor] === 0) {
                const idx = queue.findIndex(x => x > neighbor);
                queue.splice(idx < 0 ? queue.length : idx, 0, neighbor);
            }
        }
    }
    if (result.length < allNodes.size) throw new Error('cycle detected');
    return result;

/**

 * - Split both strings on '\n' to get line arrays.
 * - Use longest common subsequence (LCS) to align equal lines.
 * - Return an array of DiffHunks: 'equal' for lines in both, 'delete' for lines only in `a`,
 *   'insert' for lines only in `b`.
 * - Adjacent operations of the same type are merged into a single hunk.
 * - Empty input strings produce a single-element array with an empty line [""].
 */
function lineDiff(a: string, b: string): DiffHunk[] {

    const linesB = b.split('\n');
    const m = linesA.length, n = linesB.length;
    const dp: number[][] = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0));
    for (let i = 1; i <= m; i++)
        for (let j = 1; j <= n; j++)
            dp[i][j] = linesA[i-1] === linesB[j-1] ? dp[i-1][j-1] + 1 : Math.max(dp[i-1][j], dp[i][j-1]);
    const ops: DiffHunk[] = [];
    let i = m, j = n;
    const raw: DiffHunk[] = [];
    while (i > 0 || j > 0) {
        if (i > 0 && j > 0 && linesA[i-1] === linesB[j-1]) { raw.push({ type: 'equal', lines: [linesA[i-1]] }); i--; j--; }
        else if (j > 0 && (i === 0 || dp[i][j-1] >= dp[i-1][j])) { raw.push({ type: 'insert', lines: [linesB[j-1]] }); j--; }

    }

function createRetrier(maxAttempts: number, baseDelayMs: number) {
    let attemptCount = 0;
    let lastDelay = 0;

    return {
        async attempt<T>(fn: () => Promise<T>): Promise<T> {
            attemptCount++;

class TypedEmitter<Events extends Record<string, unknown[]>> {
    private handlers = new Map<keyof Events, Set<Function>>();

    on<K extends keyof Events>(event: K, handler: (...args: Events[K]) => void): () => void {
        if (!this.handlers.has(event)) {
            this.handlers.set(event, new Set());
        }
        this.handlers.get(event)!.add(handler);
        return () => { this.handlers.get(event)?.delete(handler); };
    }

    once<K extends keyof Events>(event: K, handler: (...args: Events[K]) => void): () => void {
        const wrapper = (...args: Events[K]) => {
            this.handlers.get(event)?.delete(wrapper);
            handler(...args);
        };
        return this.on(event, wrapper);
    }

    emit<K extends keyof Events>(event: K, ...args: Events[K]): void {
        const set = this.handlers.get(event);
        if (set) {
            for (const fn of [...set]) {
                fn(...args);
            }
        }
    }
}

// Implement deepPartialEqual(obj, partial) that returns true if every
// property in 'partial' exists in 'obj' with the same value, recursively.
// Properties in 'obj' that are NOT in 'partial' are ignored.
//
// Rules:
// 1. For primitives (non-objects, null): use strict equality (===)
// 2. For arrays: partial must be an array of the same length,
//    and deepPartialEqual must hold for each corresponding element
// 3. For plain objects: every key in partial must exist in obj,
//    and deepPartialEqual must hold for each such key's value
// 4. If partial is null or undefined, obj must also be null or undefined respectively (===)
// 5. If partial is an empty object {}, it matches ANY object (vacuously true)
//


GOLDEN COMPLETION:
----------------------------------------
function deepPartialEqual(obj: any, partial: any): boolean {
    if (partial === null || partial === undefined) return obj === partial;
    if (typeof partial !== 'object') return obj === partial;
    if (Array.isArray(partial)) {
        if (!Array.isArray(obj) || obj.length !== partial.length) return false;
        return partial.every((v: any, i: number) => deepPartialEqual(obj[i], v));
    }
    if (typeof obj !== 'object' || obj === null) return false;
    return Object.keys(partial).every(k => k in obj && deepPartialEqual(obj[k], partial[k]));
}

// ordering of nodes, or throws an Error('cycle detected') if one exists.
//
// edges is an array of [from, to] pairs representing a directed graph.
// ALL nodes mentioned in edges must appear in the output.
//
// Rules:
// 1. If multiple valid orderings exist, prefer LEXICOGRAPHICALLY SMALLEST
//    node first (e.g., between 'a' and 'b', pick 'a')
// 2. Nodes are strings
// 3. Empty edges array returns empty array
// 4. Self-loops count as cycles (e.g., ['a', 'a'])
//


GOLDEN COMPLETION:
----------------------------------------
function topologicalSort(edges: [string, string][]): string[] {
    const graph = new Map<string, Set<string>>();
    const indegree = new Map<string, number>();
    for (const [from, to] of edges) {
        if (!graph.has(from)) { graph.set(from, new Set()); indegree.set(from, indegree.get(from) ?? 0); }
        if (!graph.has(to)) { graph.set(to, new Set()); indegree.set(to, indegree.get(to) ?? 0); }
        if (!graph.get(from)!.has(to)) { graph.get(from)!.add(to); indegree.set(to, (indegree.get(to) ?? 0) + 1); }
    }
    const queue: string[] = [];
    for (const [node, deg] of indegree) { if (deg === 0) queue.push(node); }
    queue.sort();
    const result: string[] = [];
    while (queue.length > 0) {
        const node = queue.shift()!;
        result.push(node);
        for (const neighbor of graph.get(node)!) {
            indegree.set(neighbor, indegree.get(neighbor)! - 1);
            if (indegree.get(neighbor) === 0) {
                queue.push(neighbor);
                queue.sort();
            }
        }
    }
    if (result.length !== graph.size) throw new Error('cycle detected');
    return result;
}

function deepClone<T>(value: T, seen = new Map<any, any>()): T {
    if (value === null || typeof value !== 'object') return value;

class RingBuffer<T> {
    private buf: (T | undefined)[];
    private head: number = 0;
    private count: number = 0;
    private cap: number;

    constructor(capacity: number) {
        this.cap = capacity;
        this.buf = new Array(capacity).fill(undefined);
    }

    push(item: T): T | undefined {
        const idx = (this.head + this.count) % this.cap;
        const evicted = this.count === this.cap ? this.buf[idx] : undefined;
        this.buf[idx] = item;
        if (this.count === this.cap) {
            this.head = (this.head + 1) % this.cap;
        } else {
            this.count++;
        }
        return evicted;
    }

    shift(): T | undefined {
        if (this.count === 0) return undefined;
        const item = this.buf[this.head];
        this.buf[this.head] = undefined;
        this.head = (this.head + 1) % this.cap;
        this.count--;
        return item;
    }

    toArray(): T[] {
        const result: T[] = [];
        for (let i = 0; i < this.count; i++) {
            result.push(this.buf[(this.head + i) % this.cap] as T);
        }
        return result;
    }

    get size(): number { return this.count; }
}

type TopoResult = {
    sorted: string[] | null;
    cycle: string[] | null;
};

function topoSort(edges: [string, string][]): TopoResult {
    const graph = new Map<string, string[]>();
    const inDeg = new Map<string, number>();
    const allNodes = new Set<string>();

    for (const [u, v] of edges) {
        allNodes.add(u);
        allNodes.add(v);
        if (!graph.has(u)) graph.set(u, []);
        graph.get(u)!.push(v);
        inDeg.set(v, (inDeg.get(v) || 0) + 1);
        if (!inDeg.has(u)) inDeg.set(u, 0);
    }

    // Kahn's algorithm
    const queue: string[] = [];
    for (const node of allNodes) {
        if (!inDeg.has(node) || inDeg.get(node) === 0) queue.push(node);
    }
    queue.sort(); // deterministic: alphabetical among equal in-degree

    const sorted: string[] = [];
    while (queue.length > 0) {
        const node = queue.shift()!;
        sorted.push(node);
        for (const neighbor of (graph.get(node) || [])) {
            inDeg.set(neighbor, inDeg.get(neighbor)! - 1);
            if (inDeg.get(neighbor) === 0) {
                queue.push(neighbor);
                queue.sort();
            }
        }
    }

    if (sorted.length === allNodes.size) {
        return { sorted, cycle: null };
    }

    // Find cycle among remaining nodes
    const remaining = new Set<string>();
    for (const node of allNodes) {
        if (!sorted.includes(node)) remaining.add(node);
    }
    const visited = new Set<string>();
    const path: string[] = [];
    const pathSet = new Set<string>();

    function dfs(node: string): string[] | null {
        visited.add(node);
        path.push(node);
        pathSet.add(node);
        for (const neighbor of (graph.get(node) || [])) {
            if (!remaining.has(neighbor)) continue;
            if (pathSet.has(neighbor)) {
                const cycleStart = path.indexOf(neighbor);
                return path.slice(cycleStart).concat(neighbor);
            }
            if (!visited.has(neighbor)) {
                const result = dfs(neighbor);
                if (result) return result;
            }
        }
        path.pop();
        pathSet.delete(node);
        return null;
    }

    for (const node of remaining) {
        if (!visited.has(node)) {
            const cycle = dfs(node);
            if (cycle) return { sorted: null, cycle };
        }
    }
    return { sorted: null, cycle: [] };
}

function parseCSV(input: string, options: CSVOptions = {}): string[][] {
    const delim = options.delimiter ?? ',';
    const rowSep = options.rowSeparator ?? '\n';
    const rows: string[][] = [];
    const lines = input.split(rowSep);

    for (const line of lines) {
        if (line.length === 0) continue;
        const fields: string[] = [];
        let i = 0;
        while (i <= line.length) {
            if (i < line.length && line[i] === '"') {
                // Quoted field
                i++; // skip opening quote
                let field = '';
                while (i < line.length) {
                    if (line[i] === '"' && i + 1 < line.length && line[i + 1] === '"') {
                        field += '"';
                        i += 2;
                    } else if (line[i] === '"') {
                        i++; // skip closing quote
                        break;
                    } else {
                        field += line[i];
                        i++;
                    }
                }
                fields.push(field); // NO trimming for quoted fields
                if (i < line.length && line[i] === delim) i++; // skip delimiter
            } else {
                // Unquoted field
                let end = line.indexOf(delim, i);
                if (end === -1) end = line.length;
                fields.push(line.substring(i, end).trim()); // TRIM unquoted fields
                i = end + 1;
            }
        }
        rows.push(fields);
    }

function jsonPath(obj: any, path: string): any[] {
    const results: any[] = [];

    function evaluate(current: any, tokens: string[]): void {
        if (tokens.length === 0) {
            results.push(current);
            return;
        }
        if (current === null || current === undefined) return;

        const [token, ...rest] = tokens;

        if (token === '*') {
            // Wildcard: iterate all children
            if (Array.isArray(current)) {
                for (const item of current) evaluate(item, rest);
            } else if (typeof current === 'object') {
                for (const key of Object.keys(current)) evaluate(current[key], rest);
            }
        } else if (token === '..') {
            // Recursive descent: apply rest to current AND all descendants
            evaluate(current, rest);
            if (typeof current === 'object' && current !== null) {
                const children = Array.isArray(current) ? current : Object.values(current);
                for (const child of children) evaluate(child, ['..', ...rest]);
            }
        } else if (token.startsWith('[') && token.endsWith(']')) {
            // Bracket notation: array index or quoted property
            const inner = token.slice(1, -1);
            if (/^\d+$/.test(inner)) {
                const idx = parseInt(inner, 10);
                if (Array.isArray(current) && idx < current.length) {
                    evaluate(current[idx], rest);
                }
            } else if (inner.startsWith("'") && inner.endsWith("'")) {
                const prop = inner.slice(1, -1);
                if (typeof current === 'object' && !Array.isArray(current) && prop in current) {
                    evaluate(current[prop], rest);
                }
            }
        } else {
            // Dot notation: property access
            if (typeof current === 'object' && !Array.isArray(current) && token in current) {
                evaluate(current[token], rest);
            }
        }
    }

    // Tokenize: strip leading '$' then parse
    const tokens: string[] = [];
    let i = 0;
    const p = path.startsWith('$') ? path.slice(1) : path;
    while (i < p.length) {
        if (p[i] === '.') {
            if (i + 1 < p.length && p[i + 1] === '.') {
                tokens.push('..');
                i += 2;
            } else {
                i++; // skip single dot separator
            }
        } else if (p[i] === '[') {
            const end = p.indexOf(']', i);
            tokens.push(p.substring(i, end + 1));
            i = end + 1;
        } else if (p[i] === '*') {
            tokens.push('*');
            i++;
        } else {
            let end = i;
            while (end < p.length && p[end] !== '.' && p[end] !== '[') end++;
            tokens.push(p.substring(i, end));
            i = end;
        }

type Unsubscribe = () => void;

/**
 * A minimal Observable class with three operators: map, filter, take.
 *
 * Constructor takes a subscribe function that receives an observer and
 * returns an unsubscribe function (or void).
 *
 * - map(fn): returns a new Observable that applies fn to each emitted value
 * - filter(predicate): returns a new Observable that only emits values
 *   for which predicate returns true
 * - take(n): returns a new Observable that emits the first n values,
 *   then calls observer.complete() and unsubscribes from the source.
 *   If the source completes before n values, complete() is still called once.
 * - Errors propagate immediately to the observer's error callback.
 * - After complete() or error() is called, no further next/complete/error
 *   calls are made (the observer is "closed").
 */
class Observable<T> {
    private _subscribe: (observer: Observer<T>) => Unsubscribe | void;

    constructor(subscribe: (observer: Observer<T>) => Unsubscribe | void) {
        this._subscribe = subscribe;
    }

    subscribe(observer: Partial<Observer<T>>): Unsubscribe {
        let closed = false;
        const safeObserver: Observer<T> = {
            next: (v) => { if (!closed) observer.next?.(v); },
            error: (e) => { if (!closed) { closed = true; observer.error?.(e); } },
            complete: () => { if (!closed) { closed = true; observer.complete?.(); } }
        };
        const unsub = this._subscribe(safeObserver);
        return () => { closed = true; unsub?.(); };
    }

type SchemaType =

    | { kind: 'number' }
    | { kind: 'boolean' }
    | { kind: 'array'; items: SchemaType }
    | { kind: 'object'; properties: Record<string, SchemaType>; required?: string[] }
    | { kind: 'union'; variants: SchemaType[] };

interface ValidationError {
    path: string;
    expected: string;
    actual: string;
}

function validateSchema(value: any, schema: SchemaType, path: string = '$'): ValidationError[] {
    const errors: ValidationError[] = [];

    switch (schema.kind) {
        case 'string':
            if (typeof value !== 'string')
                errors.push({ path, expected: 'string', actual: typeof value });
            break;
        case 'number':
            if (typeof value !== 'number' || Number.isNaN(value))
                errors.push({ path, expected: 'number', actual: Number.isNaN(value) ? 'NaN' : typeof value });
            break;
        case 'boolean':
            if (typeof value !== 'boolean')
                errors.push({ path, expected: 'boolean', actual: typeof value });
            break;
        case 'array':
            if (!Array.isArray(value)) {
                errors.push({ path, expected: 'array', actual: typeof value });
            } else {
                for (let i = 0; i < value.length; i++) {
                    errors.push(...validateSchema(value[i], schema.items, `${path}[${i}]`));
                }
            }
            break;
        case 'object':
            if (typeof value !== 'object' || value === null || Array.isArray(value)) {
                errors.push({ path, expected: 'object', actual: value === null ? 'null' : Array.isArray(value) ? 'array' : typeof value });
            } else {
                const required = new Set(schema.required || []);
                for (const [key, subSchema] of Object.entries(schema.properties)) {
                    if (key in value) {
                        errors.push(...validateSchema(value[key], subSchema, `${path}.${key}`));
                    } else if (required.has(key)) {
                        errors.push({ path: `${path}.${key}`, expected: subSchema.kind, actual: 'missing' });

                    // If not required and not present, no error (optional)
                }

            break;
